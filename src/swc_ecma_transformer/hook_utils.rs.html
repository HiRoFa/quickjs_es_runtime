6macro_rules! chained_method {
7    ($enter_name:ident, $exit_name:ident, $T:ty) => {
8        #[inline]
9        fn $enter_name(&mut self, node: &mut $T, ctx: &mut TraverseCtx) {
10            self.0.$enter_name(node, ctx);
11            self.1.$enter_name(node, ctx);
12        }
13
14        #[inline]
15        fn $exit_name(&mut self, node: &mut $T, ctx: &mut TraverseCtx) {
16            self.0.$exit_name(node, ctx);
17            self.1.$exit_name(node, ctx);
18        }
19    };
20}

141pub(crate) struct HookBuilder<H>
142where
143    H: VisitMutHook<TraverseCtx>,
144{
145    hook: H,
146}
147
148impl<H> HookBuilder<H>
149where
150    H: VisitMutHook<TraverseCtx>,
151{
152    pub fn new(hook: H) -> Self {
153        Self { hook }
154    }
155
156    pub fn chain<B>(self, hook: B) -> HookBuilder<impl VisitMutHook<TraverseCtx>>
157    where
158        B: VisitMutHook<TraverseCtx>,
159    {
160        HookBuilder {
161            hook: OrderedChain(self.hook, hook),
162        }
163    }
164
165    pub fn chain_optional<B>(self, hook: Option<B>) -> HookBuilder<impl VisitMutHook<TraverseCtx>>
166    where
167        B: VisitMutHook<TraverseCtx>,
168    {
169        self.chain(hook)
170    }
171
172    pub fn build(self) -> impl VisitMutHook<TraverseCtx> {
173        self.hook
174    }
