1//! ES2015: Duplicate Keys
2//!
3//! This plugin transforms duplicate keys in object literals to computed
4//! properties.
5//!
6//! > This plugin is included in `preset-env`, in ES2015
7//!
8//! ## Example
9//!
10//! Input:
11//! ```js
12//! var x = { a: 5, a: 6 };
13//! var y = {
14//!   get a() {},
15//!   set a(x) {},
16//!   a: 3
17//! };
18//! ```
19//!
20//! Output:
21//! ```js
22//! var x = { a: 5, ["a"]: 6 };
23//! var y = {
24//!   get a() {},
25//!   set a(x) {},
26//!   ["a"]: 3
27//! };
28//! ```
29//!
30//! ## Implementation
31//!
32//! Implementation based on [@babel/plugin-transform-duplicate-keys](https://babel.dev/docs/babel-plugin-transform-duplicate-keys).
33
34use rustc_hash::FxHashSet;
35use swc_atoms::{Atom, Wtf8Atom};
36use swc_common::Spanned;
37use swc_ecma_ast::*;
38use swc_ecma_hooks::VisitMutHook;
39use swc_ecma_utils::quote_str;
40
41use crate::TraverseCtx;
42
43pub fn hook() -> impl VisitMutHook<TraverseCtx> {
44    DuplicateKeysPass
45}
46
47struct DuplicateKeysPass;
48
49impl VisitMutHook<TraverseCtx> for DuplicateKeysPass {
50    fn exit_expr(&mut self, expr: &mut Expr, _ctx: &mut TraverseCtx) {
51        if let Expr::Object(ObjectLit { props, .. }) = expr {
52            let mut folder = PropFolder::default();
53            folder.visit_props(props);
54        }
55    }
56}
57
58#[derive(Default)]
59struct PropFolder {
60    getter_props: FxHashSet<Atom>,
61    setter_props: FxHashSet<Atom>,
62}
63
64#[inline]
65fn atom_from_wtf8(value: &Wtf8Atom) -> Atom {
66    value
67        .as_str()
68        .map(Atom::from)
69        .unwrap_or_else(|| Atom::from(value.to_string_lossy()))
70}
71
72impl PropFolder {
73    fn visit_props(&mut self, props: &mut [PropOrSpread]) {
74        for prop_or_spread in props.iter_mut() {
75            if let PropOrSpread::Prop(prop) = prop_or_spread {
76                self.visit_prop(prop);
77            }
78        }
79    }
80
81    fn visit_prop(&mut self, prop: &mut Prop) {
82        match prop {
83            Prop::Shorthand(ident) => {
84                if !self.getter_props.insert(ident.sym.clone())
85                    || !self.setter_props.insert(ident.sym.clone())
86                {
87                    *prop = Prop::KeyValue(KeyValueProp {
88                        key: PropName::Computed(ComputedPropName {
89                            span: ident.span,
90                            expr: quote_str!(ident.sym.clone()).into(),
91                        }),
92                        value: ident.clone().into(),
93                    })
94                }
95            }
96
97            Prop::Assign(..) => {}
98
99            Prop::Getter(GetterProp { key, .. }) => {
100                self.visit_prop_name(key, PropKind::Getter);
101            }
102            Prop::Setter(SetterProp { key, .. }) => {
103                self.visit_prop_name(key, PropKind::Setter);
104            }
105            _ => {
106                self.visit_prop_name_for_key_value_or_method(prop);
107            }
108        }
109    }
110
111    fn visit_prop_name_for_key_value_or_method(&mut self, prop: &mut Prop) {
112        match prop {
113            Prop::KeyValue(KeyValueProp { key, .. }) | Prop::Method(MethodProp { key, .. }) => {
114                // Check in both getter and setter props
115                let mut should_transform = false;
116                let span = key.span();
117
118                match key {
119                    PropName::Ident(ident) => {
120                        if !self.getter_props.insert(ident.sym.clone()) {
121                            should_transform = true;
122                        }
123                        if !self.setter_props.insert(ident.sym.clone()) {
124                            should_transform = true;
125                        }
126                        if should_transform {
127                            *key = PropName::Computed(ComputedPropName {
128                                span,
129                                expr: Lit::Str(Str {
130                                    span,
131                                    raw: None,
132                                    value: ident.sym.clone().into(),
133                                })
134                                .into(),
135                            });
136                        }
137                    }
138                    PropName::Str(s) => {
139                        let atom = atom_from_wtf8(&s.value);
140                        if !self.getter_props.insert(atom.clone()) {
141                            should_transform = true;
142                        }
143                        if !self.setter_props.insert(atom) {
144                            should_transform = true;
145                        }
146                        if should_transform {
147                            *key = PropName::Computed(ComputedPropName {
148                                span: s.span,
149                                expr: s.clone().into(),
150                            });
151                        }
152                    }
153                    PropName::Computed(ComputedPropName { expr, .. }) => {
154                        // Computed property might collide
155                        if let Expr::Lit(Lit::Str(Str { ref value, .. })) = &**expr {
156                            let atom = atom_from_wtf8(value);
157                            self.getter_props.insert(atom.clone());
158                            self.setter_props.insert(atom);
159                        }
160                    }
161                    _ => {}
162                }
163            }
164            _ => {}
165        }
166    }
167
168    fn visit_prop_name(&mut self, name: &mut PropName, kind: PropKind) {
169        let props = match kind {
170            PropKind::Getter => &mut self.getter_props,
171            PropKind::Setter => &mut self.setter_props,
172        };
173
174        let span = name.span();
175
176        match name {
177            PropName::Ident(ident) if !props.insert(ident.sym.clone()) => {
178                *name = PropName::Computed(ComputedPropName {
179                    span,
180                    expr: Lit::Str(Str {
181                        span,
182                        raw: None,
183                        value: ident.sym.clone().into(),
184                    })
185                    .into(),
186                })
187            }
188            PropName::Str(s) if !props.insert(atom_from_wtf8(&s.value)) => {
189                *name = PropName::Computed(ComputedPropName {
190                    span: s.span,
191                    expr: s.clone().into(),
192                })
193            }
194            PropName::Ident(_) | PropName::Str(_) => {}
195            PropName::Computed(ComputedPropName { expr, .. }) => {
196                // Computed property might collide
197                if let Expr::Lit(Lit::Str(Str { ref value, .. })) = &**expr {
198                    props.insert(atom_from_wtf8(value));
199                }
200            }
201            _ => {}
202        }
203    }
204}
205
206enum PropKind {
207    Getter,
208    Setter,
209}
