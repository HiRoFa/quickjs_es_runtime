1//! ES2016: Exponentiation Operator
2//!
3//! This plugin transforms the exponentiation operator (`**`) to `Math.pow`.
4//!
5//! > This plugin is included in `preset-env`, in ES2016
6//!
7//! ## Example
8//!
9//! Input:
10//! ```js
11//! let x = 10 ** 2;
12//! x **= 3;
13//! obj.prop **= 4;
14//! ```
15//!
16//! Output:
17//! ```js
18//! let x = Math.pow(10, 2);
19//! x = Math.pow(x, 3);
20//! obj["prop"] = Math.pow(obj["prop"], 4);
21//! ```
22//!
23//! ## Implementation
24//!
25//! Implementation based on [@babel/plugin-transform-exponentiation-operator](https://babel.dev/docs/babel-plugin-transform-exponentiation-operator).
26//!
27//! ## References:
28//!
29//! * Babel plugin implementation: <https://github.com/babel/babel/blob/v7.26.2/packages/babel-plugin-transform-exponentiation-operator>
30//!   <https://github.com/babel/babel/tree/v7.26.2/packages/babel-helper-builder-binary-assignment-operator-visitor>
31//! * Exponentiation operator TC39 proposal: <https://github.com/tc39/proposal-exponentiation-operator>
32//! * Exponentiation operator specification: <https://tc39.es/ecma262/#sec-exp-operator>
33

40pub fn hook() -> impl VisitMutHook<TraverseCtx> {
41    ExponentiationOperatorPass
42}

44struct ExponentiationOperatorPass;
45
46impl VisitMutHook<TraverseCtx> for ExponentiationOperatorPass {
47    fn enter_expr(&mut self, expr: &mut Expr, ctx: &mut TraverseCtx) {
48        match expr {
49            // `left ** right` -> `Math.pow(left, right)`
50            Expr::Bin(bin_expr) if bin_expr.op == BinaryOp::Exp => {
51                // Do not transform BigInt
52                if is_bigint_literal(&bin_expr.left) || is_bigint_literal(&bin_expr.right) {
53                    return;
54                }
55
56                let left = bin_expr.left.take();
57                let right = bin_expr.right.take();
58                *expr = create_math_pow(left, right);
59            }
60            // `left **= right` -> various transformations
61            Expr::Assign(assign_expr) if assign_expr.op == AssignOp::ExpAssign => {
62                // Do not transform BigInt
63                if is_bigint_literal(&assign_expr.right) {
64                    return;
65                }
66
67                self.transform_exp_assign(expr, ctx);
68            }
69            _ => {}
70        }
71    }
72}
73
74impl ExponentiationOperatorPass {
75    /// Transform exponentiation assignment expression
76    ///
77    /// Transforms based on the type of left side:
78    /// - Identifier: `x **= 2` -> `x = Math.pow(x, 2)`
79    /// - Member expression: `obj.prop **= 2` -> `obj.prop = Math.pow(obj.prop,
80    ///   2)` or with temp var
81    fn transform_exp_assign(&self, expr: &mut Expr, ctx: &mut TraverseCtx) {
82        let Expr::Assign(assign_expr) = expr else {
83            return;
84        };
85
86        match &mut assign_expr.left {
87            // Simple identifier: `x **= 2` -> `x = Math.pow(x, 2)`
88            AssignTarget::Simple(SimpleAssignTarget::Ident(ident)) => {
89                let left = ident.id.clone();
90                let right = assign_expr.right.take();
91
92                let pow_left = Box::new(Expr::Ident(left.clone()));
93                let pow_call = create_math_pow(pow_left, right);
94
95                *expr = Expr::Assign(AssignExpr {
96                    span: DUMMY_SP,
97                    op: AssignOp::Assign,
98                    left: AssignTarget::Simple(SimpleAssignTarget::Ident(BindingIdent {
99                        id: left,
100                        type_ann: None,
101                    })),
102                    right: Box::new(pow_call),
103                });
104            }
105            // Member expression: needs special handling to avoid side effects
106            AssignTarget::Simple(SimpleAssignTarget::Member(member_expr)) => {
107                // Clone the data we need before borrowing mutably
108                let obj = member_expr.obj.take();
109                let prop = member_expr.prop.take();
110                let right = assign_expr.right.take();
111
112                self.transform_member_exp_assign_impl(expr, ctx, obj, prop, right);
113            }
114            _ => {}
115        }
116    }
117
118    /// Transform member expression exponentiation assignment implementation
119    ///
120    /// Uses temporary variables to avoid side effects:
121    /// - `obj.prop **= 2` -> `(_obj = obj, _obj.prop = Math.pow(_obj.prop, 2))`
122    /// - `obj[prop] **= 2` -> `(_obj = obj, _prop = prop, _obj[_prop] =
123    ///   Math.pow(_obj[_prop], 2))`
124    fn transform_member_exp_assign_impl(
125        &self,
126        expr: &mut Expr,
127        ctx: &mut TraverseCtx,
128        obj: Box<Expr>,
129        prop: MemberProp,
130        right: Box<Expr>,
131    ) {
132        // Check if we need a temp variable for the object
133        let needs_obj_temp = !matches!(*obj, Expr::Ident(_));
134
135        let mut sequence_exprs = Vec::new();
136
137        // Create temp variable for object if needed
138        let (final_obj, obj_for_pow) = if needs_obj_temp {
139            let temp_name = utils::generate_temp_var_name(&obj);
140            let temp_ident = utils::create_private_ident(&temp_name);
141
142            // Add variable declaration
143            ctx.var_declarations
144                .insert_var(temp_ident.clone().into(), None);
145
146            // Add assignment to sequence: _obj = obj
147            sequence_exprs.push(Expr::Assign(AssignExpr {
148                span: DUMMY_SP,
149                op: AssignOp::Assign,
150                left: AssignTarget::Simple(SimpleAssignTarget::Ident(BindingIdent {
151                    id: temp_ident.clone(),
152                    type_ann: None,
153                })),
154                right: obj,
155            }));
156
157            let obj_expr = Box::new(Expr::Ident(temp_ident.clone()));
158            (obj_expr.clone(), obj_expr)
159        } else {
160            (obj.clone(), obj)
161        };
162
163        match prop {
164            // obj.prop **= right -> (_obj = obj, _obj.prop = Math.pow(_obj.prop, right))
165            MemberProp::Ident(prop_name) => {
166                // Create Math.pow(obj.prop, right)
167                let pow_left = Box::new(Expr::Member(MemberExpr {
168                    span: DUMMY_SP,
169                    obj: obj_for_pow,
170                    prop: MemberProp::Ident(prop_name.clone()),
171                }));
172                let pow_call = create_math_pow(pow_left, right);
173
174                // obj.prop = Math.pow(obj.prop, right)
175                let assignment = Expr::Assign(AssignExpr {
176                    span: DUMMY_SP,
177                    op: AssignOp::Assign,
178                    left: AssignTarget::Simple(SimpleAssignTarget::Member(MemberExpr {
179                        span: DUMMY_SP,
180                        obj: final_obj,
181                        prop: MemberProp::Ident(prop_name),
182                    })),
183                    right: Box::new(pow_call),
184                });
185
186                if needs_obj_temp {
187                    sequence_exprs.push(assignment);
188                    *expr = Expr::Seq(SeqExpr {
189                        span: DUMMY_SP,
190                        exprs: sequence_exprs.into_iter().map(Box::new).collect(),
191                    });
192                } else {
193                    *expr = assignment;
194                }
195            }
196            // obj[computed] **= right -> (_obj = obj, _prop = prop, _obj[_prop] =
197            // Math.pow(_obj[_prop], right))
198            MemberProp::Computed(computed) => {
199                let computed_prop = computed.expr;
200
201                // Check if we need a temp variable for the computed property
202                let needs_prop_temp = !is_literal_expr(&computed_prop);
203
204                let (final_prop, prop_for_pow) = if needs_prop_temp {
205                    let temp_name = match &*computed_prop {
206                        Expr::Ident(ident) => format!("_{}", ident.sym),
207                        _ => "_prop".to_string(),
208                    };
209                    let temp_ident = utils::create_private_ident(&temp_name);
210
211                    // Add variable declaration
212                    ctx.var_declarations
213                        .insert_var(temp_ident.clone().into(), None);
214
215                    // Add assignment to sequence: _prop = prop
216                    sequence_exprs.push(Expr::Assign(AssignExpr {
217                        span: DUMMY_SP,
218                        op: AssignOp::Assign,
219                        left: AssignTarget::Simple(SimpleAssignTarget::Ident(BindingIdent {
220                            id: temp_ident.clone(),
221                            type_ann: None,
222                        })),
223                        right: computed_prop,
224                    }));
225
226                    let prop_expr = Box::new(Expr::Ident(temp_ident.clone()));
227                    (prop_expr.clone(), prop_expr)
228                } else {
229                    (computed_prop.clone(), computed_prop)
230                };
231
232                // Create Math.pow(obj[computed], right)
233                let pow_left = Box::new(Expr::Member(MemberExpr {
234                    span: DUMMY_SP,
235                    obj: obj_for_pow,
236                    prop: MemberProp::Computed(ComputedPropName {
237                        span: DUMMY_SP,
238                        expr: prop_for_pow,
239                    }),
240                }));
241                let pow_call = create_math_pow(pow_left, right);
242
243                // obj[computed] = Math.pow(obj[computed], right)
244                let assignment = Expr::Assign(AssignExpr {
245                    span: DUMMY_SP,
246                    op: AssignOp::Assign,
247                    left: AssignTarget::Simple(SimpleAssignTarget::Member(MemberExpr {
248                        span: DUMMY_SP,
249                        obj: final_obj,
250                        prop: MemberProp::Computed(ComputedPropName {
251                            span: DUMMY_SP,
252                            expr: final_prop,
253                        }),
254                    })),
255                    right: Box::new(pow_call),
256                });
257
258                if needs_obj_temp || needs_prop_temp {
259                    sequence_exprs.push(assignment);
260                    *expr = Expr::Seq(SeqExpr {
261                        span: DUMMY_SP,
262                        exprs: sequence_exprs.into_iter().map(Box::new).collect(),
263                    });
264                } else {
265                    *expr = assignment;
266                }
267            }
268            _ => {}
269        }
270    }
271}

273/// Check if an expression is a literal (and safe to evaluate multiple times)
274fn is_literal_expr(expr: &Expr) -> bool {
275    matches!(
276        expr,
277        Expr::Lit(_) | Expr::Ident(_) | Expr::This(_) | Expr::PrivateName(_)
278    )
279}

281/// Create `Math.pow(left, right)` call expression
282fn create_math_pow(left: Box<Expr>, right: Box<Expr>) -> Expr {
283    Expr::Call(CallExpr {
284        span: DUMMY_SP,
285        ctxt: SyntaxContext::empty(),
286        callee: Callee::Expr(Box::new(Expr::Member(MemberExpr {
287            span: DUMMY_SP,
288            obj: Box::new(Expr::Ident(Ident {
289                span: DUMMY_SP,
290                ctxt: SyntaxContext::empty(),
291                sym: "Math".into(),
292                optional: false,
293            })),
294            prop: MemberProp::Ident(IdentName {
295                span: DUMMY_SP,
296                sym: "pow".into(),
297            }),
298        }))),
299        args: vec![
300            ExprOrSpread {
301                spread: None,
302                expr: left,
303            },
304            ExprOrSpread {
305                spread: None,
306                expr: right,
307            },
308        ],
309        type_args: None,
310    })
311}

313/// Check if an expression is a BigInt literal
314fn is_bigint_literal(expr: &Expr) -> bool {
315    matches!(expr, Expr::Lit(Lit::BigInt(_)))
