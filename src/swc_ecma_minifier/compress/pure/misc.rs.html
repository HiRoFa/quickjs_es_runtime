26fn is_definitely_string(expr: &Expr) -> bool {
27    match expr {
28        Expr::Lit(Lit::Str(_)) => true,
29        Expr::Tpl(_) => true,
30        Expr::Bin(BinExpr {
31            op: BinaryOp::Add,
32            left,
33            right,
34            ..
35        }) => is_definitely_string(left) || is_definitely_string(right),
36        Expr::Paren(ParenExpr { expr, .. }) => is_definitely_string(expr),
37        _ => false,
38    }
39}

41/// Check whether we can compress `new RegExp(…)` to `RegExp(…)`. That's sound
42/// unless the first argument is already a RegExp object and the second is
43/// undefined. We check for the common case where we can prove one of the first
44/// two arguments is a string.
45fn can_compress_new_regexp(args: Option<&[ExprOrSpread]>) -> bool {
46    if let Some(args) = args {
47        if let Some(first) = args.first() {
48            if first.spread.is_some() {
49                false
50            } else if is_definitely_string(&first.expr) {
51                true
52            } else if let Some(second) = args.get(1) {
53                second.spread.is_none() && is_definitely_string(&second.expr)
54            } else {
55                false
56            }
57        } else {
58            true
59        }
60    } else {
61        true
62    }
63}

65fn collect_exprs_from_object(obj: &mut ObjectLit) -> Vec<Box<Expr>> {
66    let mut exprs = Vec::new();
67
68    for prop in obj.props.take() {
69        if let PropOrSpread::Prop(p) = prop {
70            match *p {
71                Prop::Shorthand(p) => {
72                    exprs.push(p.into());
73                }
74                Prop::KeyValue(p) => {
75                    if let PropName::Computed(e) = p.key {
76                        exprs.push(e.expr);
77                    }
78
79                    exprs.push(p.value);
80                }
81                Prop::Getter(p) => {
82                    if let PropName::Computed(e) = p.key {
83                        exprs.push(e.expr);
84                    }
85                }
86                Prop::Setter(p) => {
87                    if let PropName::Computed(e) = p.key {
88                        exprs.push(e.expr);
89                    }
90                }
91                Prop::Method(p) => {
92                    if let PropName::Computed(e) = p.key {
93                        exprs.push(e.expr);
94                    }
95                }
96                _ => {}
97            }
98        }
99    }
100
101    exprs
102}

112    /// not to have observable side effects, according to the `pure_getters`

114    fn can_assume_pure_getter(&self, prop: &MemberProp) -> bool {
115        match &self.options.pure_getters {
116            PureGetterOption::Bool(true) => true,
117            // `strict` relaxes nullish checks in terser, not getter effects.
118            PureGetterOption::Bool(false) | PureGetterOption::Strict => false,
119            PureGetterOption::Str(allowed) => match prop {
120                MemberProp::Ident(i) => allowed.contains(&i.sym),
121                // Private names are not supported.
122                MemberProp::PrivateName(..) => false,
123                MemberProp::Computed(c) => match &*c.expr {
124                    Expr::Lit(Lit::Str(s)) => s
125                        .value
126                        .as_str()
127                        .is_some_and(|v| allowed.iter().any(|a| a == v)),
128                    _ => false,
129                },
130                #[cfg(swc_ast_unknown)]
131                _ => false,
132            },
133        }
134    }

144    pub(super) fn compress_bin_assignment_to_left(&mut self, e: &mut AssignExpr) {
145        if e.op != op!("=") {
146            return;
147        }
148
149        // TODO: Handle pure properties.
150        let lhs = match &e.left {
151            AssignTarget::Simple(SimpleAssignTarget::Ident(i)) => i,
152            _ => return,
153        };
154
155        // If left operand of a binary expression is not same as lhs, this method has
156        // nothing to do.
157        let (op, right) = match &mut *e.right {
158            Expr::Bin(BinExpr {
159                left, op, right, ..
160            }) => match &**left {
161                Expr::Ident(r) if lhs.sym == r.sym && lhs.ctxt == r.ctxt => (op, right),
162                _ => return,
163            },
164            _ => return,
165        };
166
167        // Don't break code for old browsers.
168        match op {
169            BinaryOp::LogicalOr => return,
170            BinaryOp::LogicalAnd => return,
171            BinaryOp::Exp => return,
172            BinaryOp::NullishCoalescing => return,
173            _ => {}
174        }
175
176        let op = match op {
177            BinaryOp::In | BinaryOp::InstanceOf => return,
178
179            BinaryOp::EqEq | BinaryOp::NotEq | BinaryOp::EqEqEq | BinaryOp::NotEqEq => {
180                // TODO(kdy1): Check if this is optimizable.
181                return;
182            }
183
184            BinaryOp::Lt | BinaryOp::LtEq | BinaryOp::Gt | BinaryOp::GtEq => return,
185
186            BinaryOp::LShift => op!("<<="),
187            BinaryOp::RShift => {
188                op!(">>=")
189            }
190            BinaryOp::ZeroFillRShift => {
191                op!(">>>=")
192            }
193            BinaryOp::Add => {
194                op!("+=")
195            }
196            BinaryOp::Sub => {
197                op!("-=")
198            }
199            BinaryOp::Mul => {
200                op!("*=")
201            }
202            BinaryOp::Div => {
203                op!("/=")
204            }
205            BinaryOp::Mod => {
206                op!("%=")
207            }
208            BinaryOp::BitOr => {
209                op!("|=")
210            }
211            BinaryOp::BitXor => {
212                op!("^=")
213            }
214            BinaryOp::BitAnd => {
215                op!("&=")
216            }
217            BinaryOp::LogicalOr => {
218                op!("||=")
219            }
220            BinaryOp::LogicalAnd => {
221                op!("&&=")
222            }
223            BinaryOp::Exp => {
224                op!("**=")
225            }
226            BinaryOp::NullishCoalescing => {
227                op!("??=")
228            }
229            #[cfg(swc_ast_unknown)]
230            _ => panic!("unable to access unknown nodes"),
231        };
232
233        e.op = op;
234        e.right = right.take();
235        // Now we can compress it to an assignment
236    }

238    /// This method does
239    ///
240    /// - `x *= 3` => `x = 3 * x`
241    /// - `x = 3 | x` `x |= 3`
242    /// - `x = 3 & x` => `x &= 3;`
243    /// - `x ^= 3` => `x = 3 ^ x`
244    pub(super) fn compress_bin_assignment_to_right(&mut self, e: &mut AssignExpr) {
245        if e.op != op!("=") {
246            return;
247        }
248
249        // TODO: Handle pure properties.
250        let lhs = match &e.left {
251            AssignTarget::Simple(SimpleAssignTarget::Ident(i)) => i,
252            _ => return,
253        };
254
255        let (op, left) = match &mut *e.right {
256            Expr::Bin(BinExpr {
257                left, op, right, ..
258            }) => match &**right {
259                Expr::Ident(r) if lhs.sym == r.sym && lhs.ctxt == r.ctxt => {
260                    // We need this check because a function call like below can change value of
261                    // operand.
262                    //
263                    // x = g() * x;
264
265                    match &**left {
266                        Expr::This(..) | Expr::Ident(..) | Expr::Lit(..) => {}
267                        _ => return,
268                    }
269
270                    (op, left)
271                }
272                _ => return,
273            },
274            _ => return,
275        };
276
277        let op = match op {
278            BinaryOp::Mul => {
279                op!("*=")
280            }
281            BinaryOp::BitOr => {
282                op!("|=")
283            }
284            BinaryOp::BitXor => {
285                op!("^=")
286            }
287            BinaryOp::BitAnd => {
288                op!("&=")
289            }
290            _ => return,
291        };
292
293        report_change!("Compressing: `e = 3 & e` => `e &= 3`");
294
295        self.changed = true;
296        e.op = op;
297        e.right = left.take();
298    }

300    pub(super) fn eval_spread_object(&mut self, e: &mut ObjectLit) {
301        fn is_proto_key(key: &PropName) -> bool {
302            match key {
303                PropName::Ident(i) => &*i.sym == "__proto__",
304                PropName::Str(s) => s.value.as_str() == Some("__proto__"),
305                _ => false,
306            }
307        }
308
309        // Spreading a primitive of these types copies zero own enumerable
310        // properties (`ToObject` on them yields a wrapper whose properties are
311        // all non-enumerable), so `{ ...x }` where `x` is one of them
312        // contributes nothing.
313        //
314        // Strings are intentionally EXCLUDED: they expose indexed own
315        // enumerable properties (`{ ..."ab" }` === `{ 0: "a", 1: "b" }`).
316        // Objects, arrays, functions, etc. are also excluded as they carry
317        // real properties.
318        fn spreads_no_props(expr: &Expr, expr_ctx: ExprCtx) -> bool {
319            matches!(
320                expr.get_type(expr_ctx),
321                Value::Known(Type::Undefined | Type::Null | Type::Bool | Type::Num | Type::Symbol)
322            )
323        }
324
325        fn can_flatten_spread_expr(expr: &Expr, expr_ctx: ExprCtx) -> bool {
326            match expr {
327                Expr::Object(ObjectLit { props, .. }) => {
328                    props.iter().all(|p| can_flatten_spread_prop(p, expr_ctx))
329                }
330                // Spreading a side-effect-free expression that has no own
331                // enumerable properties contributes nothing. We require purity
332                // so that we never discard observable side effects
333                // (e.g. `{ ...void foo() }`).
334                _ => spreads_no_props(expr, expr_ctx) && !expr.may_have_side_effects(expr_ctx),
335            }
336        }
337
338        fn can_flatten_spread_prop(p: &PropOrSpread, expr_ctx: ExprCtx) -> bool {
339            match p {
340                PropOrSpread::Prop(p) => match &**p {
341                    Prop::KeyValue(KeyValueProp { key, value, .. }) => {
342                        !key.is_computed()
343                            && !is_proto_key(key)
344                            && !value.may_have_side_effects(expr_ctx)
345                    }
346                    Prop::Assign(AssignProp { key, value, .. }) => {
347                        key.sym != *"__proto__" && !value.may_have_side_effects(expr_ctx)
348                    }
349                    Prop::Shorthand(i) => i.sym != *"__proto__",
350
351                    // Moving methods between object literals can change their
352                    // [[HomeObject]], which matters for `super`.
353                    Prop::Method(..) | Prop::Getter(..) | Prop::Setter(..) => false,
354
355                    #[cfg(swc_ast_unknown)]
356                    _ => false,
357                },
358
359                PropOrSpread::Spread(SpreadElement { expr, .. }) => {
360                    can_flatten_spread_expr(expr, expr_ctx)
361                }
362                #[cfg(swc_ast_unknown)]
363                _ => panic!("unable to access unknown nodes"),
364            }
365        }
366
367        fn should_skip_target_prop(p: &PropOrSpread, expr_ctx: ExprCtx) -> bool {
368            match p {
369                PropOrSpread::Prop(p) => match &**p {
370                    Prop::KeyValue(KeyValueProp { key, value, .. }) => {
371                        key.is_computed() || value.may_have_side_effects(expr_ctx)
372                    }
373                    Prop::Assign(AssignProp { value, .. }) => value.may_have_side_effects(expr_ctx),
374                    Prop::Method(method) => method.key.is_computed(),
375
376                    Prop::Getter(..) | Prop::Setter(..) => true,
377
378                    _ => false,
379                },
380
381                PropOrSpread::Spread(SpreadElement { expr, .. }) => match &**expr {
382                    Expr::Object(..) => !can_flatten_spread_expr(expr, expr_ctx),
383                    _ => false,
384                },
385                #[cfg(swc_ast_unknown)]
386                _ => panic!("unable to access unknown nodes"),
387            }
388        }
389
390        if e.props
391            .iter()
392            .any(|p| should_skip_target_prop(p, self.expr_ctx))
393            || !e.props.iter().any(|p| match p {
394                PropOrSpread::Spread(SpreadElement { expr, .. }) => {
395                    can_flatten_spread_expr(expr, self.expr_ctx)
396                }
397                _ => false,
398            })
399        {
400            return;
401        }
402
403        let mut new_props = Vec::with_capacity(e.props.len());
404
405        for prop in e.props.take() {
406            match prop {
407                PropOrSpread::Spread(SpreadElement { expr, .. })
408                    if can_flatten_spread_expr(&expr, self.expr_ctx) =>
409                {
410                    // A plain object literal is flattened into the target; any
411                    // other flattenable spread is a no-property primitive
412                    // (including `null`) that contributes nothing, so it is
413                    // simply dropped.
414                    if let Expr::Object(ObjectLit { props, .. }) = *expr {
415                        new_props.extend(props);
416                    }
417                }
418
419                _ => {
420                    new_props.push(prop);
421                }
422            }
423        }
424
425        e.props = new_props;
426        self.changed = true;
427        report_change!("evaluate: Folded object spread from a plain object literal");
428    }

467                },

529    pub(super) fn remove_invalid(&mut self, e: &mut Expr) {
530        match e {
531            Expr::Seq(seq) => {
532                for e in &mut seq.exprs {
533                    self.remove_invalid(e);
534                }
535
536                if seq.exprs.len() == 1 {
537                    *e = *seq.exprs.pop().unwrap();
538                }
539            }
540
541            Expr::Bin(BinExpr { left, right, .. }) => {
542                self.remove_invalid(left);
543                self.remove_invalid(right);
544
545                if left.is_invalid() {
546                    *e = *right.take();
547                    self.remove_invalid(e);
548                } else if right.is_invalid() {
549                    *e = *left.take();
550                    self.remove_invalid(e);
551                }
552            }
553
554            _ => {}
555        }
556    }

558    pub(super) fn compress_array_join(&mut self, e: &mut Expr) {
559        let call = match e {
560            Expr::Call(e) => e,
561            _ => return,
562        };
563
564        let callee = match &mut call.callee {
565            Callee::Super(_) | Callee::Import(_) => return,
566            Callee::Expr(callee) => &mut **callee,
567            #[cfg(swc_ast_unknown)]
568            _ => panic!("unable to access unknown nodes"),
569        };
570
571        let separator = if call.args.is_empty() {
572            Wtf8Atom::from(",")
573        } else if call.args.len() == 1 {
574            if call.args[0].spread.is_some() {
575                return;
576            }
577
578            if is_pure_undefined(self.expr_ctx, &call.args[0].expr) {
579                Wtf8Atom::from(",")
580            } else {
581                match &*call.args[0].expr {
582                    Expr::Lit(Lit::Str(s)) => s.value.clone(),
583                    Expr::Lit(Lit::Null(..)) => Wtf8Atom::from("null"),
584                    _ => return,
585                }
586            }
587        } else {
588            return;
589        };
590
591        let arr = match callee {
592            Expr::Member(MemberExpr {
593                obj,
594                prop: MemberProp::Ident(IdentName { sym, .. }),
595                ..
596            }) if *sym == *"join" => {
597                if let Expr::Array(arr) = &mut **obj {
598                    arr
599                } else {
600                    return;
601                }
602            }
603            _ => return,
604        };
605
606        if arr.elems.iter().any(|elem| {
607            matches!(
608                elem,
609                Some(ExprOrSpread {
610                    spread: Some(..),
611                    ..
612                })
613            )
614        }) {
615            return;
616        }
617
618        // Handle empty array case first
619        if arr.elems.is_empty() {
620            report_change!("Compressing empty array.join()");
621            self.changed = true;
622            *e = Lit::Str(Str {
623                span: call.span,
624                raw: None,
625                value: atom!("").into(),
626            })
627            .into();
628            return;
629        }
630
631        let cannot_join_as_str_lit = arr
632            .elems
633            .iter()
634            .filter_map(|v| v.as_ref())
635            .any(|v| match &*v.expr {
636                e if is_pure_undefined(self.expr_ctx, e) => false,
637                Expr::Lit(lit) => !matches!(lit, Lit::Str(..) | Lit::Num(..) | Lit::Null(..)),
638                _ => true,
639            });
640
641        if cannot_join_as_str_lit {
642            if let Some(new_expr) =
643                self.compress_array_join_as_tpl(arr.span, &mut arr.elems, &separator)
644            {
645                self.changed = true;
646                *e = new_expr;
647                return;
648            }
649
650            // Try partial optimization (grouping consecutive literals)
651            if let Some(new_expr) =
652                self.compress_array_join_partial(arr.span, &mut arr.elems, &separator)
653            {
654                self.changed = true;
655                report_change!("Compressing array.join() with partial optimization");
656                *e = new_expr;
657                return;
658            }
659
660            if !self.options.unsafe_passes {
661                return;
662            }
663
664            if arr
665                .elems
666                .iter()
667                .filter_map(|v| v.as_ref())
668                .any(|v| match &*v.expr {
669                    e if is_pure_undefined(self.expr_ctx, e) => false,
670                    Expr::Lit(lit) => !matches!(lit, Lit::Str(..) | Lit::Num(..) | Lit::Null(..)),
671                    // All other expressions can potentially be null/undefined
672                    _ => true,
673                })
674            {
675                return;
676            }
677
678            let sep: Box<Expr> = Lit::Str(Str {
679                span: DUMMY_SP,
680                raw: None,
681                value: separator,
682            })
683            .into();
684            let mut res = Lit::Str(Str {
685                span: DUMMY_SP,
686                raw: None,
687                value: atom!("").into(),
688            })
689            .into();
690
691            fn add(to: &mut Expr, right: Box<Expr>) {
692                let lhs = to.take();
693                *to = BinExpr {
694                    span: DUMMY_SP,
695                    left: Box::new(lhs),
696                    op: op!(bin, "+"),
697                    right,
698                }
699                .into();
700            }
701
702            for (last, elem) in arr.elems.take().into_iter().identify_last() {
703                if let Some(ExprOrSpread { spread: None, expr }) = elem {
704                    match &*expr {
705                        e if is_pure_undefined(self.expr_ctx, e) => {
706                            // null and undefined should become empty strings in
707                            // join
708                            // Don't add anything for empty string join
709                        }
710                        Expr::Lit(Lit::Null(..)) => {
711                            // null and undefined should become empty strings in
712                            // join
713                            // Don't add anything for empty string join
714                        }
715                        _ => {
716                            add(&mut res, expr);
717                        }
718                    }
719                }
720
721                if !last {
722                    add(&mut res, sep.clone());
723                }
724            }
725
726            *e = res;
727
728            return;
729        }
730
731        let mut res = Wtf8Buf::default();
732        for (last, elem) in arr.elems.iter().identify_last() {
733            if let Some(elem) = elem {
734                debug_assert_eq!(elem.spread, None);
735
736                match &*elem.expr {
737                    Expr::Lit(Lit::Str(s)) => {
738                        res.push_wtf8(&s.value);
739                    }
740                    Expr::Lit(Lit::Num(n)) => {
741                        res.push_str(&n.value.to_js_string());
742                    }
743                    e if is_pure_undefined(self.expr_ctx, e) => {}
744                    Expr::Lit(Lit::Null(..)) => {}
745                    _ => {
746                        unreachable!(
747                            "Expression {:#?} cannot be joined and it should be filtered out",
748                            elem.expr
749                        )
750                    }
751                }
752            }
753
754            if !last {
755                res.push_wtf8(&separator);
756            }
757        }
758
759        report_change!("Compressing array.join()");
760
761        self.changed = true;
762        *e = Lit::Str(Str {
763            span: call.span,
764            raw: None,
765            value: res.into(),
766        })
767        .into()
768    }

770    /// Performs partial optimization on array.join() when there are mixed
771    /// literals and expressions. Groups consecutive literals into string
772    /// concatenations.
773    fn compress_array_join_partial(
774        &mut self,
775        _span: Span,
776        elems: &mut Vec<Option<ExprOrSpread>>,
777        separator: &Wtf8,
778    ) -> Option<Expr> {
779        if !self.options.evaluate {
780            return None;
781        }
782
783        // Check if we have any non-literal elements
784        let has_non_literals = elems.iter().flatten().any(|elem| match &*elem.expr {
785            Expr::Lit(Lit::Str(..) | Lit::Num(..) | Lit::Null(..)) => false,
786            e if is_pure_undefined(self.expr_ctx, e) => false,
787            _ => true,
788        });
789
790        if !has_non_literals {
791            return None; // Pure literal case will be handled elsewhere
792        }
793
794        // For non-empty separators, only optimize if we have at least 2 consecutive
795        // literals This prevents infinite loop and ensures meaningful
796        // optimization
797        if !separator.is_empty() {
798            let mut consecutive_literals = 0;
799            let mut max_consecutive = 0;
800
801            for elem in elems.iter() {
802                let is_literal = match elem {
803                    None => true,
804                    Some(elem) => match &*elem.expr {
805                        Expr::Lit(Lit::Str(..) | Lit::Num(..) | Lit::Null(..)) => true,
806                        e if is_pure_undefined(self.expr_ctx, e) => true,
807                        _ => false,
808                    },
809                };
810
811                if is_literal {
812                    consecutive_literals += 1;
813                    max_consecutive = max_consecutive.max(consecutive_literals);
814                } else {
815                    consecutive_literals = 0;
816                }
817            }
818
819            if max_consecutive < 2 {
820                return None;
821            }
822
823            // Only optimize for single-character separators to avoid bloating the code
824            // Long separators like "really-long-separator" should not be optimized
825            if separator.len() > 1 {
826                return None;
827            }
828
829            // For comma separator, require a higher threshold to avoid infinite loops
830            if separator == "," && max_consecutive < 6 {
831                return None;
832            }
833        } else {
834            // For empty string joins, optimize more aggressively since we're
835            // doing string concatenation We can always optimize
836            // these as long as there are mixed expressions and literals
837        }
838
839        // Group consecutive literals and create a string concatenation expression
840        let mut groups = Vec::new();
841        let mut current_group = Vec::new();
842
843        for elem in elems.iter() {
844            let is_literal = match elem {
845                None => true,
846                Some(elem) => match &*elem.expr {
847                    Expr::Lit(Lit::Str(..) | Lit::Num(..) | Lit::Null(..)) => true,
848                    e if is_pure_undefined(self.expr_ctx, e) => true,
849                    _ => false,
850                },
851            };
852
853            if is_literal {
854                // An elision is an empty join element, so it must stay in the
855                // group to preserve its surrounding separator positions.
856                current_group.push(elem.as_ref());
857            } else {
858                if !current_group.is_empty() {
859                    groups.push(GroupType::Literals(current_group));
860                    current_group = Vec::new();
861                }
862                groups.push(GroupType::Expression(elem.as_ref().unwrap()));
863            }
864        }
865
866        if !current_group.is_empty() {
867            groups.push(GroupType::Literals(current_group));
868        }
869
870        // If we don't have any grouped literals, no optimization possible
871        if groups.iter().all(|g| matches!(g, GroupType::Expression(_))) {
872            return None;
873        }
874
875        // Handle different separators
876        let is_string_concat = separator.is_empty();
877
878        if is_string_concat {
879            if !self.options.unsafe_passes
880                && groups.iter().any(|group| match group {
881                    GroupType::Literals(_) => false,
882                    GroupType::Expression(expr) => matches!(
883                        &expr.expr.get_type(self.expr_ctx,),
884                        Value::Known(Type::Null | Type::Undefined) | Value::Unknown
885                    ),
886                })
887            {
888                return None;
889            }
890
891            // Convert to string concatenation
892            let mut result_parts = Vec::new();
893
894            // Only add empty string prefix when the first element is a non-string
895            // expression that needs coercion to string AND there's no string
896            // literal early enough to provide coercion.
897            let needs_empty_string_prefix = match groups.first() {
898                Some(GroupType::Expression(first_expr)) => {
899                    let first_needs_coercion = match &*first_expr.expr {
900                        Expr::Bin(BinExpr {
901                            op: op!(bin, "+"), ..
902                        }) => {
903                            // `+` is only already a string concatenation when its
904                            // result is proven to be a string. Otherwise adjacent
905                            // join elements could be added numerically first.
906                            first_expr.expr.get_type(self.expr_ctx) != Value::Known(Type::Str)
907                        }
908                        Expr::Lit(Lit::Str(..)) => false,
909                        Expr::Call(..) => true,
910                        _ => true,
911                    };
912
913                    // A following literal provides the string coercion for the
914                    // first element before the next dynamic element is evaluated.
915                    if first_needs_coercion {
916                        !matches!(groups.get(1), Some(GroupType::Literals(_)))
917                    } else {
918                        false
919                    }
920                }
921                _ => false,
922            };
923
924            if needs_empty_string_prefix {
925                result_parts.push(Box::new(Expr::Lit(Lit::Str(Str {
926                    span: DUMMY_SP,
927                    raw: None,
928                    value: atom!("").into(),
929                }))));
930            }
931
932            for group in groups {
933                match group {
934                    GroupType::Literals(literals) => {
935                        let mut joined = Wtf8Buf::new();
936                        for literal in literals.iter() {
937                            let Some(literal) = literal else {
938                                continue;
939                            };
940
941                            match &*literal.expr {
942                                Expr::Lit(Lit::Str(s)) => joined.push_wtf8(&s.value),
943                                Expr::Lit(Lit::Num(n)) => joined.push_str(&n.value.to_js_string()),
944                                Expr::Lit(Lit::Null(..)) => {
945                                    // For string concatenation, null becomes
946                                    // empty string
947                                }
948                                e if is_pure_undefined(self.expr_ctx, e) => {
949                                    // undefined becomes empty string in string
950                                    // context
951                                }
952                                _ => unreachable!(),
953                            }
954                        }
955
956                        result_parts.push(Box::new(Expr::Lit(Lit::Str(Str {
957                            span: DUMMY_SP,
958                            raw: None,
959                            value: joined.into(),
960                        }))));
961                    }
962                    GroupType::Expression(expr) => {
963                        result_parts.push(expr.expr.clone());
964                    }
965                }
966            }
967
968            // Create string concatenation expression
969            if result_parts.len() == 1 {
970                return Some(*result_parts.into_iter().next().unwrap());
971            }
972
973            let mut result = *result_parts.remove(0);
974            for part in result_parts {
975                result = Expr::Bin(BinExpr {
976                    span: DUMMY_SP,
977                    left: Box::new(result),
978                    op: op!(bin, "+"),
979                    right: part,
980                });
981            }
982
983            Some(result)
984        } else {
985            // For non-empty separator, create a more compact array
986            let mut new_elems = Vec::new();
987
988            for group in groups {
989                match group {
990                    GroupType::Literals(literals) => {
991                        let mut joined = Wtf8Buf::new();
992                        for (idx, literal) in literals.iter().enumerate() {
993                            if idx > 0 {
994                                joined.push_wtf8(separator);
995                            }
996
997                            let Some(literal) = literal else {
998                                continue;
999                            };
1000
1001                            match &*literal.expr {
1002                                Expr::Lit(Lit::Str(s)) => joined.push_wtf8(&s.value),
1003                                Expr::Lit(Lit::Num(n)) => joined.push_str(&n.value.to_js_string()),
1004                                Expr::Lit(Lit::Null(..)) => {
1005                                    // null becomes empty string
1006                                }
1007                                e if is_pure_undefined(self.expr_ctx, e) => {
1008                                    // undefined becomes empty string
1009                                }
1010                                _ => unreachable!(),
1011                            }
1012                        }
1013
1014                        new_elems.push(Some(ExprOrSpread {
1015                            spread: None,
1016                            expr: Box::new(Expr::Lit(Lit::Str(Str {
1017                                span: DUMMY_SP,
1018                                raw: None,
1019                                value: joined.into(),
1020                            }))),
1021                        }));
1022                    }
1023                    GroupType::Expression(expr) => {
1024                        new_elems.push(Some(ExprOrSpread {
1025                            spread: None,
1026                            expr: expr.expr.clone(),
1027                        }));
1028                    }
1029                }
1030            }
1031
1032            // Create a new array.join() call with the original separator
1033            let new_array = Expr::Array(ArrayLit {
1034                span: _span,
1035                elems: new_elems,
1036            });
1037
1038            // For comma separator, use .join() without arguments (shorter)
1039            let args = if separator == "," {
1040                vec![]
1041            } else {
1042                vec![ExprOrSpread {
1043                    spread: None,
1044                    expr: Box::new(Expr::Lit(Lit::Str(Str {
1045                        span: DUMMY_SP,
1046                        raw: None,
1047                        value: separator.into(),
1048                    }))),
1049                }]
1050            };
1051
1052            Some(Expr::Call(CallExpr {
1053                span: _span,
1054                ctxt: Default::default(),
1055                callee: Callee::Expr(Box::new(Expr::Member(MemberExpr {
1056                    span: _span,
1057                    obj: Box::new(new_array),
1058                    prop: MemberProp::Ident(IdentName::new(atom!("join"), _span)),
1059                }))),
1060                args,
1061                ..Default::default()
1062            }))
1063        }
1064    }

1088    /// `new RegExp("([Sap]+)", "ig")` => `/([Sap]+)/gi`
1089    fn optimize_regex(&mut self, args: &mut Vec<ExprOrSpread>, span: &mut Span) -> Option<Expr> {
1090        fn valid_pattern(pattern: &Expr) -> Option<Atom> {
1091            if let Expr::Lit(Lit::Str(s)) = pattern {
1092                if s.value.code_points().any(|c| {
1093                    let Some(c) = c.to_char() else {
1094                        return true;
1095                    };
1096
1097                    // allowlist
1098                    !c.is_ascii_alphanumeric()
1099                        && !matches!(c, '$' | '[' | ']' | '(' | ')' | '{' | '}' | '-' | '+' | '_')
1100                }) {
1101                    None
1102                } else {
1103                    // SAFETY: We've verified that the string is valid UTF-8 in the if condition
1104                    // above. For this branch, the string contains only ASCII alphanumeric
1105                    // characters and a few special characters.
1106                    Some(unsafe { Atom::from_wtf8_unchecked(s.value.clone()) })
1107                }
1108            } else {
1109                None
1110            }
1111        }
1112        fn valid_flag(flag: &Expr, es_version: EsVersion) -> Option<Atom> {
1113            if let Expr::Lit(Lit::Str(s)) = flag {
1114                let mut set = FxHashSet::default();
1115                for c in s.value.code_points() {
1116                    if !(matches!(c.to_char()?, 'g' | 'i' | 'm')
1117                        || (es_version >= EsVersion::Es2015 && matches!(c.to_char()?, 'u' | 'y'))
1118                        || (es_version >= EsVersion::Es2018 && matches!(c.to_char()?, 's')))
1119                        || (es_version >= EsVersion::Es2022 && matches!(c.to_char()?, 'd'))
1120                    {
1121                        return None;
1122                    }
1123
1124                    if !set.insert(c) {
1125                        return None;
1126                    }
1127                }
1128
1129                // SAFETY: matches above ensure that the string is valid UTF-8 ('g', 'i', 'm',
1130                // 'u', 'y', 's', 'd')
1131                Some(unsafe { Atom::from_wtf8_unchecked(s.value.clone()) })
1132            } else {
1133                None
1134            }
1135        }
1136
1137        let (pattern, flag) = match args.as_slice() {
1138            [ExprOrSpread { spread: None, expr }] => (valid_pattern(expr)?, atom!("")),
1139            [ExprOrSpread {
1140                spread: None,
1141                expr: pattern,
1142            }, ExprOrSpread {
1143                spread: None,
1144                expr: flag,
1145            }] => (
1146                valid_pattern(pattern)?,
1147                valid_flag(flag, self.options.ecma)?,
1148            ),
1149            _ => return None,
1150        };
1151
1152        if pattern.is_empty() {
1153            // For some expressions `RegExp()` and `RegExp("")`
1154            // Theoretically we can use `/(?:)/` to achieve shorter code
1155            // But some browsers released in 2015 don't support them yet.
1156            return None;
1157        }
1158
1159        report_change!("Optimized regex");
1160
1161        Some(
1162            Lit::Regex(Regex {
1163                span: *span,
1164                exp: pattern,
1165                flags: {
1166                    let flag = flag.to_string();
1167                    let mut bytes = flag.into_bytes();
1168                    bytes.sort_unstable();
1169
1170                    String::from_utf8(bytes).unwrap().into()
1171                },
1172            })
1173            .into(),
1174        )
1175    }

1177    /// Array() -> []
1178    fn optimize_array(&mut self, args: &mut Vec<ExprOrSpread>, span: &mut Span) -> Option<Expr> {
1179        // Literal array spreads are expanded before this optimization. Any
1180        // remaining spread has unknown runtime arity, so it may expand to one
1181        // numeric argument and invoke Array's length-constructor behavior.
1182        if args.iter().any(|arg| arg.spread.is_some()) {
1183            return None;
1184        }
1185
1186        if args.len() == 1 {
1187            if let ExprOrSpread { spread: None, expr } = &args[0] {
1188                match &**expr {
1189                    Expr::Lit(Lit::Num(num)) => {
1190                        let length = num.value;
1191
1192                        if !(0_f64..=5_f64).contains(&length) || length.fract() != 0_f64 {
1193                            return None;
1194                        }
1195
1196                        Some(
1197                            ArrayLit {
1198                                span: *span,
1199                                elems: vec![None; length as usize],
1200                            }
1201                            .into(),
1202                        )
1203                    }
1204                    Expr::Lit(_) => Some(
1205                        ArrayLit {
1206                            span: *span,
1207                            elems: vec![args.take().into_iter().next()],
1208                        }
1209                        .into(),
1210                    ),
1211                    _ => None,
1212                }
1213            } else {
1214                None
1215            }
1216        } else {
1217            Some(
1218                ArrayLit {
1219                    span: *span,
1220                    elems: args.take().into_iter().map(Some).collect(),
1221                }
1222                .into(),
1223            )
1224        }
1225    }

1269        if !self.options.pristine_globals {
1270            return;
1271        }
1272
1273        match e {
1274            Expr::New(NewExpr {
1275                span,
1276                callee,
1277                args: Some(args),
1278                ..
1279            })
1280            | Expr::Call(CallExpr {
1281                span,
1282                callee: Callee::Expr(callee),
1283                args,
1284                ..
1285            }) if callee.is_one_of_global_ref_to(self.expr_ctx, &["Array", "Object", "RegExp"]) => {
1286                let new_expr = match &**callee {
1287                    Expr::Ident(Ident { sym, .. }) if &**sym == "RegExp" => {
1288                        self.optimize_regex(args, span)
1289                    }
1290                    Expr::Ident(Ident { sym, .. }) if &**sym == "Array" => {
1291                        self.optimize_array(args, span)
1292                    }
1293                    Expr::Ident(Ident { sym, .. }) if &**sym == "Object" => {
1294                        self.optimize_object(args, span)
1295                    }
1296                    _ => unreachable!(),
1297                };
1298
1299                if let Some(new_expr) = new_expr {
1300                    report_change!(
1301                        "Converting Regexp/Array/Object call to native constructor into literal"
1302                    );
1303                    self.changed = true;
1304                    *e = new_expr;
1305                    return;
1306                }
1307            }
1308            Expr::Call(CallExpr {
1309                span,
1310                callee: Callee::Expr(callee),
1311                args,
1312                ..
1313            }) if callee.is_one_of_global_ref_to(
1314                self.expr_ctx,
1315                &["Boolean", "Number", "String", "Symbol"],
1316            ) =>
1317            {
1318                let new_expr = match &**callee {
1319                    Expr::Ident(Ident { sym, .. }) if &**sym == "Boolean" => match &mut args[..] {
1320                        [] => Some(
1321                            Lit::Bool(Bool {
1322                                span: *span,
1323                                value: false,
1324                            })
1325                            .into(),
1326                        ),
1327                        [ExprOrSpread { spread: None, expr }] => Some(
1328                            UnaryExpr {
1329                                span: *span,
1330                                op: op!("!"),
1331                                arg: UnaryExpr {
1332                                    span: *span,
1333                                    op: op!("!"),
1334                                    arg: expr.take(),
1335                                }
1336                                .into(),
1337                            }
1338                            .into(),
1339                        ),
1340                        _ => None,
1341                    },
1342                    Expr::Ident(Ident { sym, .. }) if &**sym == "Number" => match &mut args[..] {
1343                        [] => Some(
1344                            Lit::Num(Number {
1345                                span: *span,
1346                                value: 0.0,
1347                                raw: None,
1348                            })
1349                            .into(),
1350                        ),
1351                        // This is indeed very unsafe in case of BigInt, so it
1352                        // requires both `unsafe` and `unsafe_math`, matching
1353                        // terser (since v4.3.11).
1354                        [ExprOrSpread { spread: None, expr }]
1355                            if self.options.unsafe_passes && self.options.unsafe_math =>
1356                        {
1357                            Some(
1358                                UnaryExpr {
1359                                    span: *span,
1360                                    op: op!(unary, "+"),
1361                                    arg: expr.take(),
1362                                }
1363                                .into(),
1364                            )
1365                        }
1366                        _ => None,
1367                    },
1368                    Expr::Ident(Ident { sym, .. }) if &**sym == "String" => match &mut args[..] {
1369                        [] => Some(
1370                            Lit::Str(Str {
1371                                span: *span,
1372                                value: atom!("").into(),
1373                                raw: None,
1374                            })
1375                            .into(),
1376                        ),
1377                        // this is also very unsafe in case of Symbol
1378                        [ExprOrSpread { spread: None, expr }] if self.options.unsafe_passes => {
1379                            Some(
1380                                BinExpr {
1381                                    span: *span,
1382                                    left: expr.take(),
1383                                    op: op!(bin, "+"),
1384                                    right: Lit::Str(Str {
1385                                        span: *span,
1386                                        value: atom!("").into(),
1387                                        raw: None,
1388                                    })
1389                                    .into(),
1390                                }
1391                                .into(),
1392                            )
1393                        }
1394                        _ => None,
1395                    },
1396                    Expr::Ident(Ident { sym, .. }) if &**sym == "Symbol" => {
1397                        if let [ExprOrSpread { spread: None, .. }] = &mut args[..] {
1398                            if self.options.unsafe_symbols {
1399                                args.clear();
1400                                report_change!("Remove Symbol call parameter");
1401                                self.changed = true;
1402                            }
1403                        }
1404                        None
1405                    }
1406                    _ => unreachable!(),
1407                };
1408
1409                if let Some(new_expr) = new_expr {
1410                    report_change!(
1411                        "Converting Boolean/Number/String/Symbol call to native constructor to \
1412                         literal"
1413                    );
1414                    self.changed = true;
1415                    *e = new_expr;
1416                    return;
1417                }
1418            }
1419            _ => {}
1420        };
1421
1422        match e {
1423            Expr::New(NewExpr {
1424                span,
1425                ctxt,
1426                callee,
1427                args,
1428                ..
1429            }) if callee.is_one_of_global_ref_to(
1430                self.expr_ctx,
1431                &[
1432                    "Object",
1433                    // https://262.ecma-international.org/12.0/#sec-array-constructor
1434                    "Array",
1435                    // https://262.ecma-international.org/12.0/#sec-function-constructor
1436                    "Function",
1437                    // https://262.ecma-international.org/12.0/#sec-error-constructor
1438                    "Error",

1470    /// value of function is ignored.
1471    ///
1472    /// Returns true if something is modified.
1473    fn drop_return_value(&mut self, stmts: &mut Vec<Stmt>) -> bool {
1474        for s in stmts.iter_mut() {
1475            if let Stmt::Return(ReturnStmt {
1476                arg: arg @ Some(..),
1477                ..
1478            }) = s
1479            {
1480                self.ignore_return_value(
1481                    arg.as_deref_mut().unwrap(),
1482                    DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
1483                        .union(DropOpts::DROP_NUMBER)
1484                        .union(DropOpts::DROP_STR_LIT),
1485                );
1486
1487                if let Some(Expr::Invalid(..)) = arg.as_deref() {
1488                    self.changed = true;
1489                    *arg = None;
1490                }
1491            }
1492        }
1493
1494        if let Some(last) = stmts.last_mut() {
1495            self.drop_return_value_of_stmt(last)
1496        } else {
1497            false
1498        }

1501    fn compress_array_join_as_tpl(
1502        &mut self,
1503        span: Span,
1504        elems: &mut Vec<Option<ExprOrSpread>>,
1505        sep: &Wtf8,
1506    ) -> Option<Expr> {
1507        if !self.options.evaluate {
1508            return None;
1509        }
1510
1511        if elems.iter().flatten().any(|elem| match &*elem.expr {
1512            Expr::Tpl(t) => t.quasis.iter().any(|q| q.cooked.is_none()),
1513            Expr::Lit(Lit::Str(..)) => false,
1514            _ => true,
1515        }) {
1516            return None;
1517        }
1518
1519        self.changed = true;
1520        report_change!("Compressing array.join() as template literal");
1521
1522        let mut new_tpl = Tpl {
1523            span,
1524            quasis: Vec::new(),
1525            exprs: Vec::new(),
1526        };
1527        let mut cur_raw = String::new();
1528        let mut cur_cooked = Wtf8Buf::default();
1529        let mut first = true;
1530
1531        for elem in elems.take() {
1532            if first {
1533                first = false;
1534            } else {
1535                cur_raw.push_str(&convert_str_value_to_tpl_raw(sep));
1536                cur_cooked.push_wtf8(sep);
1537            }
1538
1539            let Some(elem) = elem else {
1540                continue;
1541            };
1542
1543            match *elem.expr {
1544                Expr::Tpl(mut tpl) => {
1545                    //
1546                    for idx in 0..(tpl.quasis.len() + tpl.exprs.len()) {
1547                        if idx % 2 == 0 {
1548                            // quasis
1549                            let e = tpl.quasis[idx / 2].take();
1550
1551                            cur_cooked.push_wtf8(&e.cooked.unwrap());
1552                            cur_raw.push_str(&e.raw);
1553                        } else {
1554                            new_tpl.quasis.push(TplElement {
1555                                span: DUMMY_SP,
1556                                tail: false,
1557                                cooked: Some((&*cur_cooked).into()),
1558                                raw: (&*cur_raw).into(),
1559                            });
1560
1561                            cur_raw.clear();
1562                            cur_cooked.clear();
1563
1564                            let e = tpl.exprs[idx / 2].take();
1565
1566                            new_tpl.exprs.push(e);
1567                        }
1568                    }
1569                }
1570                Expr::Lit(Lit::Str(s)) => {
1571                    cur_cooked.push_wtf8(&Cow::Borrowed(&s.value));
1572                    cur_raw.push_str(&convert_str_value_to_tpl_raw(&s.value));
1573                }
1574                _ => {
1575                    unreachable!()
1576                }
1577            }
1578        }
1579
1580        new_tpl.quasis.push(TplElement {
1581            span: DUMMY_SP,
1582            tail: false,
1583            cooked: Some(cur_cooked.into()),
1584            raw: cur_raw.into(),
1585        });
1586
1587        Some(new_tpl.into())
1588    }

1590    /// Returns true if something is modified.
1591    fn drop_return_value_of_stmt(&mut self, s: &mut Stmt) -> bool {
1592        match s {
1593            Stmt::Block(s) => self.drop_return_value(&mut s.stmts),
1594            Stmt::Return(ret) => {
1595                self.changed = true;
1596                report_change!("Dropping `return` token");
1597
1598                let span = ret.span;
1599                match ret.arg.take() {
1600                    Some(arg) => {
1601                        *s = ExprStmt { span, expr: arg }.into();
1602                    }
1603                    None => {
1604                        *s = EmptyStmt { span }.into();
1605                    }
1606                }
1607
1608                true
1609            }
1610
1611            Stmt::Labeled(s) => self.drop_return_value_of_stmt(&mut s.body),
1612            Stmt::If(s) => {
1613                let c = self.drop_return_value_of_stmt(&mut s.cons);
1614                let a = s
1615                    .alt
1616                    .as_deref_mut()
1617                    .map(|s| self.drop_return_value_of_stmt(s))
1618                    .unwrap_or_default();
1619
1620                c || a
1621            }
1622
1623            Stmt::Try(s) => {
1624                let a = if s.finalizer.is_none() {
1625                    self.drop_return_value(&mut s.block.stmts)
1626                } else {
1627                    false
1628                };
1629
1630                if let Some(s) = &mut s.finalizer {
1631                    // A finalizer return overrides the pending completion, so only its value can
1632                    // be removed when the IIFE result is ignored.
1633                    for stmt in &mut s.stmts {
1634                        self.ignore_return_value_of_return_stmt(
1635                            stmt,
1636                            DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
1637                                .union(DropOpts::DROP_NUMBER)
1638                                .union(DropOpts::DROP_STR_LIT),
1639                        );
1640                    }
1641                }
1642
1643                a
1644            }
1645
1646            _ => false,
1647        }
1648    }

1650    fn make_ignored_expr(
1651        &mut self,
1652        span: Span,
1653        exprs: impl Iterator<Item = Box<Expr>>,
1654    ) -> Option<Expr> {
1655        let mut exprs = exprs
1656            .filter_map(|mut e| {
1657                self.ignore_return_value(
1658                    &mut e,
1659                    DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
1660                        .union(DropOpts::DROP_NUMBER)
1661                        .union(DropOpts::DROP_STR_LIT),
1662                );
1663
1664                if let Expr::Invalid(..) = &*e {
1665                    None
1666                } else {
1667                    Some(e)
1668                }
1669            })
1670            .collect::<Vec<_>>();
1671
1672        if exprs.is_empty() {
1673            return None;
1674        }
1675        if exprs.len() == 1 {
1676            let mut new = *exprs.remove(0);
1677            new.set_span(span);
1678            return Some(new);
1679        }
1680
1681        Some(
1682            SeqExpr {
1683                span: DUMMY_SP,
1684                exprs,
1685            }
1686            .into(),
1687        )
1688    }

1690    /// Calls [`Self::ignore_return_value`] on the arguments of return
1691    /// statemetns.
1692    ///
1693    /// This function is recursive but does not go into nested scopes.
1694    pub(super) fn ignore_return_value_of_return_stmt(&mut self, s: &mut Stmt, opts: DropOpts) {
1695        match s {
1696            Stmt::Return(s) => {
1697                if let Some(arg) = &mut s.arg {
1698                    self.ignore_return_value(arg, opts);
1699                    if arg.is_invalid() {
1700                        report_change!(
1701                            "Dropped the argument of a return statement because the return value \
1702                             is ignored"
1703                        );
1704                        s.arg = None;
1705                    }
1706                }
1707            }
1708
1709            Stmt::Block(s) => {
1710                for stmt in &mut s.stmts {
1711                    self.ignore_return_value_of_return_stmt(stmt, opts);
1712                }
1713            }
1714
1715            Stmt::If(s) => {
1716                self.ignore_return_value_of_return_stmt(&mut s.cons, opts);
1717                if let Some(alt) = &mut s.alt {
1718                    self.ignore_return_value_of_return_stmt(alt, opts);
1719                }
1720            }
1721
1722            Stmt::Switch(s) => {
1723                for case in &mut s.cases {
1724                    for stmt in &mut case.cons {
1725                        self.ignore_return_value_of_return_stmt(stmt, opts);
1726                    }
1727                }
1728            }
1729
1730            _ => {}
1731        }
1732    }

1734    pub(super) fn ignore_return_value(&mut self, e: &mut Expr, opts: DropOpts) {
1735        if e.is_invalid() {
1736            return;
1737        }
1738
1739        if self.ctx.contains(Ctx::IN_DELETE) {
1740            return;
1741        }
1742
1743        debug_assert_valid(e);
1744
1745        self.optimize_expr_in_bool_ctx(e, true);
1746
1747        match e {
1748            Expr::Seq(seq) => {
1749                if let Some(last) = seq.exprs.last_mut() {
1750                    self.ignore_return_value(last, opts);
1751                }
1752
1753                seq.exprs.retain(|expr| !expr.is_invalid());
1754
1755                if seq.exprs.is_empty() {
1756                    e.take();
1757                    return;
1758                }
1759                if seq.exprs.len() == 1 {
1760                    *e = *seq.exprs.remove(0);
1761                }
1762                return;
1763            }
1764
1765            Expr::Call(CallExpr {
1766                span, ctxt, args, ..
1767            }) if ctxt.has_mark(self.marks.pure) => {
1768                report_change!("ignore_return_value: Dropping a pure call");
1769                self.changed = true;
1770
1771                let new =
1772                    self.make_ignored_expr(*span, args.take().into_iter().map(|arg| arg.expr));
1773
1774                *e = new.unwrap_or(Invalid { span: DUMMY_SP }.into());
1775                return;
1776            }
1777
1778            Expr::Call(CallExpr {
1779                span,
1780                callee: Callee::Expr(callee),
1781                args,
1782                ..
1783            }) if callee.is_pure_callee(self.expr_ctx) => {
1784                report_change!("ignore_return_value: Dropping a pure call (callee is pure)");
1785                self.changed = true;
1786
1787                let new =
1788                    self.make_ignored_expr(*span, args.take().into_iter().map(|arg| arg.expr));
1789
1790                *e = new.unwrap_or(Invalid { span: DUMMY_SP }.into());
1791                return;
1792            }
1793
1794            Expr::TaggedTpl(TaggedTpl {
1795                span, ctxt, tpl, ..
1796            }) if ctxt.has_mark(self.marks.pure) => {
1797                report_change!("ignore_return_value: Dropping a pure call");
1798                self.changed = true;
1799
1800                let new = self.make_ignored_expr(*span, tpl.exprs.take().into_iter());
1801
1802                *e = new.unwrap_or(Invalid { span: DUMMY_SP }.into());
1803                return;
1804            }
1805
1806            Expr::New(NewExpr {
1807                callee,
1808                span,
1809                ctxt,
1810                args,
1811                ..
1812            }) if callee.is_pure_callee(self.expr_ctx) || ctxt.has_mark(self.marks.pure) => {
1813                report_change!("ignore_return_value: Dropping a pure call");
1814                self.changed = true;
1815
1816                let new = self.make_ignored_expr(
1817                    *span,
1818                    args.take().into_iter().flatten().map(|arg| arg.expr),
1819                );
1820
1821                *e = new.unwrap_or(Invalid { span: DUMMY_SP }.into());
1822                return;
1823            }
1824
1825            _ => {}
1826        }
1827
1828        match e {
1829            Expr::Call(CallExpr {
1830                span,
1831                callee: Callee::Expr(callee),
1832                args,
1833                ..
1834            }) if callee.is_pure_callee(self.expr_ctx) => {
1835                self.changed = true;
1836                report_change!("Dropping pure call as callee is pure");
1837                *e = self
1838                    .make_ignored_expr(*span, args.take().into_iter().map(|arg| arg.expr))
1839                    .unwrap_or(Invalid { span: DUMMY_SP }.into());
1840                return;
1841            }
1842
1843            Expr::TaggedTpl(TaggedTpl {
1844                span,
1845                tag: callee,
1846                tpl,
1847                ..
1848            }) if callee.is_pure_callee(self.expr_ctx) => {
1849                self.changed = true;
1850                report_change!("Dropping pure tag tpl as callee is pure");
1851                *e = self
1852                    .make_ignored_expr(*span, tpl.exprs.take().into_iter())
1853                    .unwrap_or(Invalid { span: DUMMY_SP }.into());
1854                return;
1855            }
1856            _ => (),
1857        }
1858
1859        if self.options.conditionals || self.options.expr {
1860            if let Expr::Cond(CondExpr {
1861                span,
1862                test,
1863                cons,
1864                alt,
1865                ..
1866            }) = e
1867            {
1868                self.ignore_return_value(cons, opts);
1869                self.ignore_return_value(alt, opts);
1870
1871                if cons.is_invalid() && alt.is_invalid() {
1872                    report_change!("Dropping a conditional expression");
1873                    *e = *test.take();
1874                    self.changed = true;
1875                    return;
1876                }
1877
1878                if cons.is_invalid() {
1879                    *e = Expr::Bin(BinExpr {
1880                        span: *span,
1881                        op: op!("||"),
1882                        left: test.take(),
1883                        right: alt.take(),
1884                    });
1885                    report_change!("Dropping the `then` branch of a conditional expression");
1886                    self.changed = true;
1887                    return;
1888                }
1889
1890                if alt.is_invalid() {
1891                    *e = Expr::Bin(BinExpr {
1892                        span: *span,
1893                        op: op!("&&"),
1894                        left: test.take(),
1895                        right: cons.take(),
1896                    });
1897                    report_change!("Dropping the `else` branch of a conditional expression");
1898                    self.changed = true;
1899                    return;
1900                }
1901            }
1902        }
1903
1904        if opts.contains(DropOpts::DROP_NUMBER) {
1905            if let Expr::Lit(Lit::Num(n)) = e {
1906                // Skip 0
1907                if n.value != 0.0 && n.value.classify() == FpCategory::Normal {
1908                    report_change!("Dropping a number");
1909                    *e = Invalid { span: DUMMY_SP }.into();
1910                    return;
1911                }
1912            }
1913        }
1914
1915        if let Expr::Ident(i) = e {
1916            // If it's not a top level, it's a reference to a declared variable.
1917            if i.ctxt.outer() == self.marks.unresolved_mark {
1918                if self.options.side_effects
1919                    || (self.options.unused && opts.contains(DropOpts::DROP_GLOBAL_REFS_IF_UNUSED))
1920                {
1921                    if is_global_var_with_pure_property_access(&i.sym) {
1922                        report_change!("Dropping a reference to a global variable");
1923                        *e = Invalid { span: DUMMY_SP }.into();
1924                        return;
1925                    }
1926                }
1927            } else {
1928                report_change!("Dropping an identifier as it's declared");
1929                *e = Invalid { span: DUMMY_SP }.into();
1930                return;
1931            }
1932        }
1933
1934        if self.options.side_effects
1935            || self.options.dead_code
1936            || self.options.collapse_vars
1937            || self.options.expr
1938        {
1939            match e {
1940                Expr::Unary(UnaryExpr {
1941                    op: op!("!"), arg, ..
1942                }) => {
1943                    self.ignore_return_value(
1944                        arg,
1945                        DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
1946                            .union(DropOpts::DROP_NUMBER)
1947                            .union(DropOpts::DROP_STR_LIT),
1948                    );
1949
1950                    if arg.is_invalid() {
1951                        report_change!("Dropping an unary expression");
1952                        *e = Invalid { span: DUMMY_SP }.into();
1953                        return;
1954                    }
1955                }
1956
1957                Expr::Unary(UnaryExpr {
1958                    op: op!("void") | op!(unary, "+") | op!(unary, "-") | op!("~"),
1959                    arg,
1960                    ..
1961                }) => {
1962                    self.ignore_return_value(
1963                        arg,
1964                        DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
1965                            .union(DropOpts::DROP_NUMBER)
1966                            .union(DropOpts::DROP_STR_LIT),
1967                    );
1968
1969                    if arg.is_invalid() {
1970                        report_change!("Dropping an unary expression");
1971                        *e = Invalid { span: DUMMY_SP }.into();
1972                        return;
1973                    }
1974
1975                    report_change!("Dropping an unary operator");
1976                    *e = *arg.take();
1977                    return;
1978                }
1979
1980                Expr::Bin(
1981                    be @ BinExpr {
1982                        op: op!("||") | op!("&&"),
1983                        ..
1984                    },
1985                ) => {
1986                    self.ignore_return_value(&mut be.right, opts);
1987
1988                    if be.right.is_invalid() {
1989                        report_change!("Dropping the RHS of a binary expression ('&&' / '||')");
1990                        *e = *be.left.take();
1991                        return;
1992                    }
1993                }
1994
1995                Expr::Tpl(tpl) => {
1996                    self.changed = true;
1997                    report_change!("Dropping a template literal");
1998
1999                    for expr in tpl.exprs.iter_mut() {
2000                        self.ignore_return_value(&mut *expr, opts);
2001                    }
2002                    tpl.exprs.retain(|expr| !expr.is_invalid());
2003                    if tpl.exprs.is_empty() {
2004                        e.take();
2005                    } else {
2006                        if tpl.exprs.len() == 1 {
2007                            *e = *tpl.exprs.remove(0);
2008                        } else {
2009                            *e = SeqExpr {
2010                                span: tpl.span,
2011                                exprs: tpl.exprs.take(),
2012                            }
2013                            .into();
2014                        }
2015                    }
2016
2017                    return;
2018                }
2019
2020                Expr::Member(MemberExpr {
2021                    obj,
2022                    prop: MemberProp::Ident(prop),
2023                    ..
2024                }) if obj.is_ident_ref_to("arguments") => {
2025                    if &*prop.sym == "callee" {
2026                        return;
2027                    }
2028                    e.take();
2029                    return;
2030                }
2031                // Unused member accesses can be dropped only if they are
2032                // pure. The object and a computed key are still evaluated:
2033                // `x().y` must keep `x()`.
2034                Expr::Member(member) if self.is_pure_member_access(member) => {
2035                    let MemberExpr {
2036                        span, obj, prop, ..
2037                    } = member;
2038                    let span = *span;
2039                    let computed_key = match prop {
2040                        MemberProp::Computed(c) => Some(c.expr.take()),
2041                        _ => None,
2042                    };
2043
2044                    report_change!("ignore_return_value: Dropping a pure property access");
2045                    self.changed = true;
2046
2047                    *e = self
2048                        .make_ignored_expr(
2049                            span,
2050                            [Some(obj.take()), computed_key].into_iter().flatten(),
2051                        )
2052                        .unwrap_or(Invalid { span: DUMMY_SP }.into());
2053                    return;
2054                }
2055
2056                _ => {}
2057            }
2058        }
2059
2060        match e {
2061            Expr::Lit(Lit::Num(n)) if n.value == 0.0 && opts.contains(DropOpts::DROP_NUMBER) => {
2062                report_change!("Dropping a zero number");
2063                *e = Invalid { span: DUMMY_SP }.into();
2064                return;
2065            }
2066            Expr::Lit(Lit::Num(..)) => {}
2067
2068            Expr::Ident(i) if i.ctxt.outer() != self.marks.unresolved_mark => {
2069                report_change!("Dropping an identifier as it's declared");
2070
2071                self.changed = true;
2072                *e = Invalid { span: DUMMY_SP }.into();
2073                return;
2074            }
2075            Expr::Ident(..) => {}
2076
2077            Expr::Lit(Lit::Null(..) | Lit::BigInt(..) | Lit::Bool(..) | Lit::Regex(..))
2078            | Expr::This(..) => {
2079                report_change!("Dropping meaningless values");
2080
2081                self.changed = true;
2082                *e = Expr::dummy();
2083                return;
2084            }
2085
2086            Expr::Bin(
2087                bin @ BinExpr {
2088                    op:
2089                        op!(bin, "+")
2090                        | op!(bin, "-")
2091                        | op!("*")
2092                        | op!("%")
2093                        | op!("**")
2094                        | op!("^")
2095                        | op!("&")
2096                        | op!("|")
2097                        | op!(">>")
2098                        | op!("<<")
2099                        | op!(">>>")
2100                        | op!("===")
2101                        | op!("!==")
2102                        | op!("==")
2103                        | op!("!=")
2104                        | op!("<")
2105                        | op!("<=")
2106                        | op!(">")
2107                        | op!(">="),
2108                    ..
2109                },
2110            ) => {
2111                self.ignore_return_value(
2112                    &mut bin.left,
2113                    DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
2114                        .union(DropOpts::DROP_NUMBER)
2115                        .union(DropOpts::DROP_STR_LIT),
2116                );
2117                self.ignore_return_value(
2118                    &mut bin.right,
2119                    DropOpts::DROP_GLOBAL_REFS_IF_UNUSED
2120                        .union(DropOpts::DROP_NUMBER)
2121                        .union(DropOpts::DROP_STR_LIT),
2122                );
2123
2124                if bin.left.is_invalid() && bin.right.is_invalid() {
2125                    *e = Invalid { span: DUMMY_SP }.into();
2126                    return;
2127                } else if bin.right.is_invalid() {
2128                    *e = *bin.left.take();
2129                    return;
2130                } else if bin.left.is_invalid() {
2131                    *e = *bin.right.take();
2132                    return;
2133                }
2134
