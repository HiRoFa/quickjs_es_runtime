15    /// This only handles `'foo' + ('bar' + baz) because others are handled by
16    /// expression simplifier.
17    pub(super) fn eval_str_addition(&mut self, e: &mut Expr) {
18        let (span, l_l, r_l, r_r) = match e {
19            Expr::Bin(
20                e @ BinExpr {
21                    op: op!(bin, "+"), ..
22                },
23            ) => match &mut *e.right {
24                Expr::Bin(
25                    r @ BinExpr {
26                        op: op!(bin, "+"), ..
27                    },
28                ) => (e.span, &mut *e.left, &mut *r.left, &mut r.right),
29                _ => return,
30            },
31            _ => return,
32        };
33
34        match l_l.get_type(self.expr_ctx) {
35            Known(Type::Str) => {}
36            _ => return,
37        }
38        match r_l.get_type(self.expr_ctx) {
39            Known(Type::Str) => {}
40            _ => return,
41        }
42
43        let lls = l_l.as_pure_string(self.expr_ctx);
44        let rls = r_l.as_pure_string(self.expr_ctx);
45
46        if let (Known(lls), Known(rls)) = (lls, rls) {
47            self.changed = true;
48            report_change!("evaluate: 'foo' + ('bar' + baz) => 'foobar' + baz");
49
50            let s = lls.into_owned() + &*rls;
51            *e = BinExpr {
52                span,
53                op: op!(bin, "+"),
54                left: s.into(),
55                right: r_r.take(),
56            }
57            .into();
58        }
59    }

61    pub(super) fn eval_tpl_as_str(&mut self, e: &mut Expr) {
62        if !self.options.evaluate {
63            return;
64        }
65
66        fn need_unsafe(e: &Expr) -> bool {
67            match e {
68                Expr::Lit(..) => false,
69                Expr::Bin(e) => need_unsafe(&e.left) || need_unsafe(&e.right),
70                _ => true,
71            }
72        }
73
74        let tpl = match e {
75            Expr::Tpl(e) => e,
76            _ => return,
77        };
78
79        if tpl.quasis.len() == 2
80            && (tpl.quasis[0].cooked.is_some() || !tpl.quasis[0].raw.contains('\\'))
81            && tpl.quasis[1].raw.is_empty()
82        {
83            if !self.options.unsafe_passes && need_unsafe(&tpl.exprs[0]) {
84                return;
85            }
86
87            self.changed = true;
88            report_change!("evaluating a template to a string");
89            *e = BinExpr {
90                span: tpl.span,
91                op: op!(bin, "+"),
92                left: tpl.quasis[0]
93                    .cooked
94                    .clone()
95                    .unwrap_or_else(|| tpl.quasis[0].raw.clone().into())
96                    .into(),
97                right: tpl.exprs[0].take(),
98            }
99            .into();
100        }
101    }

103    pub(super) fn eval_nested_tpl(&mut self, e: &mut Expr) {
104        let tpl = match e {
105            Expr::Tpl(e) => e,
106            _ => return,
107        };
108
109        if !tpl.exprs.iter().any(|e| match &**e {
110            Expr::Tpl(t) => t
111                .quasis
112                .iter()
113                .all(|q| q.cooked.is_some() && !q.raw.contains('\\')),
114            _ => false,
115        }) {
116            return;
117        }
118
119        self.changed = true;
120        report_change!("evaluating nested template literals");
121
122        let mut new_tpl = Tpl {
123            span: tpl.span,
124            quasis: Default::default(),
125            exprs: Default::default(),
126        };
127        let mut cur_cooked_str = Wtf8Buf::new();
128        let mut cur_raw_str = String::new();
129        let mut q_iter = tpl.quasis.take().into_iter();
130        let e_iter = tpl.exprs.take().into_iter();
131
132        macro_rules! push_str {
133            ($e:expr) => {
134                if let Some(cooked) = $e.cooked {
135                    cur_cooked_str.push_wtf8(&cooked);
136                } else {
137                    cur_cooked_str.push_wtf8(&Str::from_tpl_raw(&$e));
138                }
139                cur_raw_str.push_str(&$e.raw);
140            };
141        }
142
143        macro_rules! end_str {
144            () => {
145                let cooked = Wtf8Atom::from(&*cur_cooked_str);
146                let raw = Atom::from(&*cur_raw_str);
147                cur_cooked_str.clear();
148                cur_raw_str.clear();
149                new_tpl.quasis.push(TplElement {
150                    span: DUMMY_SP,
151                    tail: false,
152                    cooked: Some(cooked),
153                    raw,
154                });
155            };
156        }
157
158        // Consume quasis first to make sure it align with exprs
159        // quasis.len() == exprs.len() + 1
160        if let Some(q) = q_iter.next() {
161            push_str!(q);
162        }
163
164        for (q, mut e) in zip(q_iter, e_iter) {
165            self.eval_nested_tpl(&mut e);
166            match *e {
167                Expr::Tpl(mut tpl) => {
168                    // For evaluated template only the first
169                    // and the last quasi could be concat with
170                    // outside quasis.
171                    let mut quasis_taken = tpl.quasis.take();
172                    let l = quasis_taken.len();
173
174                    // Store the first quasi for later concat
175                    let first = quasis_taken[0].take();
176                    push_str!(first);
177
178                    if l > 1 {
179                        // If there are more than one quasi
180                        // Concat first with outside quasis
181                        end_str!();
182
183                        // Store the last quasi for later concat
184                        let last = quasis_taken.pop().unwrap();
185                        push_str!(last);
186
187                        // Append the rest of quasis and exprs to new_tpl
188                        new_tpl.quasis.extend(quasis_taken.into_iter().skip(1));
189                        new_tpl.exprs.extend(tpl.exprs);
190                    }
191                }
192                _ => {
193                    end_str!();
194
195                    new_tpl.exprs.push(e);
196                }
197            }
198            push_str!(q);
199        }
200
201        end_str!();
202
203        *e = new_tpl.into();
204    }

206    /// Converts template literals to string if `exprs` of [Tpl] is empty.
207    pub(super) fn convert_tpl_to_str(&mut self, e: &mut Expr) {
208        match e {
209            Expr::Tpl(t) if t.quasis.len() == 1 && t.exprs.is_empty() => {
210                let c = &t.quasis[0].raw;
211                let mut template_longer_count = 0;
212                let mut iter = c.chars().peekable();
213                while let Some(ch) = iter.next() {
214                    match ch {
215                        '\\' => {
216                            if let Some(next_ch) = iter.next() {
217                                match next_ch {
218                                    '\r' if iter.peek() == Some(&'\n') => {
219                                        iter.next();
220                                    }
221                                    '\n' | '\r' => {}
222                                    'n' | 'r' => {
223                                        template_longer_count -= 1;
224                                    }
225                                    '`' => {
226                                        template_longer_count += 1;
227                                    }
228                                    _ => {}
229                                }
230                            }
231                        }
232                        c @ '\n' | c @ '\r' => {
233                            template_longer_count -= 1;
234                            if c == '\r' && iter.peek() == Some(&'\n') {
235                                iter.next();
236                            }
237                        }
238                        _ => {
239                            // When the target environment is below ES2015
240                            // and non-BMP characters (like emojis) are encountered,
241                            // we stop the conversion.
242                            //
243                            // This is because:
244                            // 1. Tpl: `🦀` (may output directly in source code or require minimal
245                            //    escaping) -> shorter
246                            // 2. Str (in ES5 mode): `\uD83E\uDD80` (escape sequence for surrogate
247                            //    pair) -> extremely long
248                            if self.options.ecma < EsVersion::Es2015 && ch > '\u{ffff}' {
249                                return;
250                            }
251                        }
252                    }
253                }
254
255                if template_longer_count < 0 {
256                    return;
257                }
258
259                if let Some(cooked) = &t.quasis[0].cooked {
260                    report_change!("converting a template literal to a string literal");
261
262                    *e = Lit::Str(Str {
263                        span: t.span,
264                        raw: None,
265                        value: cooked.clone(),
266                    })
267                    .into();
268                    return;
269                }
270
271                let value = Str::from_tpl_raw(&t.quasis[0]);
272
273                report_change!(
274                    "converting a template literal to a string literal by Str::from_tpl_raw"
275                );
276
277                *e = Lit::Str(Str {
278                    span: t.span,
279                    raw: None,
280                    value,
281                })
282                .into();
283            }
284            _ => {}
285        }
286    }

288    /// This compresses a template literal by inlining string literals in
289    /// expresions into quasis.
290    ///
291    /// Note that this pass only cares about string literals and conversion to a
292    /// string literal should be done before calling this pass.
293    pub(super) fn compress_tpl(&mut self, tpl: &mut Tpl) {
294        debug_assert_eq!(tpl.exprs.len() + 1, tpl.quasis.len());
295        let has_str_lit = tpl
296            .exprs
297            .iter()
298            .any(|expr| matches!(&**expr, Expr::Lit(Lit::Str(..))));
299        if !has_str_lit {
300            return;
301        }
302
303        trace_op!("compress_tpl");
304
305        let mut quasis = Vec::new();
306        let mut exprs = Vec::new();
307        let mut cur_raw = String::new();
308        let mut cur_cooked = Some(Wtf8Buf::new());
309
310        for i in 0..(tpl.exprs.len() + tpl.quasis.len()) {
311            if i % 2 == 0 {
312                let i = i / 2;
313                let q = tpl.quasis[i].clone();
314
315                if q.cooked.is_some() {
316                    if let Some(cur_cooked) = &mut cur_cooked {
317                        cur_cooked.push_str("");
318                    }
319                } else {
320                    // If cooked is None, it means that the template literal contains invalid escape
321                    // sequences.
322                    cur_cooked = None;
323                }
324            } else {
325                let i = i / 2;
326                let e = &tpl.exprs[i];
327
328                match &**e {
329                    Expr::Lit(Lit::Str(s)) => {
330                        if cur_cooked.is_none() && s.raw.is_none() {
331                            return;
332                        }
333
334                        if let Some(cur_cooked) = &mut cur_cooked {
335                            cur_cooked.push_str("");
336                        }
337                    }
338                    _ => {
339                        cur_cooked = Some(Wtf8Buf::new());
340                    }
341                }
342            }
343        }
344
345        cur_cooked = Some(Default::default());
346
347        for i in 0..(tpl.exprs.len() + tpl.quasis.len()) {
348            if i % 2 == 0 {
349                let i = i / 2;
350                let q = tpl.quasis[i].take();
351
352                cur_raw.push_str(&q.raw);
353                if let Some(cooked) = q.cooked {
354                    if let Some(cur_cooked) = &mut cur_cooked {
355                        cur_cooked.push_wtf8(&cooked);
356                    }
357                } else {
358                    // If cooked is None, it means that the template literal contains invalid escape
359                    // sequences.
360                    cur_cooked = None;
361                }
362            } else {
363                let i = i / 2;
364                let e = tpl.exprs[i].take();
365
366                match *e {
367                    Expr::Lit(Lit::Str(s)) => {
368                        if let Some(cur_cooked) = &mut cur_cooked {
369                            cur_cooked.push_wtf8(&Cow::Borrowed(&s.value));
370                        }
371
372                        if let Some(raw) = &s.raw {
373                            if raw.len() >= 2 {
374                                // Exclude quotes
375                                cur_raw
376                                    .push_str(&convert_str_raw_to_tpl_raw(&raw[1..raw.len() - 1]));
377                            }
378                        } else {
379                            cur_raw.push_str(&convert_str_value_to_tpl_raw(&s.value));
380                        }
381                    }
382                    _ => {
383                        quasis.push(TplElement {
384                            span: DUMMY_SP,
385                            tail: true,
386                            cooked: cur_cooked.take().map(From::from),
387                            raw: take(&mut cur_raw).into(),
388                        });
389                        cur_cooked = Some(Wtf8Buf::new());
390
391                        exprs.push(e);
392                    }
393                }
394            }
395        }
396
397        report_change!("compressing template literals");
398
399        quasis.push(TplElement {
400            span: DUMMY_SP,
401            tail: true,
402            cooked: cur_cooked.map(From::from),
403            raw: cur_raw.into(),
404        });
405
406        debug_assert_eq!(exprs.len() + 1, quasis.len());
407
408        tpl.quasis = quasis;
409        tpl.exprs = exprs;
410    }

412    /// Called for binary operations with `+`.
413    pub(super) fn concat_tpl(&mut self, l: &mut Expr, r: &mut Expr) {
414        match (&mut *l, &mut *r) {
415            (Expr::Tpl(l), Expr::Lit(Lit::Str(rs))) => {
416                if let Some(raw) = &rs.raw {
417                    if raw.len() <= 2 {
418                        return;
419                    }
420                }
421
422                // Append
423                if let Some(l_last) = l.quasis.last_mut() {
424                    self.changed = true;
425
426                    report_change!(
427                        "template: Concatted a string (`{:?}`) on rhs of `+` to a template literal",
428                        rs.value
429                    );
430
431                    if let Some(cooked) = &mut l_last.cooked {
432                        let mut c = Wtf8Buf::from(&*cooked);
433                        c.push_wtf8(&Cow::Borrowed(&rs.value));
434                        *cooked = c.into();
435                    }
436
437                    l_last.raw = format!(
438                        "{}{}",
439                        l_last.raw,
440                        rs.raw
441                            .clone()
442                            .map(|s| convert_str_raw_to_tpl_raw(&s[1..s.len() - 1]))
443                            .unwrap_or_else(|| convert_str_value_to_tpl_raw(&rs.value).into())
444                    )
445                    .into();
446
447                    r.take();
448                }
449            }
450
451            (Expr::Lit(Lit::Str(ls)), Expr::Tpl(r)) => {
452                if let Some(raw) = &ls.raw {
453                    if raw.len() <= 2 {
454                        return;
455                    }
456                }
457
458                // Append
459                if let Some(r_first) = r.quasis.first_mut() {
460                    self.changed = true;
461
462                    report_change!(
463                        "template: Prepended a string (`{:?}`) on lhs of `+` to a template literal",
464                        ls.value
465                    );
466
467                    if let Some(cooked) = &mut r_first.cooked {
468                        let mut c = Wtf8Buf::new();
469                        c.push_wtf8(&Cow::Borrowed(&ls.value));
470                        c.push_wtf8(&*cooked);
471                        *cooked = c.into();
472                    }
473
474                    let new: Atom = format!(
475                        "{}{}",
476                        ls.raw
477                            .clone()
478                            .map(|s| convert_str_raw_to_tpl_raw(&s[1..s.len() - 1]))
479                            .unwrap_or_else(|| convert_str_value_to_tpl_raw(&ls.value).into()),
480                        r_first.raw
481                    )
482                    .into();
483                    r_first.raw = new;
484
485                    l.take();
486                }
487            }
488
489            (Expr::Tpl(l), Expr::Tpl(rt)) => {
490                // We prepend the last quasis of l to the first quasis of r.
491                // After doing so, we can append all data of r to l.
492
493                {
494                    let l_last = l.quasis.pop().unwrap();
495                    let r_first = rt.quasis.first_mut().unwrap();
496
497                    let new_raw: Atom = format!("{}{}", l_last.raw, r_first.raw).into();
498                    r_first.raw = new_raw;
499
500                    let new_cooked = match (l_last.cooked, &r_first.cooked) {
501                        (Some(l_cooked), Some(r_cooked)) => {
502                            let mut c = Wtf8Buf::from(&l_cooked);
503                            c.push_wtf8(r_cooked);
504                            Some(c.into())
505                        }
506                        _ => None,
507                    };
508                    r_first.cooked = new_cooked;
509                }
510
511                l.quasis.extend(rt.quasis.take());
512                l.exprs.extend(rt.exprs.take());
513                // Remove r
514                r.take();
515
516                debug_assert!(l.quasis.len() == l.exprs.len() + 1, "{l:?} is invalid");
517                self.changed = true;
518                report_change!("strings: Merged two template literals");
519            }
520            _ => {}
521        }
522    }

524    ///
525    /// - `a + 'foo' + 'bar'` => `a + 'foobar'`
526    pub(super) fn concat_str(&mut self, e: &mut Expr) {
527        if let Expr::Bin(
528            bin @ BinExpr {
529                op: op!(bin, "+"), ..
530            },
531        ) = e
532        {
533            if let Expr::Bin(
534                left @ BinExpr {
535                    op: op!(bin, "+"), ..
536                },
537            ) = &mut *bin.left
538            {
539                let type_of_second = left.right.get_type(self.expr_ctx);
540                let type_of_third = bin.right.get_type(self.expr_ctx);
541
542                if let Value::Known(Type::Str) = type_of_second {
543                    if let Value::Known(Type::Str) = type_of_third {
544                        if let Value::Known(second_str) = left.right.as_pure_wtf8(self.expr_ctx) {
545                            if let Value::Known(third_str) = bin.right.as_pure_wtf8(self.expr_ctx) {
546                                #[cfg(feature = "debug")]
547                                let debug_second_str = second_str.clone();
548
549                                let new_str = second_str.into_owned() + &*third_str;
550                                let left_span = left.span;
551
552                                self.changed = true;
553                                report_change!(
554                                    "strings: Concatting `{:?} + {:?}` to `{:?}`",
555                                    debug_second_str,
556                                    third_str,
557                                    new_str
558                                );
559
560                                *e = BinExpr {
561                                    span: bin.span,
562                                    op: op!(bin, "+"),
563                                    left: left.left.take(),
564                                    right: Lit::Str(Str {
565                                        span: left_span,
566                                        raw: None,
567                                        value: new_str.into(),
568                                    })
569                                    .into(),
570                                }
571                                .into();
572                            }
573                        }
574                    }
575                }
576            }
577        }
578    }

580    pub(super) fn drop_useless_addition_of_str(&mut self, e: &mut Expr) {
581        if let Expr::Bin(BinExpr {
582            op: op!(bin, "+"),
583            left,
584            right,
585            ..
586        }) = e
587        {
588            let lt = left.get_type(self.expr_ctx);
589            let rt = right.get_type(self.expr_ctx);
590            if let Value::Known(Type::Str) = lt {
591                if let Value::Known(Type::Str) = rt {
592                    match &**left {
593                        Expr::Lit(Lit::Str(Str { value, .. })) if value.is_empty() => {
594                            self.changed = true;
595                            report_change!(
596                                "string: Dropping empty string literal (in lhs) because it does \
597                                 not changes type"
598                            );
599
600                            *e = *right.take();
601                            return;
602                        }
603                        _ => (),
604                    }
605
606                    match &**right {
607                        Expr::Lit(Lit::Str(Str { value, .. })) if value.is_empty() => {
608                            self.changed = true;
609                            report_change!(
610                                "string: Dropping empty string literal (in rhs) because it does \
611                                 not changes type"
612                            );
613
614                            *e = *left.take();
615                        }
616                        _ => (),
617                    }
618                }
619            }
620        }
621    }

624pub(super) fn convert_str_value_to_tpl_raw(value: &Wtf8) -> Cow<'_, str> {
625    let mut result = String::default();
626
627    let iter = value.code_points();
628    for code_point in iter {
629        if let Some(ch) = code_point.to_char() {
630            match ch {
631                '\\' => {
632                    result.push_str("\\\\");
633                }
634                '`' => {
635                    result.push_str("\\`");
636                }
637                '$' => {
638                    result.push_str("\\$");
639                }
640                '\n' => {
641                    result.push_str("\\n");
642                }
643                '\r' => {
644                    result.push_str("\\r");
645                }
646                _ => result.push(ch),
647            }
648        } else {
649            result.push_str(&format!("\\u{:04X}", code_point.to_u32()));
650        }
651    }
652
653    result.into()
654}

656pub(super) fn convert_str_raw_to_tpl_raw(value: &str) -> Atom {
657    value.replace('`', "\\`").replace('$', "\\$").into()
658}

723            module.visit_mut_with(&mut fixer(None));
724
725            // Generate code with target version
726            let mut buf = Vec::new();
727            {
728                let wr: Box<dyn WriteJs> = Box::new(omit_trailing_semi(Box::new(JsWriter::new(
729                    cm.clone(),
730                    "\n",
731                    &mut buf,
732                    None,
733                ))));
734
735                let mut emitter = Emitter {
736                    cfg: Config::default().with_target(target).with_minify(true),
737                    cm,
738                    comments: None,
739                    wr,
740                };
741
742                emitter.emit_module(&module).expect("failed to emit module");
743            }
744
745            let output = String::from_utf8(buf).expect("invalid utf8");
746            check(output);
747
748            Ok(())
749        })
750        .unwrap();
751    }
