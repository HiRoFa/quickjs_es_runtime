14/// \return an anchor instruction for a given IR anchor.
15fn make_anchor(anchor_type: ir::AnchorType, multiline: bool) -> Insn {
16    match anchor_type {
17        ir::AnchorType::StartOfLine => Insn::StartOfLine { multiline },
18        ir::AnchorType::EndOfLine => Insn::EndOfLine { multiline },
19    }
20}

22/// Weirdly placed optimization.
23/// If the given bracket can be represented as ASCII contents, return the
24/// bitmap. Otherwise nothing.
25fn bracket_as_ascii(bc: &BracketContents) -> Option<AsciiBitmap> {
26    let mut result = AsciiBitmap::default();
27    // We just assume that inverted brackets contain non-ASCII characters.
28    if bc.invert {
29        return None;
30    }
31    for r in bc.cps.intervals() {
32        debug_assert!(r.first <= r.last);
33        if r.last >= 128 {
34            return None;
35        }
36        for bit in r.first..=r.last {
37            result.set(bit as u8)
38        }
39    }
40    Some(result)
41}

43/// Type which wraps up the context needed to emit a CompiledRegex.
44struct Emitter {
45    result: CompiledRegex,
46
47    // Number of loops seen so far.
48    next_loop_id: LoopID,
49
50    // List of group names, in order, with empties for unnamed groups.
51    group_names: Vec<Box<str>>,
52
53    // Whether the current node is within a lookbehind.
54    in_lookbehind: bool,
55}
56
57impl Emitter {
58    /// Emit a ByteSet instruction.
59    /// We awkwardly optimize it like so.
60    fn emit_byte_set_insn(&mut self, bytes: &[u8]) {
61        let insn = match bytes.len() {
62            0 => Insn::JustFail,
63            1 => Insn::ByteSeq1(bytes.try_into().unwrap()),
64            2 => Insn::ByteSet2(ByteArraySet(bytes.try_into().unwrap())),
65            3 => Insn::ByteSet3(ByteArraySet(bytes.try_into().unwrap())),
66            4 => Insn::ByteSet4(ByteArraySet(bytes.try_into().unwrap())),
67            _ => panic!("Byte set is too long"),
68        };
69        self.emit_insn(insn);
70    }
71
72    // Emit a nonempty byte sequence instruction. The sequence must be at most MAX_BYTE_SEQ_LENGTH bytes.
73    fn emit_byte_sequence_insn(&mut self, seq: &[u8]) {
74        const {
75            assert!(
76                MAX_BYTE_SEQ_LENGTH == 16,
77                "Need to update our emitting logic"
78            );
79        };
80        let insn = match seq.len() {
81            1 => Insn::ByteSeq1(seq.try_into().unwrap()),
82            2 => Insn::ByteSeq2(seq.try_into().unwrap()),
83            3 => Insn::ByteSeq3(seq.try_into().unwrap()),
84            4 => Insn::ByteSeq4(seq.try_into().unwrap()),
85            5 => Insn::ByteSeq5(seq.try_into().unwrap()),
86            6 => Insn::ByteSeq6(seq.try_into().unwrap()),
87            7 => Insn::ByteSeq7(seq.try_into().unwrap()),
88            8 => Insn::ByteSeq8(seq.try_into().unwrap()),
89            9 => Insn::ByteSeq9(seq.try_into().unwrap()),
90            10 => Insn::ByteSeq10(seq.try_into().unwrap()),
91            11 => Insn::ByteSeq11(seq.try_into().unwrap()),
92            12 => Insn::ByteSeq12(seq.try_into().unwrap()),
93            13 => Insn::ByteSeq13(seq.try_into().unwrap()),
94            14 => Insn::ByteSeq14(seq.try_into().unwrap()),
95            15 => Insn::ByteSeq15(seq.try_into().unwrap()),
96            16 => Insn::ByteSeq16(seq.try_into().unwrap()),
97            _ => panic!("Unexpected chunk size"),
98        };
99        self.emit_insn(insn);
100    }
101
102    /// Emit an instruction.
103    /// Return the "instruction" as an index.
104    fn emit_insn(&mut self, insn: Insn) {
105        self.result.insns.push(insn);
106    }
107
108    /// Get an instruction at a given index.
109    fn get_insn(&mut self, idx: u32) -> &mut Insn {
110        &mut self.result.insns[idx as usize]
111    }
112
113    /// \return the offset of the next instruction emitted.
114    fn next_offset(&self) -> u32 {
115        self.result.insns.len() as u32
116    }
117
118    fn emit_insn_offset(&mut self, insn: Insn) -> u32 {
119        let ret = self.next_offset();
120        self.emit_insn(insn);
121        ret
122    }
123
124    /// Emit instructions corresponding to a given node.
125    fn emit_node(&mut self, node: &Node) {
126        enum Emitter<'a> {
127            Node(&'a Node),
128            NodeLoopFinish {
129                loop_instruction_index: u32,
130            },
131            NodeLookaroundAssertionFinish {
132                lookaround_instruction_index: u32,
133                prev_in_lookbehind: bool,
134            },
135            NodeAltMiddle {
136                alt_instruction_index: u32,
137                right_node: &'a Node,
138            },
139            NodeAltFinish {
140                alt_instruction_index: u32,
141                jump_instruction_index: u32,
142                right_branch_index: u32,
143            },
144            EndCaptureGroup {
145                group: CaptureGroupID,
146            },
147        }
148
149        let mut stack = vec![Emitter::Node(node)];
150
151        while let Some(inst) = stack.pop() {
152            match inst {
153                Emitter::NodeLoopFinish {
154                    loop_instruction_index,
155                } => {
156                    self.emit_insn(Insn::LoopAgain {
157                        begin: loop_instruction_index,
158                    });
159                    // Fix up our loop exit.
160                    let exit = self.next_offset();
161                    match self.get_insn(loop_instruction_index) {
162                        Insn::EnterLoop(fields) => fields.exit = exit,
163                        _ => panic!("Should be an EnterLoop instruction"),
164                    }
165                }
166                Emitter::NodeLookaroundAssertionFinish {
167                    lookaround_instruction_index,
168                    prev_in_lookbehind,
169                } => {
170                    self.emit_insn(Insn::Goal);
171                    // Fix up the continuation.
172                    let next_insn = self.next_offset();
173                    match self.get_insn(lookaround_instruction_index) {
174                        Insn::Lookbehind { continuation, .. } => *continuation = next_insn,
175                        Insn::Lookahead { continuation, .. } => *continuation = next_insn,
176                        _ => panic!("Should be a Lookaround instruction"),
177                    }
178                    self.in_lookbehind = prev_in_lookbehind;
179                }
180                Emitter::NodeAltMiddle {
181                    alt_instruction_index,
182                    right_node,
183                } => {
184                    let jump_insn = self.emit_insn_offset(Insn::Jump { target: 0 });
185                    let right_branch = self.next_offset();
186                    stack.push(Emitter::NodeAltFinish {
187                        alt_instruction_index,
188                        jump_instruction_index: jump_insn,
189                        right_branch_index: right_branch,
190                    });
191                    stack.push(Emitter::Node(right_node));
192                }
193                Emitter::NodeAltFinish {
194                    alt_instruction_index,
195                    jump_instruction_index,
196                    right_branch_index,
197                } => {
198                    let exit = self.next_offset();
199                    // Fix up our jump targets.
200                    match self.get_insn(alt_instruction_index) {
201                        Insn::Alt { secondary } => *secondary = right_branch_index,
202                        _ => panic!("Should be an Alt instruction"),
203                    }
204                    match self.get_insn(jump_instruction_index) {
205                        Insn::Jump { target } => *target = exit,
206                        _ => panic!("Should be a Jump instruction"),
207                    }
208                }
209                Emitter::EndCaptureGroup { group } => self.emit_insn(Insn::EndCaptureGroup(group)),
210                Emitter::Node(node) => match node {
211                    Node::Empty => {}
212                    Node::Goal => self.emit_insn(Insn::Goal),
213                    Node::Char { c, icase } => {
214                        let c = *c;
215                        if !*icase {
216                            self.emit_insn(Insn::Char(c))
217                        } else {
218                            core::debug_assert!(
219                                unicode::fold_code_point(c, self.result.flags.unicode) == c,
220                                "Char {:x} should be folded",
221                                c
222                            );
223                            self.emit_insn(Insn::CharICase(c))
224                        }
225                    }
226                    Node::Cat(children) => {
227                        for nn in children.iter().rev() {
228                            stack.push(Emitter::Node(nn));
229                        }
230                    }
231                    Node::Alt(left, right) => {
232                        // Alternation is followed by the primary branch and has a jump to secondary
233                        // branch. After primary branch, jump to the continuation.
234                        let alt_insn = self.emit_insn_offset(Insn::Alt { secondary: 0 });
235                        stack.push(Emitter::NodeAltMiddle {
236                            alt_instruction_index: alt_insn,
237                            right_node: right,
238                        });
239                        stack.push(Emitter::Node(left));
240                    }
241                    Node::Bracket(contents) => {
242                        if let Some(ascii_contents) = bracket_as_ascii(contents) {
243                            self.emit_insn(Insn::AsciiBracket(ascii_contents))
244                        } else {
245                            let idx = self.result.brackets.len();
246                            self.result.brackets.push(contents.clone());
247                            self.emit_insn(Insn::Bracket(idx))
248                        }
249                    }
250                    Node::MatchAny => self.emit_insn(Insn::MatchAny),
251                    Node::MatchAnyExceptLineTerminator => {
252                        self.emit_insn(Insn::MatchAnyExceptLineTerminator)
253                    }
254                    Node::Anchor {
255                        anchor_type,
256                        multiline,
257                    } => self.emit_insn(make_anchor(*anchor_type, *multiline)),
258                    Node::Loop {
259                        loopee,
260                        quant,
261                        enclosed_groups,
262                    } => {
263                        let loop_id = self.next_loop_id;
264                        self.next_loop_id += 1;
265                        let loop_insn = self.emit_insn_offset(Insn::EnterLoop(LoopFields {
266                            loop_id,
267                            min_iters: quant.min,
268                            // If the loop is unbounded, just emit usize::MAX,
269                            // as we cannot match more characters than that.
270                            max_iters: quant.max.unwrap_or(usize::MAX),
271                            greedy: quant.greedy,
272                            exit: 0,
273                        }));
274                        self.result.loops += 1;
275                        // Emit a sequence of ResetCaptureGroup for any contained groups.
276                        for gid in enclosed_groups.start..enclosed_groups.end {
277                            self.emit_insn(Insn::ResetCaptureGroup(gid))
278                        }
279                        stack.push(Emitter::NodeLoopFinish {
280                            loop_instruction_index: loop_insn,
281                        });
282                        stack.push(Emitter::Node(loopee));
283                    }
284                    Node::Loop1CharBody { loopee, quant } => {
285                        self.emit_insn(Insn::Loop1CharBody {
286                            min_iters: quant.min,
287                            max_iters: quant.max.unwrap_or(usize::MAX),
288                            greedy: quant.greedy,
289                        });
290                        stack.push(Emitter::Node(loopee));
291                    }
292                    Node::CaptureGroup { id, contents, name } => {
293                        let group = *id;
294                        self.result.groups += 1;
295                        self.group_names.push(name.as_deref().unwrap_or("").into());
296                        self.emit_insn(Insn::BeginCaptureGroup(group));
297                        stack.push(Emitter::EndCaptureGroup { group });
298                        stack.push(Emitter::Node(contents));
299                    }
300                    Node::LookaroundAssertion {
301                        negate,
302                        backwards,
303                        start_group,
304                        end_group,
305                        contents,
306                    } => {
307                        let lookaround = if *backwards {
308                            self.emit_insn_offset(Insn::Lookbehind {
309                                negate: *negate,
310                                start_group: *start_group,
311                                end_group: *end_group,
312                                continuation: 0,
313                            })
314                        } else {
315                            self.emit_insn_offset(Insn::Lookahead {
316                                negate: *negate,
317                                start_group: *start_group,
318                                end_group: *end_group,
319                                continuation: 0,
320                            })
321                        };
322                        let prev_in_lookbehind = self.in_lookbehind;
323                        self.in_lookbehind = *backwards;
324                        stack.push(Emitter::NodeLookaroundAssertionFinish {
325                            lookaround_instruction_index: lookaround,
326                            prev_in_lookbehind,
327                        });
328                        stack.push(Emitter::Node(contents));
329                    }
330                    &Node::WordBoundary { invert } => {
331                        if self.result.flags.unicode && self.result.flags.icase {
332                            self.emit_insn(Insn::WordBoundaryUnicodeICase { invert })
333                        } else {
334                            self.emit_insn(Insn::WordBoundary { invert })
335                        }
336                    }
337                    &Node::BackRef(group) => {
338                        debug_assert!(group >= 1, "Group should not be zero");
339                        // -1 because \1 matches the first capture group, which has index 0.
340                        self.emit_insn(Insn::BackRef(group - 1))
341                    }
342
343                    Node::ByteSet(bytes) => self.emit_byte_set_insn(bytes),
344
345                    Node::CharSet(chars) => {
346                        debug_assert!(chars.len() <= MAX_CHAR_SET_LENGTH);
347                        if chars.is_empty() {
348                            self.emit_insn(Insn::JustFail);
349                        } else {
350                            let mut arr = [chars[0]; MAX_CHAR_SET_LENGTH];
351                            arr[..chars.len()].copy_from_slice(chars.as_slice());
352                            self.emit_insn(Insn::CharSet(arr))
353                        }
354                    }
355
356                    Node::ByteSequence(bytes) => {
357                        // In a lookbehind, instruction order is reversed, but ByteSequence contents are not,
358                        // to take advantage of optimized memcmp. For example, `(?<=abcd)` may be emitted
359                        // as "d", "c", "b", "a" char matches (traversing input from right to left), or
360                        // optimized to a single forwards ByteSequence("abcd"), which matches left-to-right.
361                        // Therefore if we split a byte sequence into multiple instructions,
362                        // we need to emit them in reverse order iff in a lookbehind.
363                        let chunks = bytes.as_slice().chunks(MAX_BYTE_SEQ_LENGTH);
364                        if self.in_lookbehind {
365                            for chunk in chunks.rev() {
366                                self.emit_byte_sequence_insn(chunk);
367                            }
368                        } else {
369                            for chunk in chunks {
370                                self.emit_byte_sequence_insn(chunk);
371                            }
372                        }
373                    }
374                },
375            }
376        }
377    }
378}

380/// Compile the given IR to a CompiledRegex.
381pub fn emit(n: &ir::Regex) -> CompiledRegex {
382    let mut emitter = Emitter {
383        next_loop_id: 0,
384        group_names: Vec::new(),
385        in_lookbehind: false,
386        result: CompiledRegex {
387            insns: Vec::new(),
388            brackets: Vec::new(),
389            loops: 0,
390            groups: 0,
391            group_names: Box::new([]),
392            flags: n.flags,
393            start_pred: startpredicate::predicate_for_re(n),
394        },
395    };
396    emitter.emit_node(&n.node);
397    let mut result = emitter.result;
398    // Populate group names, unless all are empty.
399    debug_assert!(
400        result.group_names.is_empty(),
401        "Group names should not be set"
402    );
403    if emitter.group_names.iter().any(|s| !s.is_empty()) {
404        result.group_names = emitter.group_names.into_boxed_slice();
405    }
406    debug_assert!(
407        result.group_names.is_empty() || result.group_names.len() == result.groups as usize
408    );
409    result
