1//! Transforms TypeScript syntax into standard JavaScript.
2//!
3//! Use [`strip`] when the input was parsed as TypeScript.
4//! If the input was parsed with `swc_ecma_parser::Syntax::Flow`, call
5//! [`typescript`] with [`Config`] and set [`Config::flow_syntax`] to `true` so
6//! Flow-specific cleanup runs after stripping type-only syntax.
7//!
8//! ```no_run
9//! use swc_common::{sync::Lrc, FileName, Globals, Mark, SourceMap, GLOBALS};
10//! use swc_ecma_ast::EsVersion;
11//! use swc_ecma_codegen::to_code_default;
12//! use swc_ecma_parser::{parse_file_as_program, FlowSyntax, Syntax};
13//! use swc_ecma_transforms_base::{fixer::fixer, resolver};
14//! use swc_ecma_transforms_typescript::typescript;
15//!
16//! let cm: Lrc<SourceMap> = Default::default();
17//! let fm = cm.new_source_file(
18//!     FileName::Custom("input.js".into()).into(),
19//!     "const value: number = 1;",
20//! );
21//! let mut recovered_errors = Vec::new();
22//!
23//! let program = parse_file_as_program(
24//!     &fm,
25//!     Syntax::Flow(FlowSyntax::default()),
26//!     EsVersion::latest(),
27//!     None,
28//!     &mut recovered_errors,
29//! )
30//! .expect("flow should parse");
31//! assert!(recovered_errors.is_empty());
32//!
33//! GLOBALS.set(&Globals::default(), || {
34//!     let unresolved_mark = Mark::new();
35//!     let top_level_mark = Mark::new();
36//!
37//!     let program = program
38//!         .apply(resolver(unresolved_mark, top_level_mark, false))
39//!         .apply(typescript(
40//!             typescript::Config {
41//!                 flow_syntax: true,
42//!                 ..Default::default()
43//!             },
44//!             unresolved_mark,
45//!             top_level_mark,
46//!         ))
47//!         .apply(fixer(None));
48//!
49//!     let output = to_code_default(cm, None, &program);
50//!     assert_eq!(output, "const value = 1;\n");
51//! });
52//! ```
