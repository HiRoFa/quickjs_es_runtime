31/// ## This Module will transform all TypeScript specific synatx
32///
33/// - ### [namespace]/[modules]/[enums]
34/// - ### class constructor [parameter properties]
35///
36///     ```TypeScript
37///     class Foo {
38///         constructor(public x: number) {
39///             // No body necessary
40///         }
41///     }
42///     ```
43/// - ### [export and import require]
44///
45///     ```TypeScript
46///     import foo = require("foo");
47///     export = foo;
48///     ```
49///
50/// [namespace]: https://www.typescriptlang.org/docs/handbook/namespaces.html
51/// [modules]: https://www.typescriptlang.org/docs/handbook/modules.html
52/// [enums]: https://www.typescriptlang.org/docs/handbook/enums.html
53/// [parameter properties]: https://www.typescriptlang.org/docs/handbook/2/classes.html#parameter-properties
54/// [export and import require]: https://www.typescriptlang.org/docs/handbook/modules.html#export--and-import--require
55#[derive(Default)]

124    fn visit_mut_program(&mut self, node: &mut Program) {
125        if !self.semantic.exported_binding.is_empty() {
126            self.ref_rewriter = Some(RefRewriter {

135    fn visit_mut_module(&mut self, node: &mut Module) {
136        if !self.verbatim_module_syntax {
137            self.strip_module_items_with_semantic(&mut node.body);
138        }
139
140        self.visit_mut_for_ts_import_export(node);
141
142        node.visit_mut_children_with(self);
143
144        if !self.export_var_list.is_empty() {
145            let decls = self
146                .export_var_list
147                .take()
148                .into_iter()
149                .map(id_to_var_declarator)
150                .collect();
151
152            node.body.push(
153                ExportDecl {
154                    decl: VarDecl {
155                        decls,
156                        ..Default::default()
157                    }
158                    .into(),
159                    span: DUMMY_SP,
160                }
161                .into(),
162            )
163        }
164
165        if self.flow_syntax {
166            self.normalize_module_await_bindings(node);
167        }
168    }

170    fn visit_mut_module_items(&mut self, node: &mut Vec<ModuleItem>) {
171        let var_list = self.var_list.take();
172        node.retain(|item| should_retain_module_item(item, self.in_namespace));
173        node.retain_mut(|item| {
174            let is_empty = item.as_stmt().map(Stmt::is_empty).unwrap_or(false);
175            item.visit_mut_with(self);
176            // Remove those folded into Empty
177            is_empty || !item.as_stmt().map(Stmt::is_empty).unwrap_or(false)
178        });
179        let var_list = mem::replace(&mut self.var_list, var_list);
180
181        if !var_list.is_empty() {
182            let decls = var_list.into_iter().map(id_to_var_declarator).collect();
183
184            node.push(
185                VarDecl {
186                    decls,
187                    ..Default::default()
188                }
189                .into(),
190            )
191        }
192    }

194    fn visit_mut_class_members(&mut self, node: &mut Vec<ClassMember>) {
195        let prop_list = self.in_class_prop.take();
196        let init_list = self.in_class_prop_init.take();
197
198        node.retain(|member| match member {
199            ClassMember::TsIndexSignature(..) => false,
200            ClassMember::Constructor(Constructor { body: None, .. }) => false,
201            ClassMember::Method(ClassMethod {
202                is_abstract,
203                function,
204                ..
205            })
206            | ClassMember::PrivateMethod(PrivateMethod {
207                is_abstract,
208                function,
209                ..
210            }) => !is_abstract && function.body.is_some(),
211            ClassMember::ClassProp(
212                ClassProp { declare: true, .. }
213                | ClassProp {
214                    is_abstract: true, ..
215                },
216            )
217            | ClassMember::AutoAccessor(AutoAccessor {
218                is_abstract: true, ..
219            }) => false,
220            _ => true,
221        });
222
223        node.visit_mut_children_with(self);
224        let prop_list = mem::replace(&mut self.in_class_prop, prop_list);
225        let init_list = mem::replace(&mut self.in_class_prop_init, init_list);
226
227        if !prop_list.is_empty() {
228            if self.native_class_properties {
229                self.reorder_class_prop_decls(node, prop_list, init_list);
230            } else {
231                self.reorder_class_prop_decls_and_inits(node, prop_list, init_list);
232            }
233        }
234    }

236    fn visit_mut_constructor(&mut self, node: &mut Constructor) {
237        node.accessibility = None;
238
239        node.params
240            .iter_mut()
241            .for_each(|param_or_ts_param_prop| match param_or_ts_param_prop {
242                ParamOrTsParamProp::TsParamProp(ts_param_prop) => {
243                    let TsParamProp {
244                        span,
245                        decorators,
246                        param,
247                        ..
248                    } = ts_param_prop;
249
250                    let (pat, expr, id) = match param {
251                        TsParamPropParam::Ident(binding_ident) => {
252                            let id = binding_ident.to_id();
253                            let prop_name = PropName::Ident(IdentName::from(&*binding_ident));
254                            let mut value_ident = Ident::from(&*binding_ident);
255                            value_ident.optional = false;
256                            let value = value_ident.into();
257
258                            (
259                                binding_ident.clone().into(),
260                                assign_value_to_this_prop(prop_name, value),
261                                id,
262                            )
263                        }
264                        TsParamPropParam::Assign(assign_pat) => {
265                            let AssignPat { left, .. } = &assign_pat;
266
267                            let Pat::Ident(binding_ident) = &**left else {
268                                unreachable!("destructuring pattern inside TsParameterProperty");
269                            };
270
271                            let id = binding_ident.id.to_id();
272                            let prop_name = PropName::Ident(binding_ident.id.clone().into());
273                            let mut value_ident = binding_ident.id.clone();
274                            value_ident.optional = false;
275                            let value = value_ident.into();
276
277                            (
278                                assign_pat.clone().into(),
279                                assign_value_to_this_prop(prop_name, value),
280                                id,
281                            )
282                        }
283                        #[cfg(swc_ast_unknown)]
284                        _ => panic!("unable to access unknown nodes"),
285                    };
286
287                    self.in_class_prop.push(id);
288                    self.in_class_prop_init.push(expr);
289
290                    *param_or_ts_param_prop = Param {
291                        span: *span,
292                        decorators: decorators.take(),
293                        pat,
294                    }
295                    .into();
296                }
297                ParamOrTsParamProp::Param(..) => {}
298                #[cfg(swc_ast_unknown)]
299                _ => panic!("unable to access unknown nodes"),
300            });
301
302        node.params.visit_mut_children_with(self);
303        node.body.visit_mut_with(self);
304    }

306    fn visit_mut_stmts(&mut self, node: &mut Vec<Stmt>) {
307        let var_list = self.var_list.take();
308        node.retain_mut(|stmt| {
309            let is_empty = stmt.is_empty();
310            stmt.visit_mut_with(self);
311            // Remove those folded into Empty
312            is_empty || !stmt.is_empty()
313        });
314        let var_list = mem::replace(&mut self.var_list, var_list);
315        if !var_list.is_empty() {
316            let decls = var_list.into_iter().map(id_to_var_declarator).collect();
317            node.push(
318                VarDecl {
319                    decls,
320                    ..Default::default()
321                }
322                .into(),
323            )
324        }
325
326        node.retain(|stmt| !matches!(stmt, Stmt::Empty(empty_stmt) if empty_stmt.span.is_dummy()));
327    }

348    fn visit_mut_stmt(&mut self, node: &mut Stmt) {
349        if !should_retain_stmt(node) {
350            if !node.is_empty() {
351                node.take();
352            }
353            return;
354        }
355
356        node.visit_mut_children_with(self);
357
358        let Stmt::Decl(decl) = node else {
359            return;
360        };
361
362        match self.fold_decl(decl.take(), false) {
363            FoldedDecl::Decl(var_decl) => *decl = var_decl,
364            FoldedDecl::Expr(stmt) => *node = stmt,
365            FoldedDecl::Empty => {
366                node.take();
367            }
368        }
369    }

372        node.visit_mut_children_with(self);
373
374        if let Some(ExportDecl { decl, .. }) = node

378            match self.fold_decl(decl.take(), true) {
379                FoldedDecl::Decl(var_decl) => *decl = var_decl,
380                FoldedDecl::Expr(stmt) => *node = stmt.into(),
381                FoldedDecl::Empty => {

404    fn visit_mut_export_decl(&mut self, node: &mut ExportDecl) {
405        if self.ref_rewriter.is_some() {
406            if let Decl::Var(var_decl) = &mut node.decl {
407                // visit inner directly to bypass visit_mut_var_declarator
408                for decl in var_decl.decls.iter_mut() {
409                    if self.flow_syntax {

411                    }
412                    decl.name.visit_mut_with(self);
413                    decl.init.visit_mut_with(self);
414                }
415                return;
416            }

418        node.visit_mut_children_with(self);

448    fn visit_mut_expr(&mut self, node: &mut Expr) {
449        while let Expr::TsAs(TsAsExpr { expr, .. })
450        | Expr::TsNonNull(TsNonNullExpr { expr, .. })
451        | Expr::TsTypeAssertion(TsTypeAssertion { expr, .. })
452        | Expr::TsConstAssertion(TsConstAssertion { expr, .. })
453        | Expr::TsInstantiation(TsInstantiation { expr, .. })
454        | Expr::TsSatisfies(TsSatisfiesExpr { expr, .. }) = node
455        {
456            *node = *expr.take();
457        }
458
459        self.enter_expr_for_inline_enum(node);
460
461        maybe_grow_default(|| node.visit_mut_children_with(self));
462
463        if let Some(ref_rewriter) = self.ref_rewriter.as_mut() {
464            ref_rewriter.exit_expr(node);
465        }
466    }

733    fn strip_module_items_with_semantic(&self, items: &mut Vec<ModuleItem>) {
734        items.retain_mut(|module_item| match module_item {
735            ModuleItem::ModuleDecl(ModuleDecl::Import(ImportDecl {
736                specifiers,
737                type_only: false,
738                ..
739            })) if !specifiers.is_empty() => {
740                // Match tsc behavior: keep empty original imports, but strip specifiers by
741                // usage.
742                specifiers.retain(|import_specifier| match import_specifier {
743                    ImportSpecifier::Named(named) => {
744                        if named.is_type_only {
745                            return false;
746                        }
747
748                        let id = named.local.to_id();
749

751                            return false;
752                        }

754                        self.semantic.has_usage(&id)
755                    }
756                    ImportSpecifier::Default(default) => {
757                        let id = default.local.to_id();

759                        if self.semantic.has_value(&id) {
760                            return false;
761                        }
762
763                        self.semantic.has_usage(&id)
764                    }
765                    ImportSpecifier::Namespace(namespace) => {
766                        let id = namespace.local.to_id();
767
768                        if self.semantic.has_value(&id) {
769                            return false;
770                        }
771
772                        self.semantic.has_usage(&id)
773                    }
774                    #[cfg(swc_ast_unknown)]
775                    _ => panic!("unable to access unknown nodes"),
776                });
777
778                self.import_not_used_as_values == crate::ImportsNotUsedAsValues::Preserve
779                    || !specifiers.is_empty()
780            }
781            ModuleItem::ModuleDecl(ModuleDecl::ExportNamed(NamedExport {
782                specifiers,
783                src,
784                type_only: false,

788                    ExportSpecifier::Namespace(..) | ExportSpecifier::Default(..) => true,
789                    ExportSpecifier::Named(ExportNamedSpecifier {
790                        orig: ModuleExportName::Ident(ident),
791                        is_type_only: false,
792                        ..
793                    }) if src.is_none() => !self.semantic.has_pure_type(&ident.to_id()),

796                    }

799                });

805            }

808                ..

811                .map(|ident| !self.semantic.has_pure_type(&ident.to_id()))

819                    return true;

823            }

826                    self.strip_namespace_body_with_semantic(body);
827                }

872    fn fold_decl(&mut self, node: Decl, is_export: bool) -> FoldedDecl {
873        match node {
874            Decl::TsModule(ts_module) => {
875                let id = ts_module.id.to_id();
876
877                if self.decl_id_record.insert(id.clone()) {
878                    if is_export {
879                        if self.namespace_id.is_none() {
880                            self.export_var_list.push(id);
881                        }
882                    } else {
883                        self.var_list.push(id);
884                    }
885                }

889            Decl::TsEnum(ts_enum) => {

893
894                self.transform_ts_enum(*ts_enum, is_first, is_export)

897                self.decl_id_record.insert(ident.to_id());

899            }
900            decl => FoldedDecl::Decl(decl),
901        }

904    fn normalize_module_await_bindings(&mut self, module: &mut Module) {
905        let mut collector = AwaitBindingCollector::default();
906        module.visit_with(&mut collector);
907
908        if collector.await_binding_ids.is_empty() {
909            return;
910        }
911
912        let mut used_syms = collector.used_syms;
913        let mut rename_map = FxHashMap::default();
914        let mut suffix = 0usize;
915
916        for id in collector.await_binding_ids {
917            let next_sym = loop {
918                let candidate: Atom = if suffix == 0 {
919                    "_await".into()
920                } else {
921                    format!("_await{suffix}").into()
922                };
923                suffix += 1;
924
925                if used_syms.insert(candidate.clone()) {
926                    break candidate;
927                }
928            };
929
930            rename_map.insert(id, next_sym);
931        }
932
933        if !rename_map.is_empty() {
934            module.visit_mut_with(&mut rename(&rename_map));
935        }
936    }

1016    namespace_id: Option<&'a Id>,

1019impl InitArg<'_> {
1020    // {}
1021    fn empty() -> ExprOrSpread {
1022        ExprOrSpread {
1023            spread: None,
1024            expr: ObjectLit::default().into(),
1025        }
1026    }
1027
1028    // N
1029    fn get(&self) -> ExprOrSpread {
1030        self.namespace_id
1031            .cloned()
1032            .map_or_else(
1033                || -> Expr { self.id.clone().into() },
1034                |namespace_id| namespace_id.make_member(self.id.clone().into()).into(),
1035            )
1036            .into()
1037    }
1038
1039    // N || {}
1040    fn or_empty(&self) -> ExprOrSpread {
1041        let expr = self.namespace_id.cloned().map_or_else(
1042            || -> Expr { self.id.clone().into() },
1043            |namespace_id| namespace_id.make_member(self.id.clone().into()).into(),
1044        );
1045
1046        let bin = BinExpr {
1047            op: op!("||"),
1048            left: expr.into(),
1049            right: ObjectLit::default().into(),
1050            ..Default::default()
1051        };
1052
1053        ExprOrSpread {
1054            spread: None,
1055            expr: bin.into(),
1056        }
1057    }
1058
1059    // N || (N = {})
1060    fn or_assign_empty(&self) -> ExprOrSpread {
1061        let expr = self.namespace_id.cloned().map_or_else(
1062            || -> Expr { self.id.clone().into() },
1063            |namespace_id| namespace_id.make_member(self.id.clone().into()).into(),
1064        );
1065
1066        let assign = self.namespace_id.cloned().map_or_else(
1067            || ObjectLit::default().make_assign_to(op!("="), self.id.clone().into()),
1068            |namespace_id| {
1069                ObjectLit::default().make_assign_to(
1070                    op!("="),
1071                    namespace_id.make_member(self.id.clone().into()).into(),
1072                )
1073            },
1074        );
1075
1076        let bin = BinExpr {
1077            op: op!("||"),
1078            left: expr.into(),
1079            right: assign.into(),
1080            ..Default::default()
1081        };
1082
1083        ExprOrSpread {
1084            spread: None,
1085            expr: bin.into(),
1086        }
1087    }

1090impl Transform {
1091    fn transform_ts_enum(
1092        &mut self,
1093        ts_enum: TsEnumDecl,
1094        is_first: bool,
1095        is_export: bool,
1096    ) -> FoldedDecl {
1097        let TsEnumDecl {
1098            span,
1099            declare,
1100            is_const,
1101            id,
1102            members,
1103        } = ts_enum;
1104
1105        debug_assert!(!declare);
1106
1107        let ts_enum_safe_remove = !self.verbatim_module_syntax
1108            && is_const
1109            && !is_export
1110            && !self.semantic.exported_binding.contains_key(&id.to_id());
1111
1112        let member_names = self
1113            .semantic
1114            .enum_record
1115            .keys()
1116            .filter(|k| k.enum_id == id.to_id())
1117            .map(|k| k.member_name.clone())
1118            .collect();
1119
1120        let enum_computer = EnumValueComputer {
1121            enum_id: &id.to_id(),
1122            unresolved_ctxt: self.unresolved_ctxt,
1123            record: &self.semantic.enum_record,
1124            const_vars: &self.semantic.const_vars,
1125            const_enum_only: None,
1126            ambient_record: &self.semantic.ambient_enum_record,
1127            ambient_const_enum_only: None,
1128        };
1129
1130        let member_list: Vec<_> = members
1131            .into_iter()
1132            .map(|m| {
1133                let span = m.span;
1134                let name = enum_member_name(&m.id);
1135
1136                let key = TsEnumRecordKey {
1137                    enum_id: id.to_id(),
1138                    member_name: name.clone(),
1139                };
1140
1141                let mut value = self.semantic.enum_record.get(&key).unwrap().clone();
1142
1143                if matches!(value, TsEnumRecordValue::Opaque(..)) {
1144                    if let Some(init) = m.init {
1145                        // Recompute from the original initializer so enum member
1146                        // references can be rewritten to runtime property
1147                        // accesses. Implicit Flow enum members do not have an
1148                        // initializer, so keep the semantic value as-is.
1149                        let mut recomputed = enum_computer.compute(init, EvalCtx::RECOMPUTE);
1150                        if let TsEnumRecordValue::Opaque(expr) = &mut recomputed {
1151                            expr.visit_mut_with(&mut RefRewriter {
1152                                query: EnumMemberRefQuery {
1153                                    enum_id: &id.to_id(),
1154                                    member_names: &member_names,
1155                                    unresolved_ctxt: self.unresolved_ctxt,
1156                                },
1157                            });
1158                        }
1159                        value = recomputed;
1160                    }
1161                }
1162
1163                EnumMemberItem { span, name, value }
1164            })
1165            .filter(|m| !ts_enum_safe_remove || !m.is_const())
1166            .collect();
1167
1168        if member_list.is_empty() && is_const {
1169            return FoldedDecl::Empty;
1170        }
1171
1172        let opaque = member_list
1173            .iter()
1174            .any(|item| matches!(item.value, TsEnumRecordValue::Opaque(..)));
1175
1176        let stmts = member_list
1177            .into_iter()
1178            .filter(|item| !ts_enum_safe_remove || !item.is_const())
1179            .map(|item| item.build_assign(&id.to_id()));
1180
1181        let namespace_export = self.namespace_id.is_some() && is_export;
1182        let iife = !is_first || namespace_export;
1183
1184        let body = if !iife {
1185            let return_stmt: Stmt = ReturnStmt {
1186                arg: Some(id.clone().into()),
1187                ..Default::default()
1188            }
1189            .into();
1190
1191            let stmts = stmts.chain(iter::once(return_stmt)).collect();
1192
1193            BlockStmt {
1194                stmts,
1195                ..Default::default()
1196            }
1197        } else {
1198            BlockStmt {
1199                stmts: stmts.collect(),
1200                ..Default::default()
1201            }
1202        };
1203
1204        let var_kind = if is_export || id.ctxt == self.top_level_ctxt {
1205            VarDeclKind::Var
1206        } else {
1207            VarDeclKind::Let
1208        };
1209
1210        let init_arg = 'init_arg: {
1211            let init_arg = InitArg {
1212                id: &id,
1213                namespace_id: self.namespace_id.as_ref().filter(|_| is_export),
1214            };
1215            if !is_first {
1216                break 'init_arg init_arg.get();
1217            }
1218
1219            if namespace_export {
1220                break 'init_arg init_arg.or_assign_empty();
1221            }
1222
1223            if is_export || var_kind == VarDeclKind::Let {
1224                InitArg::empty()
1225            } else {
1226                init_arg.or_empty()
1227            }
1228        };
1229
1230        let expr = Factory::function(vec![id.clone().into()], body).as_call(
1231            if iife || opaque { DUMMY_SP } else { PURE_SP },
1232            vec![init_arg],
1233        );
1234
1235        if iife {
1236            FoldedDecl::Expr(
1237                ExprStmt {
1238                    span,
1239                    expr: expr.into(),
1240                }
1241                .into(),
1242            )
1243        } else {
1244            let var_declarator = VarDeclarator {
1245                span,
1246                name: id.into(),
1247                init: Some(expr.into()),
1248                definite: false,
1249            };
1250
1251            FoldedDecl::Decl(
1252                VarDecl {
1253                    span,
1254                    kind: var_kind,
1255                    decls: vec![var_declarator],
1256                    ..Default::default()
1257                }
1258                .into(),
1259            )
1260        }
1261    }
1262}

1294    fn transform_ts_namespace_body(id: Id, body: TsNamespaceBody) -> BlockStmt {
1295        let TsNamespaceDecl {
1296            span,
1297            declare,
1298            global,
1299            id: local_name,
1300            body,
1301        } = match body {
1302            TsNamespaceBody::TsModuleBlock(ts_module_block) => {
1303                return Self::transform_ts_module_block(id, ts_module_block);
1304            }
1305            TsNamespaceBody::TsNamespaceDecl(ts_namespace_decl) => ts_namespace_decl,
1306            #[cfg(swc_ast_unknown)]
1307            _ => panic!("unable to access unknown nodes"),
1308        };
1309
1310        debug_assert!(!declare);
1311        debug_assert!(!global);
1312
1313        let body = Self::transform_ts_namespace_body(local_name.to_id(), *body);
1314
1315        let init_arg = InitArg {
1316            id: &local_name,
1317            namespace_id: Some(&id.to_id()),
1318        }
1319        .or_assign_empty();
1320
1321        let expr =
1322            Factory::function(vec![local_name.into()], body).as_call(DUMMY_SP, vec![init_arg]);
1323
1324        BlockStmt {
1325            span,
1326            stmts: vec![expr.into_stmt()],
1327            ..Default::default()
1328        }
1329    }
1330
1331    /// Note:
1332    /// All exported variable declarations are transformed into assignment to
1333    /// the namespace. All references to the exported binding will be
1334    /// replaced with qualified access to the namespace property.
1335    ///
1336    /// Exported function and class will be treat as const exported which is in
1337    /// line with how the TypeScript compiler handles exports.
1338    ///
1339    /// Inline exported syntax should not be used with function which will lead
1340    /// to issues with function hoisting.
1341    ///
1342    /// Input:
1343    /// ```TypeScript
1344    /// export const foo = init, { bar: baz = init } = init;
1345    ///
1346    /// export function a() {}
1347    ///
1348    /// export let b = init;
1349    /// ```
1350    ///
1351    /// Output:
1352    /// ```TypeScript
1353    /// NS.foo = init, { bar: NS.baz = init } = init;
1354    ///
1355    /// function a() {}
1356    /// NS.a = a;
1357    ///
1358    /// NS.b = init;
1359    /// ```
1360    fn transform_ts_module_block(id: Id, TsModuleBlock { span, body }: TsModuleBlock) -> BlockStmt {
1361        let mut stmts = Vec::new();
1362
1363        for module_item in body {
1364            match module_item {
1365                ModuleItem::Stmt(stmt) => stmts.push(stmt),
1366                ModuleItem::ModuleDecl(ModuleDecl::ExportDecl(ExportDecl {
1367                    decl, span, ..
1368                })) => match decl {
1369                    Decl::Class(ClassDecl { ref ident, .. })
1370                    | Decl::Fn(FnDecl { ref ident, .. }) => {
1371                        let assign_stmt = Self::assign_prop(&id, ident, span);
1372                        stmts.push(decl.into());
1373                        stmts.push(assign_stmt);
1374                    }
1375                    Decl::Var(var_decl) => {
1376                        let mut exprs: Vec<Box<_>> = var_decl
1377                            .decls
1378                            .into_iter()
1379                            .flat_map(
1380                                |VarDeclarator {
1381                                     span, name, init, ..
1382                                 }| {
1383                                    let right = init?;
1384                                    let left = name.try_into().unwrap();
1385
1386                                    Some(
1387                                        AssignExpr {
1388                                            span,
1389                                            left,
1390                                            op: op!("="),
1391                                            right,
1392                                        }
1393                                        .into(),
1394                                    )
1395                                },
1396                            )
1397                            .collect();
1398
1399                        if exprs.is_empty() {
1400                            continue;
1401                        }
1402
1403                        let expr = if exprs.len() == 1 {
1404                            exprs.pop().unwrap()
1405                        } else {
1406                            SeqExpr {
1407                                span: DUMMY_SP,
1408                                exprs,
1409                            }
1410                            .into()
1411                        };

1414                            ExprStmt {
1415                                span: var_decl.span,
1416                                expr,
1417                            }
1418                            .into(),
1419                        );
1420                    }
1421                    decl => unreachable!("{decl:?}"),
1422                },
1423                ModuleItem::ModuleDecl(ModuleDecl::TsImportEquals(decl)) => {

1425                        TsModuleRef::TsEntityName(ts_entity_name) => {

1428                            // export impot foo = bar.baz
1429                            let stmt = if decl.is_export {
1430                                // Foo.foo = bar.baz
1431                                let left = id.clone().make_member(decl.id.clone().into());
1432                                let expr = init.make_assign_to(op!("="), left.into());
1433
1434                                ExprStmt {
1435                                    span: decl.span,
1436                                    expr: expr.into(),
1437                                }
1438                                .into()
1439                            } else {
1440                                // const foo = bar.baz
1441                                let mut var_decl =
1442                                    init.into_var_decl(VarDeclKind::Const, decl.id.clone().into());
1443
1444                                var_decl.span = decl.span;
1445
1446                                var_decl.into()
1447                            };
1448
1449                            stmts.push(stmt);
1450                        }
1451                        TsModuleRef::TsExternalModuleRef(..) => {
1452                            // TS1147
1453                            if HANDLER.is_set() {
1454                                HANDLER.with(|handler| {
1455                                    handler
1456                                    .struct_span_err(
1457                                        decl.span,
1458                                        r#"Import declarations in a namespace cannot reference a module."#,
1459                                    )
1460                                    .emit();
1461                                });
1462                            }
1463                        }
1464                        #[cfg(swc_ast_unknown)]
1465                        _ => panic!("unable to access unknown nodes"),
1466                    }
1467                }
1468                item => {
1469                    if HANDLER.is_set() {
1470                        HANDLER.with(|handler| {
1471                            handler
1472                                .struct_span_err(
1473                                    item.span(),
1474                                    r#"ESM-style module declarations are not permitted in a namespace."#,
1475                                )
1476                                .emit();
1477                        });
1478                    }
1479                }
1480            }
1481        }
1482
1483        BlockStmt {
1484            span,
1485            stmts,
1486            ..Default::default()
1487        }
1488    }

1492    fn reorder_class_prop_decls_and_inits(
1493        &mut self,
1494        class_member_list: &mut Vec<ClassMember>,
1495        prop_list: Vec<Id>,
1496        mut init_list: Vec<Box<Expr>>,
1497    ) {
1498        let mut constructor = None;
1499        let mut cons_index = 0;
1500        for (index, member) in class_member_list.iter_mut().enumerate() {
1501            match member {
1502                ClassMember::Constructor(..) => {
1503                    let empty = EmptyStmt {
1504                        span: member.span(),
1505                    }
1506                    .into();
1507                    constructor = mem::replace(member, empty).constructor();
1508                    cons_index = index;
1509                }
1510                ClassMember::ClassProp(ClassProp {
1511                    key,
1512                    value: value @ Some(..),
1513                    is_static: false,
1514                    span,
1515                    ..
1516                }) => {
1517                    let key = match &mut *key {
1518                        PropName::Computed(ComputedPropName { span, expr })
1519                            if !is_literal(expr) =>
1520                        {

1522
1523                            self.var_list.push(ident.to_id());
1524
1525                            **expr = expr.take().make_assign_to(op!("="), ident.clone().into());
1526
1527                            PropName::Computed(ComputedPropName {
1528                                span: *span,
1529                                expr: ident.into(),
1530                            })
1531                        }
1532                        _ => key.clone(),
1533                    };
1534
1535                    let mut init = assign_value_to_this_prop(key, *value.take().unwrap());
1536                    init.set_span(*span);
1537

1539                }

1541                    key,
1542                    value: value @ Some(..),
1543                    is_static: false,

1570            0..0,
1571            prop_list
1572                .into_iter()
1573                .map(Ident::from)

1575                .map(|key| ClassProp {
1576                    key,
1577                    ..Default::default()
1578                })
1579                .map(ClassMember::ClassProp),
1580        );
1581    }

2009/// Gives a Flow component-typed arrow binding the function semantics required
2010/// by Flow's component runtime contract.
2011fn convert_flow_component_arrow(declarator: &mut VarDeclarator) {
2012    let Pat::Ident(BindingIdent {
2013        id,
2014        type_ann: Some(type_ann),
2015    }) = &declarator.name
2016    else {
2017        return;
2018    };
2019
2020    if !is_flow_component_type(&type_ann.type_ann) {
2021        return;
2022    }
2023
2024    let Some(init) = &mut declarator.init else {
2025        return;
2026    };
2027    let Expr::Arrow(arrow) = init.as_mut() else {
2028        return;
2029    };
2030
2031    let arrow = arrow.take();
2032    let body = match *arrow.body {
2033        ArrowFunctionBody::FunctionBody(body) => body,
2034        ArrowFunctionBody::Expr(expr) => {
2035            let span = expr.span();
2036            FunctionBody {
2037                span,
2038                stmts: vec![Stmt::Return(ReturnStmt {
2039                    span,
2040                    arg: Some(expr),
2041                })],
2042            }
2043        }
2044        #[cfg(swc_ast_unknown)]
2045        _ => panic!("unable to access unknown nodes"),
2046    };
2047
2048    **init = Expr::Fn(FnExpr {
2049        ident: Some(id.clone()),
2050        function: Box::new(Function {
2051            this_param: None,
2052            params: arrow
2053                .params
2054                .into_iter()
2055                .map(|pat| Param {
2056                    span: pat.span(),
2057                    decorators: Vec::new(),
2058                    pat,
2059                })
2060                .collect(),
2061            decorators: Vec::new(),
2062            span: arrow.span,
2063            ctxt: arrow.ctxt,
2064            body: Some(body),
2065            is_generator: arrow.is_generator,
2066            is_async: arrow.is_async,
2067            type_params: arrow.type_params,
2068            return_type: arrow.return_type,
2069        }),
2070    });
2071}
