33/// The kind of error that occurred.
34#[derive(Debug, Clone, Copy, PartialEq, Eq, displaydoc::Display)]

36pub enum ParseErrorKind {
37    /// An unexpected character was encountered.
38    ///
39    /// This variant implies the other variants
40    /// (notably `UnknownProperty` and `Unimplemented`) do not apply.
41    #[displaydoc("An unexpected character was encountered")]
42    UnexpectedChar(char),
43    /// The property name or value is unknown.
44    ///
45    /// For property names, make sure you use the spelling
46    /// defined in [ECMA-262](https://tc39.es/ecma262/#table-nonbinary-unicode-properties).
47    #[displaydoc("The property name or value is unknown")]
48    UnknownProperty,
49    /// A reference to an unknown variable.
50    UnknownVariable,
51    /// A variable of a certain type occurring in an unexpected context.
52    UnexpectedVariable,
53    /// The source is an incomplete unicode set.
54    Eof,
55    /// Something unexpected went wrong with our code. Please file a bug report on GitHub.
56    Internal,
57    /// The provided syntax is not supported by us.
58    ///
59    /// Note that unknown properties will return the
60    /// `UnknownProperty` variant, not this one.
61    #[displaydoc("The provided syntax is not supported by us.")]
62    Unimplemented,
63    /// The provided escape sequence is not a valid Unicode code point or represents too many code points.
64    InvalidEscape,

84/// The error type returned by the `parse` functions in this crate.
85///
86/// See [`ParseError::fmt_with_source`] for pretty-printing and [`ParseErrorKind`] of the
87/// different types of errors represented by this struct.
88#[derive(Debug, Clone, Copy, PartialEq, Eq)]
89pub struct ParseError {
90    // offset is the index to an arbitrary byte in the last character in the source that makes sense
91    // to display as location for the error, e.g., the unexpected character itself or
92    // for an unknown property name the last character of the name.
93    offset: Option<usize>,
94    kind: ParseErrorKind,
95}
96
97type Result<T, E = ParseError> = core::result::Result<T, E>;

99impl ParseError {
100    /// Pretty-prints this error and if applicable, shows where the error occurred in the source.
101    ///
102    /// Must be called with the same source that was used to parse the set.
103    ///
104    /// # Examples
105    ///
106    /// ```
107    /// use icu::properties::unicodeset_parse::*;
108    ///
109    /// let source = "[[abc]-x]";
110    /// let set = parse(source);
111    /// assert!(set.is_err());
112    /// let err = set.unwrap_err();
113    /// assert_eq!(
114    ///     err.fmt_with_source(source).to_string(),
115    ///     "[[abc]-x← error: unexpected character 'x'"
116    /// );
117    /// ```
118    ///
119    /// ```
120    /// use icu::properties::unicodeset_parse::*;
121    ///
122    /// let source = r"[\N{LATIN CAPITAL LETTER A}]";
123    /// let set = parse(source);
124    /// assert!(set.is_err());
125    /// let err = set.unwrap_err();
126    /// assert_eq!(
127    ///     err.fmt_with_source(source).to_string(),
128    ///     r"[\N← error: unimplemented"
129    /// );
130    /// ```
131    pub fn fmt_with_source(&self, source: &str) -> impl Display + use<> {
132        let ParseError { offset, kind } = *self;
133
134        if kind == ParseErrorKind::Eof {
135            return format!("{source}← error: unexpected end of input");
136        }
137        let mut s = String::new();
138        if let Some(offset) = offset
139            && offset < source.len()
140        {
141            // offset points to any byte of the last character we want to display.
142            // in the case of ASCII, this is easy - we just display bytes [..=offset].
143            // however, if the last character is more than one byte in UTF-8
144            // we cannot use ..=offset, because that would potentially include only partial
145            // bytes of last character in our string. hence we must find the start of the
146            // following character and use that as the (exclusive) end of our string.
147
148            // offset points into the last character we want to include, hence the start of the
149            // first character we want to exclude is at least offset + 1.
150            let mut exclusive_end = offset + 1;
151            // TODO: replace this loop with str::ceil_char_boundary once stable
152            for _ in 0..3 {
153                // is_char_boundary returns true at the latest once exclusive_end == source.len()
154                if source.is_char_boundary(exclusive_end) {
155                    break;
156                }
157                exclusive_end += 1;
158            }
159
160            // exclusive_end is at most source.len() due to str::is_char_boundary and at least 0 by type
161            s.push_str(&source[..exclusive_end]);
162            s.push_str("← ");
163        }
164        s.push_str("error: ");
165        match kind {
166            ParseErrorKind::UnexpectedChar(c) => {
167                s.push_str(&format!("unexpected character '{}'", c.escape_debug()));
168            }
169            ParseErrorKind::UnknownProperty => {
170                s.push_str("unknown property");
171            }
172            ParseErrorKind::UnknownVariable => {
173                s.push_str("unknown variable");
174            }
175            ParseErrorKind::UnexpectedVariable => {
176                s.push_str("unexpected variable");
177            }
178            ParseErrorKind::Eof => {
179                s.push_str("unexpected end of input");
180            }
181            ParseErrorKind::Internal => {
182                s.push_str("internal error");
183            }
184            ParseErrorKind::Unimplemented => {
185                s.push_str("unimplemented");
186            }
187            ParseErrorKind::InvalidEscape => {
188                s.push_str("invalid escape sequence");
189            }
190        }
191
192        s
193    }
194
195    /// Returns the [`ParseErrorKind`] of this error.
196    pub fn kind(&self) -> ParseErrorKind {
197        self.kind
198    }
199
200    /// Returns the offset of this error in the source string, if it was specified.
201    pub fn offset(&self) -> Option<usize> {
202        self.offset
203    }
204
205    fn or_with_offset(self, offset: usize) -> Self {
206        match self.offset {
207            Some(_) => self,
208            None => ParseError {
209                offset: Some(offset),
210                ..self
211            },
212        }
213    }
214}

251    /// Insert a `VariableValue` into the `VariableMap`.
252    ///
253    /// Returns `Err` with the old value, if it exists, and does not update the map.
254    pub fn insert(
255        &mut self,
256        key: String,
257        value: VariableValue<'a>,
258    ) -> Result<(), &VariableValue<'_>> {
259        // borrow-checker shenanigans, otherwise we could use if let
260        if self.0.contains_key(&key) {
261            // we just checked that this key exists
262            #[expect(clippy::indexing_slicing)]
263            return Err(&self.0[&key]);
264        }
265
266        if let VariableValue::String(s) = &value {
267            let mut chars = s.chars();
268            if let (Some(c), None) = (chars.next(), chars.next()) {
269                self.0.insert(key, VariableValue::Char(c));
270                return Ok(());
271            };
272        }
273
274        self.0.insert(key, value);
275        Ok(())
276    }

354// this ignores the ambiguity between \-escapes and \p{} perl properties. it assumes it is in a context where \p is just 'p'
355// returns whether the provided char signifies the start of a literal char (raw or escaped - so \ is a legal char start)
356// important: assumes c is not pattern_white_space
357fn legal_char_start(c: char) -> bool {
358    !(c == '&' || c == '-' || c == '$' || c == '^' || c == '[' || c == ']' || c == '{')
359}

361// same as `legal_char_start` but adapted to the charInString nonterminal. \ is allowed due to escapes.
362// important: assumes c is not pattern_white_space
363fn legal_char_in_string_start(c: char) -> bool {
364    c != '}'
365}

534    // the entry point, parses a full UnicodeSet. ignores remaining input
535    fn parse_unicode_set(&mut self) -> Result<()> {
536        match self.must_peek_char()? {
537            '\\' => self.parse_property_perl(),
538            '[' => {
539                self.iter.next();
540                if let Some(':') = self.peek_char() {
541                    self.parse_property_posix()
542                } else {
543                    self.parse_unicode_set_inner()
544                }
545            }
546            '$' => {
547                // must be variable ref to a UnicodeSet
548                let (offset, v) = self.parse_variable()?;
549                match v {
550                    Some(VariableValue::UnicodeSet(s)) => {
551                        self.single_set.add_set(s.code_points());
552                        self.string_set
553                            .extend(s.strings().iter().map(ToString::to_string));
554                        Ok(())
555                    }
556                    Some(_) => Err(PEK::UnexpectedVariable.with_offset(offset)),
557                    None => Err(PEK::UnexpectedChar('$').with_offset(offset)),
558                }
559            }
560            c => self.error_here(PEK::UnexpectedChar(c)),
561        }
562    }

564    // beginning [ is already consumed
565    fn parse_unicode_set_inner(&mut self) -> Result<()> {
566        // special cases for the first chars after [
567        if self.must_peek_char()? == '^' {
568            self.iter.next();
569            self.inverted = true;
570        }
571        // whitespace allowed between ^ and - in `[^ - ....]`
572        self.skip_whitespace();
573        if self.must_peek_char()? == '-' {
574            self.iter.next();
575            self.single_set.add_char('-');
576        }
577
578        // repeatedly parse the following:
579        // char
580        // char-char
581        // {string}
582        // unicodeset
583        // & and - operators, but only between unicodesets
584        // $variables in place of strings, chars, or unicodesets
585
586        #[derive(Debug, Clone, Copy)]
587        enum State {
588            // a state equivalent to the beginning
589            Begin,
590            // a state after a char. implies `prev_char` is Some(_), because we need to buffer it
591            // in case it is part of a range, e.g., a-z
592            Char,
593            // in the middle of parsing a range. implies `prev_char` is Some(_), and the next
594            // element must be a char as well
595            CharMinus,
596            // state directly after parsing a recursive unicode set. operators are only allowed
597            // in this state
598            AfterUnicodeSet,
599            // state directly after parsing an operator. forces the next element to be a recursive
600            // unicode set
601            AfterOp,
602            // state after parsing a $ (that was not a variable reference)
603            // the only valid next option is a closing bracket
604            AfterDollar,
605            // state after parsing a - in an otherwise invalid position
606            // the only valid next option is a closing bracket
607            AfterMinus,
608        }
609        use State::*;
610
611        const DEFAULT_OP: Operation = Operation::Union;
612
613        let mut state = Begin;
614        let mut prev_char = None;
615        let mut operation = Operation::Union;
616
617        loop {
618            self.skip_whitespace();
619
620            // for error messages
621            let (immediate_offset, immediate_char) = self.must_peek()?;
622
623            let (tok_offset, from_var, tok) = self.parse_main_token()?;
624            // warning: self.iter should not be advanced any more after this point on any path to
625            // MT::Literal(Literal::CharKind(SingleOrMultiChar::Multi)), because that variant
626            // expects a certain self.iter state
627
628            use MainToken as MT;
629            use SingleOrMultiChar as SMC;
630            match (state, tok) {
631                // the end of this unicode set
632                (
633                    Begin | Char | CharMinus | AfterUnicodeSet | AfterDollar | AfterMinus,
634                    MT::ClosingBracket,
635                ) => {
636                    if let Some(prev) = prev_char.take() {
637                        self.single_set.add_char(prev);
638                    }
639                    if matches!(state, CharMinus) {
640                        self.single_set.add_char('-');
641                    }
642
643                    return Ok(());
644                }
645                // special case ends for -
646                // [[a-z]-]
647                (AfterOp, MT::ClosingBracket) if matches!(operation, Operation::Difference) => {
648                    self.single_set.add_char('-');
649                    return Ok(());
650                }
651                (Begin, MT::Minus) => {
652                    self.single_set.add_char('-');
653                    state = AfterMinus;
654                }
655                // inner unicode set
656                (Begin | Char | AfterUnicodeSet | AfterOp, MT::UnicodeSet(set)) => {
657                    if let Some(prev) = prev_char.take() {
658                        self.single_set.add_char(prev);
659                    }
660
661                    self.process_chars(operation, set.code_points().clone());
662                    self.process_strings(
663                        operation,
664                        set.strings().iter().map(ToString::to_string).collect(),
665                    );
666
667                    operation = DEFAULT_OP;
668                    state = AfterUnicodeSet;
669                }
670                // a literal char (either individually or as the start of a range if char)
671                (
672                    Begin | Char | AfterUnicodeSet,
673                    MT::Literal(Literal::CharKind(SMC::Single(c))),
674                ) => {
675                    if let Some(prev) = prev_char.take() {
676                        self.single_set.add_char(prev);
677                    }
678                    prev_char = Some(c);
679                    state = Char;
680                }
681                // a bunch of literal chars as part of a multi-escape sequence
682                (
683                    Begin | Char | AfterUnicodeSet,
684                    MT::Literal(Literal::CharKind(SMC::Multi(first_c))),
685                ) => {
686                    if let Some(prev) = prev_char.take() {
687                        self.single_set.add_char(prev);
688                    }
689                    self.single_set.add_char(first_c);
690                    self.parse_multi_escape_into_set()?;
691
692                    // Note we cannot go to the Char state, because a multi-escape sequence of
693                    // length > 1 cannot initiate a range
694                    state = Begin;
695                }
696                // a literal string (length != 1, by CharOrString invariant)
697                (Begin | Char | AfterUnicodeSet, MT::Literal(Literal::String(s))) => {
698                    if let Some(prev) = prev_char.take() {
699                        self.single_set.add_char(prev);
700                    }
701
702                    self.string_set.insert(s);
703                    state = Begin;
704                }
705                // parse a literal char as the end of a range
706                (CharMinus, MT::Literal(Literal::CharKind(SMC::Single(c)))) => {
707                    let start = prev_char.ok_or(ParseError {
708                        offset: Some(tok_offset),
709                        kind: PEK::Internal,
710                    })?;
711                    let end = c;
712                    if start > end {
713                        // TODO(#3558): Better error message (e.g., "start greater than end in range")?
714                        return Err(PEK::UnexpectedChar(end).with_offset(tok_offset));
715                    }
716
717                    self.single_set.add_range(start..=end);
718                    prev_char = None;
719                    state = Begin;
720                }
721                // start parsing a char range
722                (Char, MT::Minus) => {
723                    state = CharMinus;
724                }
725                // start parsing a unicode set difference
726                (AfterUnicodeSet, MT::Minus) => {
727                    operation = Operation::Difference;
728                    state = AfterOp;
729                }
730                // start parsing a unicode set difference
731                (AfterUnicodeSet, MT::Ampersand) => {
732                    operation = Operation::Intersection;
733                    state = AfterOp;
734                }
735                (Begin | Char | AfterUnicodeSet, MT::DollarSign) => {
736                    if let Some(prev) = prev_char.take() {
737                        self.single_set.add_char(prev);
738                    }
739                    self.single_set.add_char('\u{FFFF}');
740                    state = AfterDollar;
741                }
742                _ => {
743                    // TODO(#3558): We have precise knowledge about the following MainToken here,
744                    //  should we make use of that?
745
746                    if from_var {
747                        // otherwise we get error messages such as
748                        // [$a-$← error: unexpected character '$'
749                        // for input [$a-$b], $a = 'a', $b = "string" ;
750                        return Err(PEK::UnexpectedVariable.with_offset(tok_offset));
751                    }
752                    return Err(PEK::UnexpectedChar(immediate_char).with_offset(immediate_offset));
753                }
754            }
755        }
756    }

758    fn parse_main_token(&mut self) -> Result<(usize, bool, MainToken<'a>)> {
759        let (initial_offset, first) = self.must_peek()?;
760        if first == ']' {
761            self.iter.next();
762            return Ok((initial_offset, false, MainToken::ClosingBracket));
763        }
764        let (_, second) = self.must_peek_double()?;
765        match (first, second) {
766            // variable or anchor
767            ('$', _) => {
768                let (offset, var_or_anchor) = self.parse_variable()?;
769                match var_or_anchor {
770                    None => Ok((offset, false, MainToken::DollarSign)),
771                    Some(v) => Ok((offset, true, MainToken::from_variable_value(v.clone()))),
772                }
773            }
774            // string
775            ('{', _) => self
776                .parse_string()
777                .map(|(offset, l)| (offset, false, MainToken::Literal(l))),
778            // inner set
779            ('\\', 'p' | 'P') | ('[', _) => {
780                let mut inner_builder = UnicodeSetBuilder::new_internal(
781                    self.iter,
782                    self.source,
783                    self.variable_map,
784                    self.xid_start,
785                    self.xid_continue,
786                    self.pat_ws,
787                    self.property_provider,
788                );
789                inner_builder.parse_unicode_set()?;
790                let (single, string_set) = inner_builder.finalize();
791                // note: offset - 1, because we already consumed full set
792                let offset = self.must_peek_index()? - 1;
793                let mut strings = string_set.into_iter().collect::<Vec<_>>();
794                strings.sort();
795                let cpilasl = CodePointInversionListAndStringList::try_from(
796                    single.build(),
797                    VarZeroVec::from(&strings),
798                )
799                .map_err(|_| PEK::Internal.with_offset(offset))?;
800                Ok((offset, false, MainToken::UnicodeSet(cpilasl)))
801            }
802            // note: c cannot be a whitespace, because we called skip_whitespace just before
803            // (in the main parse loop), so it's safe to call this guard function
804            (c, _) if legal_char_start(c) => self
805                .parse_char()
806                .map(|(offset, c)| (offset, false, MainToken::Literal(Literal::CharKind(c)))),
807            ('-', _) => {
808                self.iter.next();
809                Ok((initial_offset, false, MainToken::Minus))
810            }
811            ('&', _) => {
812                self.iter.next();
813                Ok((initial_offset, false, MainToken::Ampersand))
814            }
815            (c, _) => Err(PEK::UnexpectedChar(c).with_offset(initial_offset)),
816        }
817    }

819    // parses a variable or an anchor. expects '$' as next token.
820    // if this is a single $ (eg `[... $ ]` or the invalid `$ a`), then this function returns Ok(None),
821    // otherwise Ok(Some(variable_value)).
822    fn parse_variable(&mut self) -> Result<(usize, Option<&'a VariableValue<'a>>)> {
823        self.consume('$')?;
824
825        let mut res = String::new();
826        let (mut var_offset, first_c) = self.must_peek()?;
827
828        if !self.xid_start.contains(first_c) {
829            // -1 because we already consumed the '$'
830            return Ok((var_offset - 1, None));
831        }
832
833        res.push(first_c);
834        self.iter.next();
835        // important: if we are parsing a root unicodeset as a variable, we might reach EOF as
836        // a valid end of the variable name, so we cannot use must_peek here.
837        while let Some(&(offset, c)) = self.iter.peek() {
838            if !self.xid_continue.contains(c) {
839                break;
840            }
841            // only update the offset if we're adding a new char to our variable
842            var_offset = offset;
843            self.iter.next();
844            res.push(c);
845        }
846
847        if let Some(v) = self.variable_map.0.get(&res) {
848            return Ok((var_offset, Some(v)));
849        }
850
851        Err(PEK::UnknownVariable.with_offset(var_offset))
852    }

854    // parses and consumes: '{' (s charInString)* s '}'
855    fn parse_string(&mut self) -> Result<(usize, Literal)> {
856        self.consume('{')?;
857
858        let mut buffer = String::new();
859        let mut last_offset;
860
861        loop {
862            self.skip_whitespace();
863            last_offset = self.must_peek_index()?;
864            match self.must_peek_char()? {
865                '}' => {
866                    self.iter.next();
867                    break;
868                }
869                // note: c cannot be a whitespace, because we called skip_whitespace just before,
870                // so it's safe to call this guard function
871                c if legal_char_in_string_start(c) => {
872                    // don't need the offset, because '}' will always be the last char
873                    let (_, c) = self.parse_char()?;
874                    match c {
875                        SingleOrMultiChar::Single(c) => buffer.push(c),
876                        SingleOrMultiChar::Multi(first) => {
877                            buffer.push(first);
878                            self.parse_multi_escape_into_string(&mut buffer)?;
879                        }
880                    }
881                }
882                c => return self.error_here(PEK::UnexpectedChar(c)),
883            }
884        }
885
886        let mut chars = buffer.chars();
887        let literal = match (chars.next(), chars.next()) {
888            (Some(c), None) => Literal::CharKind(SingleOrMultiChar::Single(c)),
889            _ => Literal::String(buffer),
890        };
891        Ok((last_offset, literal))
892    }

894    // finishes a partial multi escape parse. in case of a parse error, self.single_set
895    // may be left in an inconsistent state
896    fn parse_multi_escape_into_set(&mut self) -> Result<()> {
897        // note: would be good to somehow merge the two multi_escape methods. splitting up the UnicodeSetBuilder into a more
898        // conventional parser + lexer combo might allow this.
899        // issue is that we cannot pass this method an argument that somehow mutates `self` in the current architecture.
900        // self.lexer.parse_multi_into_charappendable(&mut self.single_set) should work because the lifetimes are separate
901
902        // whitespace before first char of this loop (ie, second char in this multi_escape) must be
903        // enforced when creating the SingleOrMultiChar::Multi.
904        let mut first = true;
905        loop {
906            let skipped = self.skip_whitespace();
907            match self.must_peek_char()? {
908                '}' => {
909                    self.iter.next();
910                    return Ok(());
911                }
912                initial_c => {
913                    if skipped == 0 && !first {
914                        // bracketed hex code points must be separated by whitespace
915                        return self.error_here(PEK::UnexpectedChar(initial_c));
916                    }
917                    first = false;
918
919                    let (_, c) = self.parse_hex_digits_into_char(1, 6)?;
920                    self.single_set.add_char(c);
921                }
922            }
923        }
924    }

953    // starts with \ and consumes the whole escape sequence if a single
954    // char is escaped, otherwise pauses the parse after the first char
955    fn parse_escaped_char(&mut self) -> Result<(usize, SingleOrMultiChar)> {
956        self.consume('\\')?;
957
958        let (offset, next_char) = self.must_next()?;
959
960        match next_char {
961            'u' | 'x' if self.peek_char() == Some('{') => {
962                // bracketedHex
963                self.iter.next();
964
965                self.skip_whitespace();
966                let (_, first_c) = self.parse_hex_digits_into_char(1, 6)?;
967                let skipped = self.skip_whitespace();
968
969                match self.must_peek()? {
970                    (offset, '}') => {
971                        self.iter.next();
972                        Ok((offset, SingleOrMultiChar::Single(first_c)))
973                    }
974                    // note: enforcing whitespace after the first char here, because the parse_multi_escape functions
975                    // won't have access to this information anymore
976                    (offset, c) if c.is_ascii_hexdigit() && skipped > 0 => {
977                        Ok((offset, SingleOrMultiChar::Multi(first_c)))
978                    }
979                    (_, c) => self.error_here(PEK::UnexpectedChar(c)),
980                }
981            }
982            'u' => {
983                // 'u' hex{4}
984                self.parse_hex_digits_into_char(4, 4)
985                    .map(|(offset, c)| (offset, SingleOrMultiChar::Single(c)))
986            }
987            'x' => {
988                // 'x' hex{2}
989                self.parse_hex_digits_into_char(2, 2)
990                    .map(|(offset, c)| (offset, SingleOrMultiChar::Single(c)))
991            }
992            'U' => {
993                // 'U00' ('0' hex{5} | '10' hex{4})
994                self.consume('0')?;
995                self.consume('0')?;
996                self.parse_hex_digits_into_char(6, 6)
997                    .map(|(offset, c)| (offset, SingleOrMultiChar::Single(c)))
998            }
999            'N' => {
1000                // parse code point with name in {}
1001                // tracking issue: https://github.com/unicode-org/icu4x/issues/1397
1002                Err(PEK::Unimplemented.with_offset(offset))
1003            }
1004            'a' => Ok((offset, SingleOrMultiChar::Single('\u{0007}'))),
1005            'b' => Ok((offset, SingleOrMultiChar::Single('\u{0008}'))),
1006            't' => Ok((offset, SingleOrMultiChar::Single('\u{0009}'))),
1007            'n' => Ok((offset, SingleOrMultiChar::Single('\u{000A}'))),
1008            'v' => Ok((offset, SingleOrMultiChar::Single('\u{000B}'))),
1009            'f' => Ok((offset, SingleOrMultiChar::Single('\u{000C}'))),
1010            'r' => Ok((offset, SingleOrMultiChar::Single('\u{000D}'))),
1011            _ => Ok((offset, SingleOrMultiChar::Single(next_char))),
1012        }
1013    }

1045    fn parse_property_inner(&mut self, end: char) -> Result<()> {
1046        // UnicodeSet spec ignores whitespace, '-', and '_',
1047        // but ECMA-262 requires '_', so we'll allow that.
1048        // TODO(#3559): support loose matching on property names (e.g., "AS  -_-  CII_Hex_ D-igit")
1049        // TODO(#3559): support more properties than ECMA-262
1050
1051        let property_offset;
1052
1053        let mut key_buffer = String::new();
1054        let mut value_buffer = String::new();
1055
1056        enum State {
1057            // initial state, nothing parsed yet
1058            Begin,
1059            // non-empty property name
1060            PropertyName,
1061            // property name parsed, '=' or '≠' parsed, no value parsed yet
1062            PropertyValueBegin,
1063            // non-empty property name, non-empty property value
1064            PropertyValue,
1065        }
1066        use State::*;
1067
1068        let mut state = Begin;
1069        // whether '=' (true) or '≠' (false) was parsed
1070        let mut equality = true;
1071
1072        loop {
1073            self.skip_whitespace();
1074            match (state, self.must_peek_char()?) {
1075                // parse the end of the property expression
1076                (PropertyName | PropertyValue, c) if c == end => {
1077                    // byte index of (full) property name/value is one back
1078                    property_offset = self.must_peek_index()? - 1;
1079                    self.iter.next();
1080                    break;
1081                }
1082                // parse the property name
1083                // NOTE: this might be too strict, because in the case of e.g. [:value:], we might want to
1084                // allow [:lower-case-letter:] ([:gc=lower-case-letter:] works)
1085                (Begin | PropertyName, c) if c.is_ascii_alphanumeric() || c == '_' => {
1086                    key_buffer.push(c);
1087                    self.iter.next();
1088                    state = PropertyName;
1089                }
1090                // parse the name-value separator
1091                (PropertyName, c @ ('=' | '≠')) => {
1092                    equality = c == '=';
1093                    self.iter.next();
1094                    state = PropertyValueBegin;
1095                }
1096                // parse the property value
1097                (PropertyValue | PropertyValueBegin, c) if c != end => {
1098                    value_buffer.push(c);
1099                    self.iter.next();
1100                    state = PropertyValue;
1101                }
1102                (_, c) => return self.error_here(PEK::UnexpectedChar(c)),
1103            }
1104        }
1105
1106        if !equality {
1107            self.inverted = !self.inverted;
1108        }
1109
1110        let inverted = self
1111            .load_property_codepoints(&key_buffer, &value_buffer)
1112            // any error that does not already have an offset should use the appropriate property offset
1113            .map_err(|e| e.or_with_offset(property_offset))?;
1114        if inverted {
1115            self.inverted = !self.inverted;
1116        }
1117
1118        Ok(())
1119    }

1121    // returns whether the set needs to be inverted or not
1122    fn load_property_codepoints(&mut self, key: &str, value: &str) -> Result<bool> {
1123        // we support:
1124        // [:gc = value:]
1125        // [:sc = value:]
1126        // [:scx = value:]
1127        // [:Grapheme_Cluster_Break = value:]
1128        // [:Sentence_Break = value:]
1129        // [:Word_Break = value:]
1130        // [:value:] - looks up value in gc, sc
1131        // [:prop:] - binary property, returns codepoints that have the property
1132        // [:prop = truthy/falsy:] - same as above
1133
1134        let mut inverted = false;
1135
1136        // contains a value for the General_Category property that needs to be tried
1137        let mut try_gc = Err(PEK::UnknownProperty.into());
1138        // contains a value for the Script property that needs to be tried
1139        let mut try_sc = Err(PEK::UnknownProperty.into());
1140        // contains a value for the Script_Extensions property that needs to be tried
1141        let mut try_scx = Err(PEK::UnknownProperty.into());
1142        // contains a value for the Grapheme_Cluster_Break property that needs to be tried
1143        let mut try_gcb = Err(PEK::UnknownProperty.into());
1144        // contains a value for the Indic_Conjuct_Break property that needs to be tried
1145        let mut try_incb = Err(PEK::UnknownProperty.into());
1146        // contains a value for the Line_Break property that needs to be tried
1147        let mut try_lb = Err(PEK::UnknownProperty.into());
1148        // contains a value for the Sentence_Break property that needs to be tried
1149        let mut try_sb = Err(PEK::UnknownProperty.into());
1150        // contains a value for the Word_Break property that needs to be tried
1151        let mut try_wb = Err(PEK::UnknownProperty.into());
1152        // contains a value for the East_Asian_Width property that needs to be tried
1153        let mut try_eaw = Err(PEK::UnknownProperty.into());
1154        // contains a supposed binary property name that needs to be tried
1155        let mut try_binary = Err(PEK::UnknownProperty.into());
1156        // contains a supposed canonical combining class property name that needs to be tried
1157        let mut try_ccc: Result<&str, ParseError> = Err(PEK::UnknownProperty.into());
1158        // contains a supposed block property name that needs to be tried
1159        let mut try_block: Result<&str, ParseError> = Err(PEK::UnknownProperty.into());
1160
1161        if !value.is_empty() {
1162            // key is gc, sc, scx, grapheme cluster break, sentence break, word break
1163            // value is a property value
1164            // OR
1165            // key is a binary property and value is a truthy/falsy value
1166
1167            fn loose(s: &[u8]) -> impl Iterator<Item = u8> + '_ {
1168                s.iter()
1169                    .map(|&c| c.to_ascii_lowercase())
1170                    .filter(|&c| c != b'_')
1171            }
1172
1173            if loose(key.as_bytes()).eq(loose(GeneralCategory::NAME))
1174                || loose(key.as_bytes()).eq(loose(GeneralCategory::SHORT_NAME))
1175            {
1176                try_gc = Ok(value)
1177            } else if loose(key.as_bytes()).eq(loose(GraphemeClusterBreak::NAME))
1178                || loose(key.as_bytes()).eq(loose(GraphemeClusterBreak::SHORT_NAME))
1179            {
1180                try_gcb = Ok(value)
1181            } else if loose(key.as_bytes()).eq(loose(IndicConjunctBreak::NAME))
1182                || loose(key.as_bytes()).eq(loose(IndicConjunctBreak::SHORT_NAME))
1183            {
1184                try_incb = Ok(value)
1185            } else if loose(key.as_bytes()).eq(loose(LineBreak::NAME))
1186                || loose(key.as_bytes()).eq(loose(LineBreak::SHORT_NAME))
1187            {
1188                try_lb = Ok(value)
1189            } else if loose(key.as_bytes()).eq(loose(Script::NAME))
1190                || loose(key.as_bytes()).eq(loose(Script::SHORT_NAME))
1191            {
1192                try_sc = Ok(value)
1193            } else if loose(key.as_bytes()).eq(loose(SentenceBreak::NAME))
1194                || loose(key.as_bytes()).eq(loose(SentenceBreak::SHORT_NAME))
1195            {
1196                try_sb = Ok(value)
1197            } else if loose(key.as_bytes()).eq(loose(WordBreak::NAME))
1198                || loose(key.as_bytes()).eq(loose(WordBreak::SHORT_NAME))
1199            {
1200                try_wb = Ok(value)
1201            } else if loose(key.as_bytes()).eq(loose(EastAsianWidth::NAME))
1202                || loose(key.as_bytes()).eq(loose(EastAsianWidth::SHORT_NAME))
1203            {
1204                try_eaw = Ok(value)
1205            } else if loose(key.as_bytes()).eq(loose(CanonicalCombiningClass::NAME))
1206                || loose(key.as_bytes()).eq(loose(CanonicalCombiningClass::SHORT_NAME))
1207            {
1208                try_ccc = Ok(value)
1209            } else if loose(key.as_bytes()).eq(loose(b"Script_Extensions"))
1210                || loose(key.as_bytes()).eq(loose(b"scx"))
1211            {
1212                try_scx = Ok(value)
1213            } else if loose(key.as_bytes()).eq(loose(b"Block"))
1214                || loose(key.as_bytes()).eq(loose(b"blk"))
1215            {
1216                try_block = Ok(value)
1217            } else {
1218                let normalized_value = value.to_ascii_lowercase();
1219                let truthy = matches!(normalized_value.as_str(), "true" | "t" | "yes" | "y");
1220                let falsy = matches!(normalized_value.as_str(), "false" | "f" | "no" | "n");
1221                // value must either match truthy or falsy
1222                if truthy == falsy {
1223                    return Err(PEK::UnknownProperty.into());
1224                }
1225                // correctness: if we reach this point, only `try_binary` can be Ok, hence
1226                // it does not matter that further down we unconditionally return `inverted`,
1227                // because only `try_binary` can enter that code path.
1228                inverted = falsy;
1229                try_binary = Ok(key);
1230            }
1231        } else {
1232            // key is binary property
1233            // OR a value of gc, sc (only gc or sc are supported as implicit keys by UTS35!)
1234            try_gc = Ok(key);
1235            try_sc = Ok(key);
1236            try_binary = Ok(key);
1237        }
1238
1239        try_gc
1240            .and_then(|value| self.try_load_general_category_set(value))
1241            .or_else(|_| try_sc.and_then(|value| self.try_load_script_set(value)))
1242            .or_else(|_| try_scx.and_then(|value| self.try_load_script_extensions_set(value)))
1243            .or_else(|_| try_binary.and_then(|value| self.try_load_ecma262_binary_set(value)))
1244            .or_else(|_| try_gcb.and_then(|value| self.try_load_grapheme_cluster_break_set(value)))
1245            .or_else(|_| try_incb.and_then(|value| self.try_load_indic_conjunct_break_set(value)))
1246            .or_else(|_| try_lb.and_then(|value| self.try_load_line_break_set(value)))
1247            .or_else(|_| try_sb.and_then(|value| self.try_load_sentence_break_set(value)))
1248            .or_else(|_| try_wb.and_then(|value| self.try_load_word_break_set(value)))
1249            .or_else(|_| try_eaw.and_then(|value| self.try_load_east_asian_width_set(value)))
1250            .or_else(|_| try_ccc.and_then(|value| self.try_load_ccc_set(value)))
1251            .or_else(|_| try_block.and_then(|value| self.try_load_block_set(value)))?;
1252        Ok(inverted)
1253    }

1255    fn finalize(mut self) -> (CodePointInversionListBuilder, BTreeSet<String>) {
1256        if self.inverted {
1257            // code point inversion; removes all strings
1258            #[cfg(feature = "log")]
1259            {
1260                let single_set = self.single_set.clone().build();
1261                if !self
1262                    .string_set
1263                    .iter()
1264                    // if the string starts with a cp in the cp-set, then it'll be rejected through that
1265                    .all(|s| s.chars().next().is_some_and(|c| single_set.contains(c)))
1266                {
1267                    log::info!(
1268                        "Inverting a unicode set with strings. This removes all strings entirely."
1269                    );
1270                }
1271            }
1272            self.string_set.clear();
1273            self.single_set.complement();
1274        }
1275
1276        (self.single_set, self.string_set)
1277    }

1293    // by manually parsing the digits instead of using u32::from_str_radix.
1294    fn parse_hex_digits_into_char(&mut self, min: usize, max: usize) -> Result<(usize, char)> {
1295        let first_offset = self.must_peek_index()?;
1296        let end_offset = self.validate_hex_digits(min, max)?;
1297
1298        // validate_hex_digits ensures that chars (including the last one) are ascii hex digits,
1299        // which are all exactly one UTF-8 byte long, so slicing on these offsets always respects char boundaries
1300        let hex_source = &self.source[first_offset..=end_offset];
1301        let num = u32::from_str_radix(hex_source, 16).map_err(|_| PEK::Internal)?;
1302        char::try_from(num)
1303            .map(|c| (end_offset, c))
1304            .map_err(|_| PEK::InvalidEscape.with_offset(end_offset))
1305    }

1307    // validates [0-9a-fA-F]{min,max}, returns the offset of the last digit, consuming everything in the process
1308    fn validate_hex_digits(&mut self, min: usize, max: usize) -> Result<usize> {
1309        let mut last_offset = 0;
1310        for count in 0..max {
1311            let (offset, c) = self.must_peek()?;
1312            if !c.is_ascii_hexdigit() {
1313                if count < min {
1314                    return Err(PEK::UnexpectedChar(c).with_offset(offset));
1315                } else {
1316                    break;
1317                }
1318            }
1319            self.iter.next();
1320            last_offset = offset;
1321        }
1322        Ok(last_offset)
1323    }

1615///
1616/// Parses a `UnicodeSet` pattern and returns a `UnicodeSet` in the form of a [`CodePointInversionListAndStringList`](CodePointInversionListAndStringList),
1617/// as well as the number of bytes consumed from the source string.
1618///
1619/// Supports `UnicodeSets` as described in [UTS #35 - Unicode Sets](https://unicode.org/reports/tr35/#Unicode_Sets).
1620///
1621/// The error type of the returned Result can be pretty-printed with [`ParseError::fmt_with_source`].
1622///
1623/// # Variables
1624///
1625/// If you need support for variables inside `UnicodeSets` (e.g., `[$start-$end]`), use [`parse_with_variables`].
1626///
1627/// # Limitations
1628///
1629/// * Currently, we only support the [ECMA-262 properties](https://tc39.es/ecma262/#table-nonbinary-unicode-properties).
1630///   The property names must match the exact spelling listed in ECMA-262. Note that we do support UTS35 syntax for elided `General_Category`
1631///   and `Script` property names, i.e., `[:Latn:]` and `[:Ll:]` are both valid, with the former implying the `Script` property, and the latter the
1632///   `General_Category` property.
1633/// * We do not support `\N{Unicode code point name}` character escaping. Use any other escape method described in UTS35.
1634///
1635/// ✨ *Enabled with the `compiled_data` Cargo feature.*
1636///
1637/// [📚 Help choosing a constructor](icu_provider::constructors)
1638///
1639/// # Examples
1640///
1641/// Parse ranges
1642/// ```
1643/// use icu::properties::unicodeset_parse::parse;
1644///
1645/// let source = "[a-zA-Z0-9]";
1646/// let (set, consumed) = parse(source).unwrap();
1647/// let code_points = set.code_points();
1648///
1649/// assert!(code_points.contains_range('a'..='z'));
1650/// assert!(code_points.contains_range('A'..='Z'));
1651/// assert!(code_points.contains_range('0'..='9'));
1652/// assert_eq!(consumed, source.len());
1653/// ```
1654///
1655/// Parse properties, set operations, inner sets
1656/// ```
1657/// use icu::properties::unicodeset_parse::parse;
1658///
1659/// let (set, _) =
1660///     parse("[[:^ll:]-[^][:gc = Lowercase Letter:]&[^[[^]-[a-z]]]]").unwrap();
1661/// assert!(set.code_points().contains_range('a'..='z'));
1662/// assert_eq!(('a'..='z').count(), set.size());
1663/// ```
1664///
1665/// Inversions remove strings
1666/// ```
1667/// use icu::properties::unicodeset_parse::parse;
1668///
1669/// let (set, _) =
1670///     parse(r"[[a-z{hello\ world}]&[^a-y{hello\ world}]]").unwrap();
1671/// assert!(set.contains('z'));
1672/// assert_eq!(set.size(), 1);
1673/// assert!(!set.has_strings());
1674/// ```
1675///
1676/// Set operators (including the implicit union) have the same precedence and are left-associative
1677/// ```
1678/// use icu::properties::unicodeset_parse::parse;
1679///
1680/// let (set, _) = parse("[[ace][bdf] - [abc][def]]").unwrap();
1681/// assert!(set.code_points().contains_range('d'..='f'));
1682/// assert_eq!(set.size(), ('d'..='f').count());
1683/// ```
1684///
1685/// Supports partial parses
1686/// ```
1687/// use icu::properties::unicodeset_parse::parse;
1688///
1689/// let (set, consumed) = parse("[a-c][x-z]").unwrap();
1690/// let code_points = set.code_points();
1691/// assert!(code_points.contains_range('a'..='c'));
1692/// assert!(!code_points.contains_range('x'..='z'));
1693/// assert_eq!(set.size(), ('a'..='c').count());
1694/// // only the first UnicodeSet is parsed
1695/// assert_eq!(consumed, "[a-c]".len());
1696/// ```
