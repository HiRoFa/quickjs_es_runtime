11pub(super) fn baseline(
12    kind: &BaselineKind,
13    downstream: bool,
14    kaios: bool,
15    opts: &Opts,
16) -> QueryResult {
17    // The Baseline timeline is keyed by baseline-low dates, which browserslist
18    // queries with a 30-month "widely available" offset:
19    //   widely available          => today - 30 months
20    //   newly available           => (today + 30 months) - 30 months
21    //   widely available on DATE  => DATE - 30 months
22    //   YEAR                      => end of that year
23    let cutoff = match kind {
24        BaselineKind::WidelyAvailable => add_months(today(), -30),
25        BaselineKind::NewlyAvailable => add_months(add_months(today(), 30), -30),
26        BaselineKind::WidelyAvailableOnDate(date) => add_months(parse_date(date), -30),
27        BaselineKind::Year(year) => (i32::from(*year), 12, 31),
28    };
29    let cutoff = date_key(cutoff);
30
31    // KaiOS is a downstream browser, but it can be requested independently of
32    // the other downstream browsers.
33    let is_included = |browser: &str| {
34        baseline::is_core_browser(browser)
35            || (downstream && browser != "kaios")
36            || (kaios && browser == "kaios")
37    };
38
39    let mut distribs = Vec::new();
40    match baseline::min_versions_on(cutoff) {
41        Some(min_versions) => {
42            for (browser, version) in min_versions.filter(|(browser, _)| is_included(browser)) {
43                distribs.append(&mut browser_unbounded_range(
44                    browser,
45                    Comparator::GreaterOrEqual,
46                    version,
47                    opts,
48                )?);
49            }
50        }
51        // Before the first Baseline feature, every version of every browser
52        // is considered compatible.
53        None => {
54            for browser in baseline::browsers().filter(|browser| is_included(browser)) {
55                distribs.append(&mut browser_unbounded_range(
56                    browser,
57                    Comparator::GreaterOrEqual,
58                    "0",
59                    opts,
60                )?);
61            }
62        }
63    }
64    Ok(distribs)
65}

67/// Encodes a date as decimal `yyyymmdd`, the key of the Baseline timeline.
68fn date_key((year, month, day): Date) -> u32 {
69    year as u32 * 10000 + month * 100 + day
70}
71
72fn today() -> Date {
73    let now = Local::now().date_naive();
74    (now.year(), now.month(), now.day())
75}
76
77fn parse_date(date: &str) -> Date {
78    // The "YYYY-MM-DD" shape is guaranteed by the parser.
79    let mut parts = date.split('-').map(|part| part.parse().unwrap());
80    let year = parts.next().unwrap();
81    let month = parts.next().unwrap() as u32;
82    let day = parts.next().unwrap() as u32;
83    (year, month, day)
84}

86/// Adds `offset` months to a date with JavaScript `Date.prototype.setMonth`
87/// semantics: a day-of-month past the end of the target month rolls over
88/// into the following month.
89fn add_months((year, month, day): Date, offset: i32) -> Date {
90    let months = year * 12 + month as i32 - 1 + offset;
91    let mut year = months.div_euclid(12);
92    let mut month = months.rem_euclid(12) as u32 + 1;
93    let mut day = day;
94    while day > days_in_month(year, month) {
95        day -= days_in_month(year, month);
96        month += 1;
97        if month > 12 {
98            month = 1;
99            year += 1;
100        }
101    }
102    (year, month, day)
103}
104
105fn days_in_month(year: i32, month: u32) -> u32 {
106    match month {
107        4 | 6 | 9 | 11 => 30,
108        2 if (year % 4 == 0 && year % 100 != 0) || year % 400 == 0 => 29,
109        2 => 28,
110        _ => 31,
111    }
112}
