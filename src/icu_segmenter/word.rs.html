52/// Implements the [`Iterator`] trait over the word boundaries of the given string.
53///
54/// Lifetimes:
55///
56/// - `'l` = lifetime of the segmenter object from which this iterator was created
57/// - `'s` = lifetime of the string being segmented
58///
59/// The [`Iterator::Item`] is an [`usize`] representing index of a code unit
60/// _after_ the boundary (for a boundary at the end of text, this index is the length
61/// of the [`str`] or array of code units).
62///
63/// For examples of use, see [`WordSegmenter`].
64#[derive(Debug)]
65pub struct WordBreakIterator<'data, 's, Y: RuleBreakType>(WordBreakIteratorInner<'data, 's, Y>);
66
67#[derive(Debug)]
68enum WordBreakIteratorInner<'data, 's, Y: RuleBreakType> {
69    V1(crate::rule_segmenter_v1::RuleBreakIterator<'data, 's, Y>),
70    #[cfg(feature = "unstable")]
71    V2(crate::rule_segmenter_v2::RuleBreakIterator<'data, 's, Y, ComplexWord<Y>>),
72}
73
74impl<Y: RuleBreakType> Iterator for WordBreakIterator<'_, '_, Y> {
75    type Item = usize;
76    fn next(&mut self) -> Option<Self::Item> {
77        match self.0 {
78            WordBreakIteratorInner::V1(ref mut iter) => iter.next(),
79            #[cfg(feature = "unstable")]
80            WordBreakIteratorInner::V2(ref mut iter) => iter.next(),
81        }
82    }
83}

85/// The word type tag that is returned by [`WordBreakIterator::word_type()`].
86///
87/// [`WordBreakIterator::word_type()`]: WordBreakIterator::word_type
88#[non_exhaustive]
89#[derive(Copy, Clone, PartialEq, Debug)]
90#[repr(u8)]
91pub enum WordType {
92    /// No category tag.
93    None = 0,
94    /// Number category tag.
95    Number = 1,
96    /// Letter category tag, including CJK.
97    Letter = 2,
98}
99
100impl WordType {
101    /// Whether the segment is word-like; word-like segments include numbers, as
102    /// well as segments made up of letters (including CJKV ideographs).
103    pub fn is_word_like(&self) -> bool {
104        self != &WordType::None
105    }
106}

108impl<'data, 's, Y: RuleBreakType> WordBreakIterator<'data, 's, Y> {
109    /// Returns the word type of the segment preceding the current boundary.
110    #[inline]
111    pub fn word_type(&self) -> WordType {
112        let last_accepting_status = match self.0 {
113            WordBreakIteratorInner::V1(ref iter) => iter.rule_status(),
114            #[cfg(feature = "unstable")]
115            WordBreakIteratorInner::V2(ref iter) => iter.last_accepting_status(),
116        };
117        match last_accepting_status {
118            0 => WordType::None,
119            1 => WordType::Number,
120            _ => WordType::Letter,
121        }
122    }
123
124    /// Returns an iterator over pairs of boundary position and word type.
125    pub fn iter_with_word_type(self) -> WordBreakIteratorWithWordType<'data, 's, Y> {
126        WordBreakIteratorWithWordType(self)
127    }
128
129    /// Returns `true` when the segment preceding the current boundary is word-like,
130    /// such as letters, numbers, or CJKV ideographs.
131    #[inline]
132    pub fn is_word_like(&self) -> bool {
133        self.word_type().is_word_like()
134    }
135}

152/// Supports loading word break data, and creating word break iterators for different string
153/// encodings.
154///
155/// Most segmentation methods live on [`WordSegmenterBorrowed`], which can be obtained via
156/// [`WordSegmenter::new_auto()`] (etc) or [`WordSegmenter::as_borrowed()`].
157///
158/// Word segmenter is currently compatible with [Unicode Standard Annex #29][UAX29] (Version 17.0.0).
159///
160/// [UAX29]: https://www.unicode.org/reports/tr29/tr29-47.html
161///
162/// # Content Locale
163///
164/// You can optionally provide a _content locale_ to the [`WordSegmenter`] constructor. If you
165/// have information on the language of the text being segmented, providing this hint can
166/// produce higher-quality results.
167///
168/// If you have a content locale, use [`WordBreakOptions`] and a constructor beginning with `try_new`.
169/// If you do not have a content locale use [`WordBreakInvariantOptions`] and a constructor
170/// beginning with `new`.
171///
172/// # Examples
173///
174/// Segment a string:
175///
176/// ```rust
177/// use icu::segmenter::WordSegmenter;
178///
179/// let segmenter = WordSegmenter::new_auto(Default::default());
180///
181/// let breakpoints: Vec<usize> =
182///     segmenter.segment_str("Hello World").collect();
183/// assert_eq!(&breakpoints, &[0, 5, 6, 11]);
184/// ```
185///
186/// Segment a Latin1 byte string with a content locale:
187///
188/// ```rust
189/// use icu::locale::langid;
190/// use icu::segmenter::WordSegmenter;
191/// use icu::segmenter::options::WordBreakOptions;
192///
193/// let mut options = WordBreakOptions::default();
194/// let langid = &langid!("en");
195/// options.content_locale = Some(langid);
196/// let segmenter = WordSegmenter::try_new_auto(options).unwrap();
197///
198/// let breakpoints: Vec<usize> = segmenter
199///     .as_borrowed()
200///     .segment_latin1(b"Hello World")
201///     .collect();
202/// assert_eq!(&breakpoints, &[0, 5, 6, 11]);
203/// ```
204///
205/// Successive boundaries can be used to retrieve the segments.
206/// In particular, the first boundary is always 0, and the last one is the
207/// length of the segmented text in code units.
208///
209/// ```rust
210/// # use icu::segmenter::{WordSegmenter, options::WordBreakInvariantOptions};
211/// # let segmenter = WordSegmenter::new_auto(WordBreakInvariantOptions::default());
212/// use itertools::Itertools;
213/// let text = "Mark’d ye his words?";
214/// let segments: Vec<&str> = segmenter
215///     .segment_str(text)
216///     .tuple_windows()
217///     .map(|(i, j)| &text[i..j])
218///     .collect();
219/// assert_eq!(
220///     &segments,
221///     &["Mark’d", " ", "ye", " ", "his", " ", "words", "?"]
222/// );
223/// ```
224///
225/// Not all segments delimited by word boundaries are words; some are interword
226/// segments such as spaces and punctuation.
227/// The [`WordBreakIterator::word_type()`] of a boundary can be used to
228/// classify the preceding segment; [`WordBreakIterator::iter_with_word_type()`]
229/// associates each boundary with its status.
230/// ```rust
231/// # use itertools::Itertools;
232/// # use icu::segmenter::WordSegmenter;
233/// # use icu::segmenter::options::{WordType, WordBreakInvariantOptions};
234/// # let segmenter = WordSegmenter::new_auto(WordBreakInvariantOptions::default());
235/// # let text = "Mark’d ye his words?";
236/// let words: Vec<&str> = segmenter
237///     .segment_str(text)
238///     .iter_with_word_type()
239///     .tuple_windows()
240///     .filter(|(_, (_, segment_type))| segment_type.is_word_like())
241///     .map(|((i, _), (j, _))| &text[i..j])
242///     .collect();
243/// assert_eq!(&words, &["Mark’d", "ye", "his", "words"]);
244/// ```
245#[derive(Debug)]
246pub struct WordSegmenter(WordSegmenterInner);
247
248#[derive(Debug)]
249enum WordSegmenterInner {
250    V1 {
251        payload: DataPayload<SegmenterBreakWordV1>,
252        complex: ComplexPayloads,
253        payload_locale_override: Option<DataPayload<SegmenterBreakWordOverrideV1>>,
254    },
255    #[cfg(feature = "unstable")]
256    V2 {
257        payload: DataPayload<SegmenterBreakWordV2>,
258        complex: ComplexPayloads,
259    },
260}

924fn handle_complex_utf8<T>(
925    iter: &mut crate::rule_segmenter_v1::RuleBreakIterator<'_, '_, T>,
926    left_codepoint: T::CharType,
927) -> Option<usize>
928where
929    T: RuleBreakType<CharType = char>,
930{
931    // word segmenter doesn't define break rules for some scripts such as Thai.
932    let start_iter = iter.iter.clone();
933    let start_point = iter.current_pos_data;
934    let mut s = String::new();
935    s.push(left_codepoint);
936    loop {
937        debug_assert!(!iter.is_eof());
938        s.push(iter.get_current_codepoint()?);
939        iter.advance_iter();
940        if let Some(current_break_property) = iter.get_current_break_property() {
941            if current_break_property != iter.data.complex_property {
942                break;
943            }
944        } else {
945            // EOF
946            break;
947        }
948    }
949
950    // Restore iterator to move to head of complex string
951    iter.iter = start_iter;
952    iter.current_pos_data = start_point;
953    #[expect(clippy::unwrap_used)] // iter.complex present for word segmenter
954    let breaks = iter.complex.unwrap().segment_str(&s);
955    iter.result_cache = breaks;
956    let first_pos = *iter.result_cache.first()?;
957    let mut i = left_codepoint.len_utf8();
958    loop {
959        if i == first_pos {
960            // Re-calculate breaking offset
961            iter.result_cache = iter.result_cache.iter().skip(1).map(|r| r - i).collect();
962            return iter.get_current_position();
963        }
964        debug_assert!(
965            i < first_pos,
966            "we should always arrive at first_pos: near index {:?}",
967            iter.get_current_position()
968        );
969        i += iter.get_current_codepoint().map_or(0, T::char_len);
970        iter.advance_iter();
971        if iter.is_eof() {
972            iter.result_cache.clear();
973            return Some(iter.len);
974        }
975    }
976}

978fn handle_complex_utf16<T>(
979    iter: &mut crate::rule_segmenter_v1::RuleBreakIterator<'_, '_, T>,
980    left_codepoint: T::CharType,
981) -> Option<usize>
982where
983    T: RuleBreakType<CharType = u32>,
984{
985    // word segmenter doesn't define break rules for some scripts such as Thai.
986    let start_iter = iter.iter.clone();
987    let start_point = iter.current_pos_data;
988    let mut s = vec![left_codepoint as u16];
989    loop {
990        debug_assert!(!iter.is_eof());
991        s.push(iter.get_current_codepoint()? as u16);
992        iter.advance_iter();
993        if let Some(current_break_property) = iter.get_current_break_property() {
994            if current_break_property != iter.data.complex_property {
995                break;
996            }
997        } else {
998            // EOF
999            break;
1000        }
1001    }
1002
1003    // Restore iterator to move to head of complex string
1004    iter.iter = start_iter;
1005    iter.current_pos_data = start_point;
1006    #[expect(clippy::unwrap_used)] // iter.complex present for word segmenter
1007    let breaks = iter.complex.unwrap().segment_utf16(&s);
1008    iter.result_cache = breaks;
1009    // result_cache vector is utf-16 index that is in BMP.
1010    let first_pos = *iter.result_cache.first()?;
1011    let mut i = 1;
1012    loop {
1013        if i == first_pos {
1014            // Re-calculate breaking offset
1015            iter.result_cache = iter.result_cache.iter().skip(1).map(|r| r - i).collect();
1016            return iter.get_current_position();
1017        }
1018        debug_assert!(
1019            i < first_pos,
1020            "we should always arrive at first_pos: near index {:?}",
1021            iter.get_current_position()
1022        );
1023        i += 1;
1024        iter.advance_iter();
1025        if iter.is_eof() {
1026            iter.result_cache.clear();
1027            return Some(iter.len);
1028        }
1029    }
1030}
