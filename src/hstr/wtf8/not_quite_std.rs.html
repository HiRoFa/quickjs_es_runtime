24// UTF-8 ranges and tags for encoding characters
25// Copied from 48d5fe9ec560b53b1f5069219b0d62015e1de5ba^:src/libcore/char.rs
26const TAG_CONT: u8 = 0b1000_0000;
27const TAG_TWO_B: u8 = 0b1100_0000;
28const TAG_THREE_B: u8 = 0b1110_0000;
29const TAG_FOUR_B: u8 = 0b1111_0000;
30const MAX_ONE_B: u32 = 0x80;
31const MAX_TWO_B: u32 = 0x800;
32const MAX_THREE_B: u32 = 0x10000;
33
34/// Encode a code point into potentially uninitialized storage.
35///
36/// On success, exactly the first returned number of elements in `dst` are
37/// initialized. Copied from
38/// 48d5fe9ec560b53b1f5069219b0d62015e1de5ba^:src/libcore/char.rs.
39#[inline]
40fn encode_utf8_raw(code: u32, dst: &mut [MaybeUninit<u8>]) -> Option<usize> {
41    // Marked #[inline] to allow llvm optimizing it away
42    if code < MAX_ONE_B && !dst.is_empty() {
43        dst[0].write(code as u8);
44        Some(1)
45    } else if code < MAX_TWO_B && dst.len() >= 2 {
46        dst[0].write((code >> 6 & 0x1f) as u8 | TAG_TWO_B);
47        dst[1].write((code & 0x3f) as u8 | TAG_CONT);
48        Some(2)
49    } else if code < MAX_THREE_B && dst.len() >= 3 {
50        dst[0].write((code >> 12 & 0x0f) as u8 | TAG_THREE_B);
51        dst[1].write((code >> 6 & 0x3f) as u8 | TAG_CONT);
52        dst[2].write((code & 0x3f) as u8 | TAG_CONT);
53        Some(3)
54    } else if dst.len() >= 4 {
55        dst[0].write((code >> 18 & 0x07) as u8 | TAG_FOUR_B);
56        dst[1].write((code >> 12 & 0x3f) as u8 | TAG_CONT);
57        dst[2].write((code >> 6 & 0x3f) as u8 | TAG_CONT);
58        dst[3].write((code & 0x3f) as u8 | TAG_CONT);
59        Some(4)
60    } else {
61        None
62    }
63}

65/// Copied from 48d5fe9ec560b53b1f5069219b0d62015e1de5ba^:src/libcore/char.rs
66#[inline]
67fn encode_utf16_raw(mut ch: u32, dst: &mut [u16]) -> Option<usize> {
68    // Marked #[inline] to allow llvm optimizing it away
69    if (ch & 0xffff) == ch && !dst.is_empty() {
70        // The BMP falls through (assuming non-surrogate, as it should)
71        dst[0] = ch as u16;
72        Some(1)
73    } else if dst.len() >= 2 {
74        // Supplementary planes break into surrogates.
75        ch -= 0x1_0000;
76        dst[0] = 0xd800 | ((ch >> 10) as u16);
77        dst[1] = 0xdc00 | ((ch as u16) & 0x3ff);
78        Some(2)
79    } else {
80        None
81    }
82}

84/// Copied from core::str::next_code_point
85#[inline]
86pub fn next_code_point(bytes: &mut slice::Iter<u8>) -> Option<u32> {
87    // Decode UTF-8
88    let x = match bytes.next() {
89        None => return None,
90        Some(&next_byte) if next_byte < 128 => return Some(next_byte as u32),
91        Some(&next_byte) => next_byte,
92    };
93
94    // Multibyte case follows
95    // Decode from a byte combination out of: [[[x y] z] w]
96    // NOTE: Performance is sensitive to the exact formulation here
97    let init = utf8_first_byte(x, 2);
98    let y = unwrap_or_0(bytes.next());
99    let mut ch = utf8_acc_cont_byte(init, y);
100    if x >= 0xe0 {
101        // [[x y z] w] case
102        // 5th bit in 0xE0 .. 0xEF is always clear, so `init` is still valid
103        let z = unwrap_or_0(bytes.next());
104        let y_z = utf8_acc_cont_byte((y & CONT_MASK) as u32, z);
105        ch = init << 12 | y_z;
106        if x >= 0xf0 {
107            // [x y z w] case
108            // use only the lower 3 bits of `init`
109            let w = unwrap_or_0(bytes.next());
110            ch = (init & 7) << 18 | utf8_acc_cont_byte(y_z, w);
111        }
112    }
113
114    Some(ch)
115}

117#[inline]
118fn utf8_first_byte(byte: u8, width: u32) -> u32 {
119    (byte & (0x7f >> width)) as u32
120}

122/// Return the value of `ch` updated with continuation byte `byte`.
123#[inline]
124fn utf8_acc_cont_byte(ch: u32, byte: u8) -> u32 {
125    (ch << 6) | (byte & CONT_MASK) as u32
126}

128#[inline]
129fn unwrap_or_0(opt: Option<&u8>) -> u8 {
130    match opt {
131        Some(&byte) => byte,
132        None => 0,
133    }
134}

139/// Copied from String::push
140/// This does **not** include the WTF-8 concatenation check.
141#[inline]
142pub fn push_code_point(string: &mut Wtf8Buf, code_point: CodePoint) {
143    let cur_len = string.len();
144    // This may use up to 4 bytes.
145    string.reserve(4);
146
147    let used = {
148        let spare = string.bytes.spare_capacity_mut();
149        encode_utf8_raw(code_point.to_u32(), &mut spare[..4])
150            .expect("four bytes are sufficient to encode any code point")
151    };
152
153    // SAFETY: `reserve(4)` guarantees capacity for `cur_len + 4`, and
154    // `encode_utf8_raw` initialized exactly the first `used` spare bytes.
155    unsafe {
156        string.bytes.set_len(cur_len + used);
157    }
158}

160/// Copied from core::str::StrPrelude::is_char_boundary
161#[inline]
162pub fn is_code_point_boundary(slice: &Wtf8, index: usize) -> bool {
163    if index == slice.len() {
164        return true;
165    }
166    match slice.bytes.get(index) {
167        None => false,
168        Some(&b) => !(128u8..192u8).contains(&b),
169    }
170}

175    mem::transmute(slice::from_raw_parts(

188/// Copied from core::str::Utf16CodeUnits::next
189pub fn next_utf16_code_unit(iter: &mut IllFormedUtf16CodeUnits) -> Option<u16> {
190    if iter.extra != 0 {
191        let tmp = iter.extra;
192        iter.extra = 0;
193        return Some(tmp);
194    }
195
196    let mut buf = [0u16; 2];
197    iter.code_points.next().map(|code_point| {
198        let n = encode_utf16_raw(code_point.to_u32(), &mut buf).unwrap_or(0);
199        if n == 2 {
200            iter.extra = buf[1];
201        }
202        buf[0]
203    })
204}

215/// Copied from src/librustc_unicode/char.rs
216#[inline]
217pub fn decode_utf16<I: IntoIterator<Item = u16>>(iterable: I) -> DecodeUtf16<I::IntoIter> {
218    DecodeUtf16 {
219        iter: iterable.into_iter(),
220        buf: None,
221    }
222}

224/// Copied from src/librustc_unicode/char.rs
225impl<I: Iterator<Item = u16>> Iterator for DecodeUtf16<I> {
226    type Item = Result<char, u16>;
227
228    fn next(&mut self) -> Option<Result<char, u16>> {
229        let u = match self.buf.take() {
230            Some(buf) => buf,
231            None => self.iter.next()?,
232        };
233
234        if !(0xd800..=0xdfff).contains(&u) {
235            // not a surrogate
236            Some(Ok(unsafe { char::from_u32_unchecked(u as u32) }))
237        } else if u >= 0xdc00 {
238            // a trailing surrogate
239            Some(Err(u))
240        } else {
241            let u2 = match self.iter.next() {
242                Some(u2) => u2,
243                // eof
244                None => return Some(Err(u)),
245            };
246            if !(0xdc00..=0xdfff).contains(&u2) {
247                // not a trailing surrogate so we're not a valid
248                // surrogate pair, so rewind to redecode u2 next time.
249                self.buf = Some(u2);
250                return Some(Err(u));
251            }
252
253            // all ok, so lets decode it.
254            let c = (((u - 0xd800) as u32) << 10 | (u2 - 0xdc00) as u32) + 0x1_0000;
255            Some(Ok(unsafe { char::from_u32_unchecked(c) }))
256        }
257    }
258
259    #[inline]
260    fn size_hint(&self) -> (usize, Option<usize>) {
261        let (low, high) = self.iter.size_hint();
262        // we could be entirely valid surrogates (2 elements per
263        // char), or entirely non-surrogates (1 element per char)
264        (low / 2, high)
265    }
266}

268/// Check if a byte is a valid UTF-8 continuation byte.
269#[inline]
270fn is_continuation_byte(byte: u8) -> bool {
271    byte & 0xc0 == 0x80
272}
273
274/// Check if bytes at position form a surrogate pair. A surrogate pair should be
275/// encoded as 4-byte sequence in UTF-8 instead of two separate 3-byte
276/// sequences.
277#[inline]
278fn is_surrogate_pair(bytes: &[u8], pos: usize) -> bool {
279    if pos + 5 >= bytes.len() {
280        return false;
281    }
282    bytes[pos] == 0xed
283        && (0xa0..0xb0).contains(&bytes[pos + 1])
284        && is_continuation_byte(bytes[pos + 2])
285        && bytes[pos + 3] == 0xed
286        && (0xb0..0xc0).contains(&bytes[pos + 4])
287        && is_continuation_byte(bytes[pos + 5])
288}

290/// Validate that bytes represent well-formed WTF-8.
291///
292/// This checks that:
293/// - All bytes form valid UTF-8 sequences OR valid surrogate code point
294///   encodings
295/// - Surrogate pairs are not encoded as two separate 3-byte sequences
296pub fn validate_wtf8(bytes: &[u8]) -> bool {
297    let mut i = 0;
298    while i < bytes.len() {
299        let byte = bytes[i];
300
301        // ASCII range (1-byte sequence)
302        if byte < 0x80 {
303            i += 1;
304            continue;
305        }
306
307        // 2-byte sequence
308        if byte < 0xe0 {
309            if i + 1 >= bytes.len() || !is_continuation_byte(bytes[i + 1]) || byte < 0xc2 {
310                return false; // Truncated, invalid continuation, or overlong
311            }
312            i += 2;
313            continue;
314        }
315
316        // 3-byte sequence
317        if byte < 0xf0 {
318            if i + 2 >= bytes.len()
319                || !is_continuation_byte(bytes[i + 1])
320                || !is_continuation_byte(bytes[i + 2])
321            {
322                return false; // Truncated or invalid continuation bytes
323            }
324
325            let b2 = bytes[i + 1];
326
327            // Check for overlong encoding
328            if byte == 0xe0 && b2 < 0xa0 {
329                return false;
330            }
331
332            // Check for surrogate pair (lead + trail) - invalid in WTF-8
333            if is_surrogate_pair(bytes, i) {
334                return false;
335            }
336
337            i += 3;
338            continue;
339        }
340
341        // 4-byte sequence
342        if byte < 0xf8 {
343            if i + 3 >= bytes.len()
344                || !is_continuation_byte(bytes[i + 1])
345                || !is_continuation_byte(bytes[i + 2])
346                || !is_continuation_byte(bytes[i + 3])
347            {
348                return false; // Truncated or invalid continuation bytes
349            }
350
351            let b2 = bytes[i + 1];
352
353            // Check for overlong encoding and valid range
354            if (byte == 0xf0 && b2 < 0x90) || (byte == 0xf4 && b2 >= 0x90) || byte > 0xf4 {
355                return false;
356            }
357
358            i += 4;
359            continue;
360        }
361
362        // Invalid byte (>= 0xF8)
363        return false;
364    }
365
366    true
