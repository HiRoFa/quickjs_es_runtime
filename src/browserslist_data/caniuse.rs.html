27static VERSION_LIST: LazyLock<Vec<VersionDetail>> = LazyLock::new(|| {
28    VERSION_LIST_VERSION

56/// Every version carries its own global usage, and caniuse lists a usage for exactly the
57/// versions in the version list, so the usage table is that data sorted by usage.
58static CANIUSE_GLOBAL_USAGE: LazyLock<Vec<(&'static str, &'static str, f32)>> =
59    LazyLock::new(|| {
60        let mut usage = CANIUSE_BROWSERS
61            .iter()
62            .flat_map(|(name, stat)| {
63                stat.version_list()
64                    .iter()
65                    .map(|version| (name.as_str(), version.version(), version.global_usage))
66            })
67            .collect::<Vec<_>>();
68        usage.sort_unstable_by(|(.., a), (.., b)| b.total_cmp(a));
69        usage
70    });

72/// Position of every version within its own browser's version list. The feature tables
73/// store one flag per version in that same order, so a position is also a flag offset.
74static VERSION_INDEXES: LazyLock<AHashMap<&'static str, AHashMap<&'static str, u32>>> =
75    LazyLock::new(|| {
76        CANIUSE_BROWSERS
77            .iter()
78            .map(|(name, stat)| {
79                let indexes = stat
80                    .version_list()
81                    .iter()
82                    .enumerate()
83                    .map(|(index, version)| (version.version(), index as u32))
84                    .collect();
85                (name.as_str(), indexes)
86            })
87            .collect()
88    });

90static BROWSER_VERSION_ALIASES: LazyLock<
91    AHashMap<&'static str, AHashMap<&'static str, &'static str>>,
92> = LazyLock::new(|| {
93    let mut aliases = CANIUSE_BROWSERS
94        .iter()
95        .filter_map(|(name, stat)| {
96            let name = name.as_str();
97            let aliases = stat
98                .version_list()
99                .iter()
100                .filter_map(|version| {
101                    version
102                        .version
103                        .as_str()
104                        .split_once('-')
105                        .map(|(bottom, top)| (bottom, top, version.version))
106                })
107                .fold(
108                    AHashMap::<&str, &str>::new(),
109                    move |mut aliases, (bottom, top, version)| {
110                        let _ = aliases.insert(bottom, version.as_str());
111                        let _ = aliases.insert(top, version.as_str());
112                        aliases
113                    },
114                );
115            if aliases.is_empty() {
116                None
117            } else {
118                Some((name, aliases))
119            }
120        })
121        .collect::<AHashMap<&'static str, _>>();
122    let _ = aliases.insert("op_mob", {
123        let mut aliases = AHashMap::new();
124        let _ = aliases.insert("59", "58");
125        aliases
126    });
127    aliases
128});

130static ANDROID_TO_DESKTOP: LazyLock<Vec<VersionDetail>> = LazyLock::new(|| {
131    let chrome = CANIUSE_BROWSERS.get("chrome").unwrap();
132    let android = CANIUSE_BROWSERS.get("android").unwrap();
133
134    let chrome_point = chrome
135        .version_list()
136        .binary_search_by_key(&(ANDROID_EVERGREEN_FIRST as usize), |probe| {
137            probe.version.as_str().parse::<usize>().unwrap()
138        })
139        .unwrap();
140
141    android
142        .version_list()
143        .iter()
144        .filter(|version| {
145            let version = version.version.as_str();
146            version.starts_with("2.")
147                || version.starts_with("3.")
148                || version.starts_with("4.")
149                || version == "3"
150                || version == "4"
151        })
152        .chain(chrome.version_list().iter().skip(chrome_point))
153        .cloned()
154        .collect()
155});

157pub fn get_browser_stat(
158    name: &str,
159    mobile_to_desktop: bool,
160) -> Option<(&'static str, &'static [VersionDetail])> {
161    let name = if name.bytes().all(|b| b.is_ascii_lowercase()) {
162        Cow::from(name)
163    } else {
164        Cow::from(name.to_ascii_lowercase())
165    };
166    let name = get_browser_alias(&name);
167
168    if mobile_to_desktop {
169        if let Some(desktop_name) = to_desktop_name(name) {
170            match name {
171                "android" => Some(("android", &*ANDROID_TO_DESKTOP)),
172                "op_mob" => {
173                    let stat = CANIUSE_BROWSERS.get("opera").unwrap();
174                    Some(("op_mob", stat.version_list()))
175                }
176                _ => CANIUSE_BROWSERS.get(desktop_name).map(|stat| {
177                    (
178                        get_mobile_by_desktop_name(desktop_name),
179                        stat.version_list(),
180                    )
181                }),
182            }
183        } else {
184            CANIUSE_BROWSERS
185                .get_key_value(name)
186                .map(|(k, v)| (k.as_str(), v.version_list()))
187        }
188    } else {
189        CANIUSE_BROWSERS
190            .get_key_value(name)
191            .map(|(k, v)| (k.as_str(), v.version_list()))
192    }
193}

195pub fn iter_browser_stat(
196    mobile_to_desktop: bool,
197) -> impl Iterator<Item = (&'static str, &'static [VersionDetail])> {
198    CANIUSE_BROWSERS.iter().filter_map(move |(name, stat)| {
199        match (
200            mobile_to_desktop,
201            to_desktop_name(name.as_str()),
202            name.as_str(),
203        ) {
204            (false, _, _) | (true, None, _) => Some((name.as_str(), stat.version_list())),
205            (true, Some(_), "android") => Some(("android", &*ANDROID_TO_DESKTOP)),
206            (true, Some(_), "op_mob") => {
207                let stat = CANIUSE_BROWSERS.get("opera").unwrap();
208                Some(("op_mob", stat.version_list()))
209            }
210            (true, Some(desktop_name), _) => CANIUSE_BROWSERS.get(desktop_name).map(|stat| {
211                (
212                    get_mobile_by_desktop_name(desktop_name),
213                    stat.version_list(),
214                )
215            }),
216        }
217    })
218}

228fn get_browser_alias(name: &str) -> &str {
229    match name {
230        "fx" | "ff" => "firefox",
231        "ios" => "ios_saf",
232        "explorer" => "ie",
233        "blackberry" => "bb",
234        "explorermobile" => "ie_mob",
235        "operamini" => "op_mini",
236        "operamobile" => "op_mob",
237        "chromeandroid" => "and_chr",
238        "firefoxandroid" => "and_ff",
239        "ucandroid" => "and_uc",
240        "qqandroid" => "and_qq",
241        _ => name,
242    }
243}

245pub fn to_desktop_name(name: &str) -> Option<&'static str> {
246    match name {
247        "and_chr" | "android" => Some("chrome"),
248        "and_ff" => Some("firefox"),
249        "ie_mob" => Some("ie"),
250        _ => None,
251    }
252}
253
254fn get_mobile_by_desktop_name(name: &str) -> &'static str {
255    match name {
256        "chrome" => "and_chr", // "android" has been handled as a special case
257        "firefox" => "and_ff",
258        "ie" => "ie_mob",
259        "opera" => "op_mob",
260        _ => unreachable!(),
261    }
262}

264pub fn normalize_version<'a>(
265    name: &'a str,
266    version_list: &'static [VersionDetail],
267    version: &'a str,
268) -> Option<&'a str> {
269    if version_list.iter().any(|v| v.version.as_str() == version) {
270        Some(version)
271    } else if let Some(version) = get_browser_version_alias(name, version) {
272        Some(version)
273    } else if version_list.len() == 1 {
274        version_list.first().map(|s| s.version.as_str())
275    } else {
276        None
277    }
278}
