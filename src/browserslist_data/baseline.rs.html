1//! Baseline browser data, generated from the [`baseline-browser-mapping`]
2//! npm package's Baseline support timeline.
3//!
4//! [`baseline-browser-mapping`]: https://www.npmjs.com/package/baseline-browser-mapping
5
6use crate::{decode_browser_name, utils::PooledStr};
7#[cfg(feature = "deflate")]
8use std::sync::LazyLock;

10include!("generated/baseline.rs");

12/// caniuse names of the seven core Baseline browsers; the remaining browsers
13/// in the dataset are downstream browsers sharing a core browser's engine.
14static CORE_BROWSERS: &[&str] = &[
15    "and_chr", "and_ff", "chrome", "edge", "firefox", "ios_saf", "safari",
16];
17
18pub fn is_core_browser(name: &str) -> bool {
19    CORE_BROWSERS.contains(&name)
20}
21
22/// All browsers tracked by the Baseline dataset, by caniuse name.
23pub fn browsers() -> impl Iterator<Item = &'static str> {
24    BASELINE_BROWSERS.iter().map(|&id| decode_browser_name(id))
25}
26
27/// Minimum compatible versions (by caniuse browser name) for the Baseline
28/// feature set as of `cutoff_date`, a baseline-low threshold encoded as
29/// decimal `yyyymmdd` (the query date minus 30 months for widely-available
30/// queries).
31///
32/// Returns `None` if the date predates the first Baseline feature; every
33/// version of every browser is considered compatible in that case.
34pub fn min_versions_on(
35    cutoff_date: u32,
36) -> Option<impl Iterator<Item = (&'static str, &'static str)>> {
37    let index = BASELINE_TIMELINE_DATE.partition_point(|date| *date <= cutoff_date);
38    index.checked_sub(1).map(|index| {
39        let end = BASELINE_TIMELINE_WIDTH[..=index]
40            .iter()
41            .map(|width| u16::from(*width))
42            .sum::<u16>();
43        let start = end - u16::from(BASELINE_TIMELINE_WIDTH[index]);
44        BASELINE_VERSION_BROWSER[usize::from(start)..usize::from(end)]
45            .iter()
46            .zip(&BASELINE_VERSION_VERSION[usize::from(start)..usize::from(end)])
47            .map(|(id, version)| {
48                (
49                    decode_browser_name(*id),
50                    PooledStr(BASELINE_VERSION_TABLE[usize::from(*version)]).as_str(),
51                )
52            })
53    })
54}
