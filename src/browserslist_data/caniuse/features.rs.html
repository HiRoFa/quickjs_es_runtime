1use super::PooledStr;
2use crate::{decode_browser_name, utils::BinMap};
3use ahash::AHashMap;
4use std::sync::LazyLock;
5
6#[derive(Clone, Copy)]
7pub struct Feature(u32, u32);
8
9#[derive(Clone, Copy)]
10pub struct VersionList {
11    // Where each of the browser's versions sits in its version list, which is also where
12    // that version's flag sits within the browser's run of flags.
13    indexes: &'static AHashMap<&'static str, u32>,
14    // Index of this browser's first flag within FEATURES_STAT_FLAGS.
15    base: u32,
16}
17
18// ```rust
19// static FEATURES_KEY/WIDTH: parallel feature name and browser-list widths
20//
21// static FEATURES_STAT_FLAGS: &[u8]; // support flag
22// static FEATURES_STAT_BROWSERS: &[u8]; // browser name id
23// ```
24include!("../generated/caniuse-feature-matching.rs");
25
26static FEATURES: LazyLock<Vec<(PooledStr, Feature)>> = LazyLock::new(|| {
27    FEATURES_KEY
28        .iter()
29        .zip(&*FEATURES_WIDTH)
30        .scan(0, |start, (key, width)| {
31            let end = *start + u32::from(*width);
32            let feature = (PooledStr(*key), Feature(*start, end));
33            *start = end;
34            Some(feature)
35        })
36        .collect()
37});
38
39// caniuse states every feature for every version of a browser (checked by
40// generate-data), so a browser's flags line up with its version list position by
41// position, and each browser's run of flags is as long as its version list.
42static FEATURES_STAT_FLAG_START: LazyLock<Vec<u32>> = LazyLock::new(|| {
43    let mut start = 0;
44    FEATURES_STAT_BROWSERS
45        .iter()
46        .map(|id| {
47            let base = start;
48            start += browser_of(*id).version_list().len() as u32;
49            base
50        })
51        .collect()
52});
53
54pub fn get_feature_stat(name: &str) -> Option<Feature> {
55    BinMap(&FEATURES).get(name).copied()
56}
57
58fn browser_of(id: u8) -> &'static super::BrowserStat {
59    super::browser_stat(decode_browser_name(id)).expect("feature refers to unknown browser")
60}
61
62fn version_list_at(index: usize) -> VersionList {
63    let name = decode_browser_name(FEATURES_STAT_BROWSERS[index]);
64    VersionList {
65        indexes: super::version_indexes(name).expect("feature refers to unknown browser"),
66        base: FEATURES_STAT_FLAG_START[index],
67    }
68}
69
70impl Feature {
71    pub fn get(&self, browser: &str) -> Option<VersionList> {
72        let range = (self.0 as usize)..(self.1 as usize);
73        let index = FEATURES_STAT_BROWSERS[range.clone()]
74            .binary_search_by_key(&browser, |&k| decode_browser_name(k))
75            .ok()?;
76        Some(version_list_at(range.start + index))
77    }
78
79    pub fn iter(&self) -> impl Iterator<Item = (&'static str, VersionList)> {
80        let start = self.0 as usize;
81        FEATURES_STAT_BROWSERS[start..(self.1 as usize)]
82            .iter()
83            .enumerate()
84            .map(move |(offset, &id)| (decode_browser_name(id), version_list_at(start + offset)))
85    }
86}
87
88impl VersionList {
89    pub fn get(&self, version: &str) -> Option<u8> {
90        let position = *self.indexes.get(version)?;
91        Some(FEATURES_STAT_FLAGS[self.base as usize + position as usize])
92    }
