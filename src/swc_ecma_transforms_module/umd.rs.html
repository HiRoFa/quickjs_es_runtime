61pub struct Umd {

63    unresolved_mark: Mark,
64    config: BuiltConfig,
65    resolver: Resolver,
66

74impl VisitMut for Umd {
75    noop_visit_mut_type!(fail);
76
77    fn visit_mut_module(&mut self, module: &mut Module) {
78        let module_items = &mut module.body;
79
80        if !self.config.config.allow_top_level_this {
81            top_level_this(module_items, *Expr::undefined(DUMMY_SP));
82        }
83
84        let SyntaxStrippedModule {
85            directives,
86            has_use_strict,
87            requested_modules,
88            local_export_entries,
89            export_assign,
90            body,
91            has_module_syntax,
92        } = syntax_strip::lower(
93            SourceModule::collect(module_items.take()),
94            self.const_var_kind,
95        );
96
97        let mut stmts: Vec<Stmt> = Vec::with_capacity(body.len() + 4);
98        stmts.extend(directives);
99
100        // "use strict";
101        if self.config.config.strict_mode && !has_use_strict {
102            stmts.push(use_strict());
103        }
104
105        let import_interop = self.config.config.import_interop();
106
107        let is_export_assign = export_assign.is_some();
108
109        if has_module_syntax && !import_interop.is_none() && !is_export_assign {
110            stmts.push(define_es_module(self.exports()));
111        }
112
113        let mut import_map = Default::default();
114        let ts_import_equals_exports: FxHashSet<Atom> = body
115            .iter()
116            .filter_map(exported_external_ts_import_equals_name)
117            .cloned()
118            .collect();
119
120        stmts.extend(self.handle_import_export(
121            &mut import_map,
122            requested_modules,
123            local_export_entries,
124            &ts_import_equals_exports,
125            is_export_assign,
126        ));
127
128        stmts.extend(
129            body.into_iter()
130                .filter_map(|item| self.lower_body_item(&mut import_map, item)),
131        );
132
133        if let Some(export_assign) = export_assign {
134            let return_stmt = ReturnStmt {
135                span: DUMMY_SP,
136                arg: Some(export_assign),
137            };
138
139            stmts.push(return_stmt.into());
140        }
141
142        rewrite_import_bindings(&mut stmts, import_map, Default::default());
143
144        *module_items = emit::emit(
145            stmts,
146            emit::EmitModule {
147                span: module.span,
148                is_export_assign,
149                exports: self.exports.clone(),
150                deps: self.dep_list.take(),
151            },
152            &self.cm,
153            &self.config,
154            self.unresolved_mark,
155            &self.resolver,
156        );
157    }
158}

160impl Umd {
161    fn handle_import_export(
162        &mut self,
163        import_map: &mut ImportMap,
164        requested_modules: RequestedModules,
165        local_export_entries: LocalExportEntries,
166        ts_import_equals_exports: &FxHashSet<Atom>,
167        is_export_assign: bool,
168    ) -> impl Iterator<Item = Stmt> {
169        let import_interop = self.config.config.import_interop();
170
171        let mut stmts = Vec::with_capacity(requested_modules.len());
172
173        let mut export_bindings = local_export_entries.into_iter().collect();
174
175        requested_modules.into_iter().for_each(
176            |(
177                module_request,
178                RequestedModule {
179                    span: src_span,
180                    entries: module_entries,
181                    usage: mut module_usage,
182                    ..
183                },
184            )| {
185                let src = module_request.src().clone();
186                let is_node_default = !module_usage.has_named() && import_interop.is_node();
187
188                if import_interop.is_none() {
189                    module_usage -= ModuleRequestUsage::NAMESPACE;
190                }
191
192                let need_re_export = module_usage.has_star_export();
193                let need_interop = module_usage.needs_interop();
194
195                let mod_ident = private_ident!(local_name_for_src(&src));
196
197                self.dep_list.push((mod_ident.clone(), src, src_span));
198
199                module_entries.reduce(
200                    import_map,
201                    &mut export_bindings,
202                    &mod_ident,
203                    &mut false,
204                    is_node_default,
205                );
206
207                // _export_star(mod, exports);
208                let mut import_expr: Expr = if need_re_export {
209                    helper_expr!(export_star).as_call(
210                        DUMMY_SP,
211                        vec![mod_ident.clone().as_arg(), self.exports().as_arg()],
212                    )
213                } else {
214                    mod_ident.clone().into()
215                };
216
217                // _introp(mod);
218                if need_interop {
219                    import_expr = match import_interop {
220                        ImportInterop::Swc if module_usage.needs_interop() => {
221                            if module_usage.needs_namespace_object() {
222                                helper_expr!(interop_require_wildcard)
223                            } else {
224                                helper_expr!(interop_require_default)
225                            }
226                            .as_call(PURE_SP, vec![import_expr.as_arg()])
227                        }
228                        ImportInterop::Node if module_usage.needs_namespace_object() => {
229                            helper_expr!(interop_require_wildcard)
230                                .as_call(PURE_SP, vec![import_expr.as_arg(), true.as_arg()])
231                        }
232                        _ => import_expr,
233                    }
234                }
235
236                // mod = _introp(mod);
237                if need_interop {
238                    let stmt = import_expr
239                        .make_assign_to(op!("="), mod_ident.into())
240                        .into_stmt();
241                    stmts.push(stmt);
242                } else if need_re_export {
243                    stmts.push(import_expr.into_stmt());
244                }
245            },
246        );
247
248        let mut export_stmts = Default::default();
249
250        export_bindings.retain(|(export_name, _)| !ts_import_equals_exports.contains(export_name));
251
252        if !export_bindings.is_empty() && !is_export_assign {
253            sort_export_bindings(&mut export_bindings);
254
255            let exports = self.exports();
256
257            export_stmts = emit_export_stmts(exports, export_bindings);
258        }
259
260        export_stmts.into_iter().chain(stmts)
261    }
262
263    fn lower_body_item(&mut self, import_map: &mut ImportMap, item: ModuleItem) -> Option<Stmt> {
264        match item {
265            ModuleItem::Stmt(stmt) if !stmt.is_empty() => Some(stmt),
266            ModuleItem::ModuleDecl(ModuleDecl::TsImportEquals(import)) => {
267                self.lower_ts_import_equals(import_map, *import)
268            }
269            _ => None,
270        }
271    }
272
273    fn lower_ts_import_equals(
274        &mut self,
275        import_map: &mut ImportMap,
276        import: TsImportEqualsDecl,
277    ) -> Option<Stmt> {
278        let src = external_ts_import_equals_source(&import)?;
279        let src_span = src.span;
280        let request = src.value.to_atom_lossy().into_owned();
281        let TsImportEqualsDecl {
282            id: local,
283            is_export,
284            ..
285        } = import;
286        let param = if is_export {
287            local.clone().into_private()
288        } else {
289            local.clone()
290        };
291
292        self.dep_list
293            .push((param.clone(), request, (src_span, Default::default())));
294
295        if !is_export {
296            return None;
297        }
298
299        Some(self.emit_ts_import_equals(import_map, local, is_export, param.into()))
300    }
301
302    fn emit_ts_import_equals(
303        &mut self,
304        import_map: &mut ImportMap,
305        local: Ident,
306        is_export: bool,
307        value: Expr,
308    ) -> Stmt {
309        if is_export {
310            import_map.insert(local.to_id(), (self.exports(), Some(local.sym.clone())));
311            return value
312                .make_assign_to(op!("="), self.exports().make_member(local.into()).into())
313                .into_stmt();
314        }
315
316        value
317            .into_var_decl(self.const_var_kind, local.into())
318            .into()
319    }
320
321    fn exports(&mut self) -> Ident {
322        self.exports
323            .get_or_insert_with(|| private_ident!("exports"))
324            .clone()
325    }
