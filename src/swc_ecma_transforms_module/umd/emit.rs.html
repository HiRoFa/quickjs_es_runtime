17pub(super) fn emit(
18    stmts: Vec<Stmt>,
19    module: EmitModule,
20    cm: &Lrc<SourceMap>,
21    config: &BuiltConfig,
22    unresolved_mark: Mark,
23    resolver: &Resolver,
24) -> Vec<ModuleItem> {
25    let (adapter_fn_expr, factory_params) =
26        emit_adapter(&module, cm, config, unresolved_mark, resolver);
27
28    let factory_fn_expr: Expr = Function {
29        params: factory_params,
30        decorators: Default::default(),
31        span: DUMMY_SP,
32        body: Some(FunctionBody {
33            span: DUMMY_SP,
34            stmts,
35        }),
36        is_generator: false,
37        is_async: false,
38        ..Default::default()
39    }
40    .into();
41
42    vec![adapter_fn_expr
43        .as_call(
44            DUMMY_SP,
45            vec![
46                ThisExpr { span: DUMMY_SP }.as_arg(),
47                factory_fn_expr.as_arg(),
48            ],
49        )
50        .into_stmt()
51        .into()]
52}

54/// Builds the UMD adapter function and returns it with the factory params.
55///
56/// The adapter always emits `CommonJS`, AMD, and global-fallback branches:
57/// ```javascript
58/// if (typeof module === "object" && typeof module.exports === "object") { ... }
59/// else if (typeof define === "function" && define.amd) { ... }
60/// else if (global = typeof globalThis !== "undefined" ? globalThis : global || self) { ... }
61/// ```
62///
63/// For regular ES exports, `exports` is added as the first CommonJS/AMD
64/// dependency and factory param. The global fallback creates the configured
65/// export object and passes it as the first factory argument:
66/// ```javascript
67/// factory(exports, require("mod"));
68/// define(["exports", "mod"], factory);
69/// factory((global.lib = {}), global.mod);
70/// function (exports, mod) { ... }
71/// ```
72///
73/// For `export =`, there is no `exports` param. The factory returns the module
74/// value, `CommonJS` assigns that return value to `module.exports`, AMD only
75/// registers the dependency list, and the fallback branch assigns the factory
76/// result to the inferred global export name:
77/// ```javascript
78/// module.exports = factory(require("mod"));
79/// define(["mod"], factory);
80/// global.lib = factory(global.mod);
81/// function (mod) { ... }
82/// ```
83fn emit_adapter(
84    module: &EmitModule,
85    cm: &Lrc<SourceMap>,
86    config: &BuiltConfig,
87    unresolved_mark: Mark,
88    resolver: &Resolver,
89) -> (FnExpr, Vec<Param>) {
90    macro_rules! js_typeof {
91        ($test:expr =>! $type:expr) => {
92            Expr::Unary(UnaryExpr {
93                span: DUMMY_SP,
94                op: op!("typeof"),
95                arg: Box::new(Expr::from($test)),
96            })
97            .make_bin(op!("!=="), quote_str!($type))
98        };
99
100        ($test:expr => $type:expr) => {
101            Expr::Unary(UnaryExpr {
102                span: DUMMY_SP,
103                op: op!("typeof"),
104                arg: Box::new(Expr::from($test)),
105            })
106            .make_bin(op!("==="), quote_str!($type))
107        };
108    }
109
110    let module_ident = quote_ident!(SyntaxContext::empty().apply_mark(unresolved_mark), "module");
111    let require = quote_ident!(
112        SyntaxContext::empty().apply_mark(unresolved_mark),
113        "require"
114    );
115    let define = quote_ident!(SyntaxContext::empty().apply_mark(unresolved_mark), "define");
116    let global_this = quote_ident!(
117        SyntaxContext::empty().apply_mark(unresolved_mark),
118        "globalThis"
119    );
120    let js_self = quote_ident!(SyntaxContext::empty().apply_mark(unresolved_mark), "self");
121
122    let global = private_ident!("global");
123    let factory = private_ident!("factory");
124
125    let module_exports = module_ident.clone().make_member(quote_ident!("exports"));
126    let define_amd = define.clone().make_member(quote_ident!("amd"));
127
128    let mut cjs_args = Vec::new();
129    let mut amd_dep_list = Vec::new();
130    let mut browser_args = Vec::new();
131
132    let mut factory_params = Vec::new();
133
134    if !module.is_export_assign {
135        if let Some(exports) = &module.exports {
136            cjs_args.push(quote_ident!("exports").as_arg());
137            amd_dep_list.push(Some(quote_str!("exports").as_arg()));
138            browser_args.push(
139                ObjectLit::default()
140                    .make_assign_to(
141                        op!("="),
142                        module_global_export_member(module.span, cm, config, &global).into(),
143                    )
144                    .as_arg(),
145            );
146            factory_params.push(exports.clone().into());
147        }
148    }
149
150    module.deps.iter().for_each(|(ident, src_path, src_span)| {
151        let src_path = match resolver {
152            Resolver::Real { resolver, base } => resolver
153                .resolve_import(base, src_path)
154                .with_context(|| format!("failed to resolve `{src_path}`"))
155                .unwrap(),
156            Resolver::Default => src_path.clone(),
157        };
158
159        cjs_args.push(
160            require
161                .clone()
162                .as_call(
163                    DUMMY_SP,
164                    vec![quote_str!(src_span.0, src_path.clone()).as_arg()],
165                )
166                .as_arg(),
167        );
168        amd_dep_list.push(Some(quote_str!(src_span.0, src_path.clone()).as_arg()));
169
170        let global_dep = {
171            let dep_name = config.global_name(&src_path);
172            let global = global.clone();
173            if is_valid_prop_ident(&dep_name) {
174                global.make_member(quote_ident!(dep_name))
175            } else {
176                global.computed_member(quote_str!(dep_name))
177            }
178        };
179        browser_args.push(global_dep.as_arg());
180        factory_params.push(ident.clone().into());
181    });
182
183    let cjs_if_test = js_typeof!(module_ident.clone() => "object")
184        .make_bin(op!("&&"), js_typeof!(module_exports.clone() => "object"));
185    let mut cjs_if_body = factory.clone().as_call(DUMMY_SP, cjs_args);
186    if module.is_export_assign {
187        cjs_if_body = cjs_if_body.make_assign_to(op!("="), module_exports.clone().into());
188    }
189
190    let amd_if_test = js_typeof!(define.clone() => "function").make_bin(op!("&&"), define_amd);
191    let amd_if_body = define.as_call(
192        DUMMY_SP,
193        vec![
194            ArrayLit {
195                span: DUMMY_SP,
196                elems: amd_dep_list,
197            }
198            .as_arg(),
199            factory.clone().as_arg(),
200        ],
201    );
202
203    let browser_if_test = CondExpr {
204        span: DUMMY_SP,
205        test: Box::new(js_typeof!(global_this.clone() =>! "undefined")),
206        cons: Box::new(global_this.into()),
207        alt: Box::new(global.clone().make_bin(op!("||"), js_self)),
208    }
209    .make_assign_to(op!("="), global.clone().into());
210
211    let mut browser_if_body = factory.clone().as_call(DUMMY_SP, browser_args);
212    if module.is_export_assign {
213        browser_if_body = browser_if_body.make_assign_to(
214            op!("="),
215            module_global_export_member(module.span, cm, config, &global).into(),
216        );
217    }
218
219    let browser_if = if_stmt(browser_if_test, browser_if_body, None);
220    let amd_if = if_stmt(amd_if_test, amd_if_body, Some(browser_if));
221    let adapter_body = FunctionBody {
222        span: DUMMY_SP,
223        stmts: vec![if_stmt(cjs_if_test, cjs_if_body, Some(amd_if))],
224    };
225
226    let adapter_fn_expr = Function {
227        params: vec![global.into(), factory.into()],
228        span: DUMMY_SP,
229        body: Some(adapter_body),
230        is_generator: false,
231        is_async: false,
232        ..Default::default()
233    }
234    .into();
235
236    (adapter_fn_expr, factory_params)
237}

239fn if_stmt(test: Expr, body: Expr, alt: Option<Stmt>) -> Stmt {
240    IfStmt {
241        span: DUMMY_SP,
242        test: Box::new(test),
243        cons: Box::new(body.into_stmt()),
244        alt: alt.map(Box::new),
245    }
246    .into()
247}

249/// Returns the browser fallback global member for the current module.
250fn module_global_export_member(
251    module_span: Span,
252    cm: &Lrc<SourceMap>,
253    config: &BuiltConfig,
254    global: &Ident,
255) -> MemberExpr {
256    let filename = cm.span_to_filename(module_span);
257    let exported_name = config.determine_export_name(filename);
258
259    global.clone().make_member(exported_name.into())
260}
