23// Get high half of the 128-bit result of multiplication of two 64-bit unsigned
24// integers.
25pub(crate) fn umul128_upper64(x: u64, y: u64) -> u64 {
26    let p = u128::from(x) * u128::from(y);
27    (p >> 64) as u64
28}
29
30// Get upper 128-bits of multiplication of a 64-bit unsigned integer and a
31// 128-bit unsigned integer.
32pub(crate) fn umul192_upper128(x: u64, y: u128) -> u128 {
33    let mut r = u128::from(x) * u128::from(y.high());
34    r += u128::from(umul128_upper64(x, y.low()));
35    r
36}
37
38// Get lower 128-bits of multiplication of a 64-bit unsigned integer and a
39// 128-bit unsigned integer.
40pub(crate) fn umul192_lower128(x: u64, y: u128) -> u128 {
41    let high = x.wrapping_mul(y.high());
42    let high_low = u128::from(x) * u128::from(y.low());
43    (u128::from(high) << 64).wrapping_add(high_low)
44}
