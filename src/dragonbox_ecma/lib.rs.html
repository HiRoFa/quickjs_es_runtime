109// IEEE754-binary64
110const TOTAL_BITS: usize = 64;
111const SIGNIFICAND_BITS: usize = 52;
112const EXPONENT_BITS: usize = 11;
113const MIN_EXPONENT: i32 = -1022;
114const MAX_EXPONENT: i32 = 1023;
115const EXPONENT_BIAS: i32 = -1023;

121// Defines an unsigned integer type that is large enough
122// to carry a variable of type f64.
123// Most of the operations will be done on this integer type.
124type CarrierUint = u64;
125
126// Number of bits in the above unsigned integer type.
127const CARRIER_BITS: usize = 64;
128
129// Specifies the signed integer type to hold the exponent of Float.
130type ExponentInt = i32;
131
132// Extract exponent bits from a bit pattern. The result must be aligned to the
133// LSB so that there is no additional zero paddings on the right. This function
134// does not do bias adjustment.
135const fn extract_exponent_bits(u: CarrierUint) -> ExponentInt {
136    (u >> SIGNIFICAND_BITS) as ExponentInt & (((1 as ExponentInt) << EXPONENT_BITS) - 1)
137}
138
139// Remove the exponent bits and extract significand bits together with the sign
140// bit.
141const fn remove_exponent_bits(u: CarrierUint) -> CarrierUint {
142    u & !(((1 << EXPONENT_BITS) - 1) << SIGNIFICAND_BITS)
143}
144
145// Shift the obtained signed significand bits to the left by 1 to remove the
146// sign bit.
147const fn remove_sign_bit_and_shift(u: CarrierUint) -> CarrierUint {
148    (u << 1) & ((((1 << (TOTAL_BITS - 1)) - 1) << 1) | 1)
149}
150
151const fn is_nonzero(u: CarrierUint) -> bool {
152    (u & ((1 << (SIGNIFICAND_BITS + EXPONENT_BITS)) - 1)) != 0
153}
154
155const fn is_positive(u: CarrierUint) -> bool {
156    u < (1 << (SIGNIFICAND_BITS + EXPONENT_BITS))
157}
158
159const fn is_negative(u: CarrierUint) -> bool {
160    !is_positive(u)
161}
162
163const fn has_even_significand_bits(s: CarrierUint) -> bool {
164    s % 2 == 0
165}

167const fn compute_power32<const K: u32>(a: u32) -> u32 {
168    let mut p = 1;
169    let mut i = 0;
170    while i < K {
171        p *= a;
172        i += 1;
173    }
174    p
175}

177const fn compute_power64<const K: u32>(a: u64) -> u64 {
178    let mut p = 1;
179    let mut i = 0;
180    while i < K {
181        p *= a;
182        i += 1;
183    }
184    p

187const fn count_factors<const A: usize>(mut n: usize) -> u32 {
188    const {
189        assert!(A > 1);
190    }
191    let mut c = 0;
192    while n % A == 0 {
193        n /= A;
194        c += 1;
195    }
196    c
197}

205    { log::FLOOR_LOG10_POW2_MIN_EXPONENT },
206    { log::FLOOR_LOG10_POW2_MAX_EXPONENT },

235const MIN_K: i32 = min(
236    -log::floor_log10_pow2_minus_log10_4_over_3::<
237        { log::FLOOR_LOG10_POW2_MINUS_LOG10_4_OVER_3_MIN_EXPONENT },
238        { log::FLOOR_LOG10_POW2_MINUS_LOG10_4_OVER_3_MAX_EXPONENT },
239    >(MAX_EXPONENT - SIGNIFICAND_BITS as i32),
240    -log::floor_log10_pow2::<
241        { log::FLOOR_LOG10_POW2_MIN_EXPONENT },
242        { log::FLOOR_LOG10_POW2_MAX_EXPONENT },
243    >(MAX_EXPONENT - SIGNIFICAND_BITS as i32)
244        + KAPPA as i32,
245);
246const _: () = assert!(MIN_K >= cache::MIN_K);
247
248// We do invoke shorter_interval_case for exponent == min_exponent case, so we
249// should not add 1 here.
250const MAX_K: i32 = max(
251    -log::floor_log10_pow2_minus_log10_4_over_3::<
252        { log::FLOOR_LOG10_POW2_MINUS_LOG10_4_OVER_3_MIN_EXPONENT },
253        { log::FLOOR_LOG10_POW2_MINUS_LOG10_4_OVER_3_MAX_EXPONENT },
254    >(MIN_EXPONENT - SIGNIFICAND_BITS as i32 /* + 1*/),
255    -log::floor_log10_pow2::<
256        { log::FLOOR_LOG10_POW2_MIN_EXPONENT },
257        { log::FLOOR_LOG10_POW2_MAX_EXPONENT },
258    >(MIN_EXPONENT - SIGNIFICAND_BITS as i32)
259        + KAPPA as i32,
260);
261const _: () = assert!(MAX_K <= cache::MAX_K);

263struct ComputeMulResult {
264    integer_part: CarrierUint,
265    is_integer: bool,

268fn compute_mul(u: CarrierUint, cache: &cache::EntryType) -> ComputeMulResult {
269    let r = wuint::umul192_upper128(u, *cache);
270    ComputeMulResult {
271        integer_part: r.high(),
272        is_integer: r.low() == 0,
273    }
274}

276fn compute_delta(cache: &cache::EntryType, beta: i32) -> u32 {
277    (cache.high() >> ((CARRIER_BITS - 1) as i32 - beta)) as u32
278}

280struct ComputeMulParityResult {
281    parity: bool,
282    is_integer: bool,
283}
284
285fn compute_mul_parity(
286    two_f: CarrierUint,
287    cache: &cache::EntryType,
288    beta: i32,
289) -> ComputeMulParityResult {
290    debug_assert!(beta >= 1);
291    debug_assert!(beta < 64);
292
293    let r = wuint::umul192_lower128(two_f, *cache);
294    ComputeMulParityResult {
295        parity: ((r.high() >> (64 - beta)) & 1) != 0,
296        is_integer: ((r.high() << beta) | (r.low() >> (64 - beta))) == 0,
297    }
298}

300fn compute_left_endpoint_for_shorter_interval_case(
301    cache: &cache::EntryType,
302    beta: i32,
303) -> CarrierUint {
304    (cache.high() - (cache.high() >> (SIGNIFICAND_BITS + 2)))
305        >> ((CARRIER_BITS - SIGNIFICAND_BITS - 1) as i32 - beta)
306}
307
308fn compute_right_endpoint_for_shorter_interval_case(
309    cache: &cache::EntryType,
310    beta: i32,
311) -> CarrierUint {
312    (cache.high() + (cache.high() >> (SIGNIFICAND_BITS + 1)))
313        >> ((CARRIER_BITS - SIGNIFICAND_BITS - 1) as i32 - beta)
314}
315
316fn compute_round_up_for_shorter_interval_case(cache: &cache::EntryType, beta: i32) -> CarrierUint {
317    (cache.high() >> ((CARRIER_BITS - SIGNIFICAND_BITS - 2) as i32 - beta)).div_ceil(2)
318}

320const fn floor_log2(mut n: u64) -> i32 {
321    let mut count = -1;
322    while n != 0 {
323        count += 1;
324        n >>= 1;
325    }
326    count
327}

329fn is_left_endpoint_integer_shorter_interval(exponent: i32) -> bool {
330    const CASE_SHORTER_INTERVAL_LEFT_ENDPOINT_LOWER_THRESHOLD: i32 = 2;
331    const CASE_SHORTER_INTERVAL_LEFT_ENDPOINT_UPPER_THRESHOLD: i32 = 2 + floor_log2(
332        compute_power64::<
333            {
334                count_factors::<5>((((1 as CarrierUint) << (SIGNIFICAND_BITS + 2)) - 1) as usize)
335                    + 1
336            },
337        >(10)
338            / 3,
339    );
340    (CASE_SHORTER_INTERVAL_LEFT_ENDPOINT_LOWER_THRESHOLD
341        ..=CASE_SHORTER_INTERVAL_LEFT_ENDPOINT_UPPER_THRESHOLD)
342        .contains(&exponent)
343}

345// The main algorithm assumes the input is a normal/subnormal finite number.
346fn compute_nearest(signed_significand_bits: CarrierUint, exponent_bits: ExponentInt) -> Decimal {
347    let mut two_fc = remove_sign_bit_and_shift(signed_significand_bits);
348    let mut binary_exponent = exponent_bits;
349
350    // Is the input a normal number?
351    if binary_exponent != 0 {
352        binary_exponent += EXPONENT_BIAS - SIGNIFICAND_BITS as i32;
353
354        // Shorter interval case; proceed like Schubfach. One might think this
355        // condition is wrong, since when exponent_bits == 1 and two_fc == 0,
356        // the interval is actually regular. However, it turns out that this
357        // seemingly wrong condition is actually fine, because the end result is
358        // anyway the same.
359        //
360        // [binary32]
361        // (fc-1/2) * 2^e = 1.175'494'28... * 10^-38
362        // (fc-1/4) * 2^e = 1.175'494'31... * 10^-38
363        //    fc    * 2^e = 1.175'494'35... * 10^-38
364        // (fc+1/2) * 2^e = 1.175'494'42... * 10^-38
365        //
366        // Hence, shorter_interval_case will return 1.175'494'4 * 10^-38.
367        // 1.175'494'3 * 10^-38 is also a correct shortest representation that
368        // will be rejected if we assume shorter interval, but 1.175'494'4 *
369        // 10^-38 is closer to the true value so it doesn't matter.
370        //
371        // [binary64]
372        // (fc-1/2) * 2^e = 2.225'073'858'507'201'13... * 10^-308
373        // (fc-1/4) * 2^e = 2.225'073'858'507'201'25... * 10^-308
374        //    fc    * 2^e = 2.225'073'858'507'201'38... * 10^-308
375        // (fc+1/2) * 2^e = 2.225'073'858'507'201'63... * 10^-308
376        //
377        // Hence, shorter_interval_case will return 2.225'073'858'507'201'4 * 10^-308.
378        // This is indeed of the shortest length, and it is the unique one
379        // closest to the true value among valid representations of the same
380        // length.
381
382        // Shorter interval case.
383        if two_fc == 0 {
384            // Compute k and beta.
385            let minus_k = log::floor_log10_pow2_minus_log10_4_over_3::<
386                { MIN_EXPONENT - SIGNIFICAND_BITS as i32 },
387                { MAX_EXPONENT - SIGNIFICAND_BITS as i32 },
388            >(binary_exponent);
389            let beta = binary_exponent + log::floor_log2_pow10::<MIN_K, MAX_K>(-minus_k);
390
391            // Compute xi and zi.
392            let cache = unsafe { cache::get(-minus_k) };
393
394            let mut xi = compute_left_endpoint_for_shorter_interval_case(&cache, beta);
395            let zi = compute_right_endpoint_for_shorter_interval_case(&cache, beta);
396
397            // If the left endpoint is not an integer, increase it.
398            if !is_left_endpoint_integer_shorter_interval(binary_exponent) {
399                xi += 1;
400            }
401
402            // Try bigger divisor.
403            // zi is at most floor((f_c + 1/2) * 2^e * 10^k0).
404            // Substituting f_c = 2^p and k0 = -floor(log10(3 * 2^(e-2))), we get
405            // zi <= floor((2^(p+1) + 1) * 20/3) <= ceil((2^(p+1) + 1)/3) * 20.
406            // This computation does not overflow for any of the formats I care about.
407            let mut decimal_significand = div::divide_by_pow10::<
408                1,
409                { ((((1 << (SIGNIFICAND_BITS + 1)) + 1) / 3) + 1) * 20 },
410            >(zi);
411
412            // If succeed, remove trailing zeros if necessary and return.
413            if decimal_significand * 10 >= xi {
414                return Decimal {
415                    significand: decimal_significand,
416                    exponent: minus_k + 1,
417                };
418            }
419
420            // Otherwise, compute the round-up of y.
421            decimal_significand = compute_round_up_for_shorter_interval_case(&cache, beta);
422
423            // When tie occurs, choose one of them according to the rule.
424            const SHORTER_INTERVAL_TIE_LOWER_THRESHOLD: i32 =
425                -log::floor_log5_pow2_minus_log5_3(SIGNIFICAND_BITS as i32 + 4)
426                    - 2
427                    - SIGNIFICAND_BITS as i32;
428            const SHORTER_INTERVAL_TIE_UPPER_THRESHOLD: i32 =
429                -log::floor_log5_pow2(SIGNIFICAND_BITS as i32 + 2) - 2 - SIGNIFICAND_BITS as i32;
430            if policy::prefer_round_down(decimal_significand)
431                && binary_exponent >= SHORTER_INTERVAL_TIE_LOWER_THRESHOLD
432                && binary_exponent <= SHORTER_INTERVAL_TIE_UPPER_THRESHOLD
433            {
434                decimal_significand -= 1;
435            } else if decimal_significand < xi {
436                decimal_significand += 1;
437            }
438            return Decimal {
439                significand: decimal_significand,
440                exponent: minus_k,
441            };
442        }
443
444        two_fc |= 1 << (SIGNIFICAND_BITS + 1);
445    }
446    // Is the input a subnormal number?
447    else {
448        binary_exponent = MIN_EXPONENT - SIGNIFICAND_BITS as i32;
449    }
450
451    //////////////////////////////////////////////////////////////////////
452    // Step 1: Schubfach multiplier calculation.
453    //////////////////////////////////////////////////////////////////////
454
455    let has_even_significand_bits = has_even_significand_bits(signed_significand_bits);
456
457    // Compute k and beta.
458    let minus_k = log::floor_log10_pow2::<
459        { MIN_EXPONENT - SIGNIFICAND_BITS as i32 },
460        { MAX_EXPONENT - SIGNIFICAND_BITS as i32 },
461    >(binary_exponent)
462        - KAPPA as i32;
463    let cache = unsafe { cache::get(-minus_k) };
464    let beta = binary_exponent + log::floor_log2_pow10::<MIN_K, MAX_K>(-minus_k);
465
466    // Compute zi and deltai.
467    // 10^kappa <= deltai < 10^(kappa + 1)
468    let deltai = compute_delta(&cache, beta);
469    // For the case of binary32, the result of integer check is not correct for
470    // 29711844 * 2^-82
471    // = 6.1442653300000000008655037797566933477355632930994033813476... * 10^-18
472    // and 29711844 * 2^-81
473    // = 1.2288530660000000001731007559513386695471126586198806762695... * 10^-17,
474    // and they are the unique counterexamples. However, since 29711844 is even,
475    // this does not cause any problem for the endpoints calculations; it can
476    // only cause a problem when we need to perform integer check for the
477    // center. Fortunately, with these inputs, that branch is never executed, so
478    // we are fine.
479    let z_result = compute_mul((two_fc | 1) << beta, &cache);
480
481    //////////////////////////////////////////////////////////////////////
482    // Step 2: Try larger divisor; remove trailing zeros if necessary.
483    //////////////////////////////////////////////////////////////////////
484
485    const BIG_DIVISOR: u32 = compute_power32::<{ KAPPA + 1 }>(10);
486    const SMALL_DIVISOR: u32 = compute_power32::<KAPPA>(10);
487
488    // Using an upper bound on zi, we might be able to optimize the division
489    // better than the compiler; we are computing zi / big_divisor here.
490    let mut decimal_significand = div::divide_by_pow10::<
491        { KAPPA + 1 },
492        { (1 << (SIGNIFICAND_BITS + 1)) * BIG_DIVISOR as u64 - 1 },
493    >(z_result.integer_part);
494    let mut r = (z_result.integer_part - u64::from(BIG_DIVISOR) * decimal_significand) as u32;
495
496    loop {
497        if r < deltai {
498            // Exclude the right endpoint if necessary.
499            if (r | u32::from(!z_result.is_integer) | u32::from(has_even_significand_bits)) == 0 {
500                decimal_significand -= 1;
501                r = BIG_DIVISOR;
502                break;
503            }
504        } else if r > deltai {
505            break;
506        } else {
507            // r == deltai; compare fractional parts.
508            let x_result = compute_mul_parity(two_fc - 1, &cache, beta);
509
510            if !(x_result.parity | (x_result.is_integer & has_even_significand_bits)) {
511                break;
512            }
513        }
514
515        return Decimal {
516            significand: decimal_significand,
517            exponent: minus_k + KAPPA as i32 + 1,
518        };
519    }
520
521    //////////////////////////////////////////////////////////////////////
522    // Step 3: Find the significand with the smaller divisor.
523    //////////////////////////////////////////////////////////////////////
524
525    decimal_significand *= 10;
526
527    // delta is equal to 10^(kappa + elog10(2) - floor(elog10(2))), so dist
528    // cannot be larger than r.
529    let mut dist = r - (deltai / 2) + (SMALL_DIVISOR / 2);
530    let approx_y_parity = ((dist ^ (SMALL_DIVISOR / 2)) & 1) != 0;
531
532    // Is dist divisible by 10^kappa?
533    let divisible_by_small_divisor = div::check_divisibility_and_divide_by_pow10(&mut dist);
534
535    // Add dist / 10^kappa to the significand.
536    decimal_significand += CarrierUint::from(dist);
537
538    if divisible_by_small_divisor {
539        // Check z^(f) >= epsilon^(f).
540        // We have either yi == zi - epsiloni or yi == (zi - epsiloni) - 1,
541        // where yi == zi - epsiloni if and only if z^(f) >= epsilon^(f).
542        // Since there are only 2 possibilities, we only need to care about the parity.
543        // Also, zi and r should have the same parity since the divisor
544        // is an even number.
545        let y_result = compute_mul_parity(two_fc, &cache, beta);
546        if y_result.parity != approx_y_parity {
547            decimal_significand -= 1;
548        } else {
549            // If z^(f) >= epsilon^(f), we might have a tie
550            // when z^(f) == epsilon^(f), or equivalently, when y is an integer.
551            // For tie-to-up case, we can just choose the upper one.
552            if policy::prefer_round_down(decimal_significand) & y_result.is_integer {
553                decimal_significand -= 1;
554            }
555        }
556    }
557
558    Decimal {
559        significand: decimal_significand,
560        exponent: minus_k + KAPPA as i32,
561    }
562}

564#[cfg(not(feature = "ecma"))]
565fn to_decimal(x: f64) -> Decimal {
566    let br = x.to_bits();
567    let exponent_bits = extract_exponent_bits(br);
568    let s = remove_exponent_bits(br);
569
570    compute_nearest(s, exponent_bits)
571}

573#[cfg(feature = "ecma")]
574fn to_decimal(x: f64) -> Decimal {
575    let br = x.to_bits();
576    let exponent_bits = extract_exponent_bits(br);
577    let s = remove_exponent_bits(br);
578
579    let mut decimal = compute_nearest(s, exponent_bits);
580    policy::remove_trailing_zeros(&mut decimal.significand, &mut decimal.exponent);
581    decimal
582}
