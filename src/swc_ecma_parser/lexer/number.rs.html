13pub(super) fn parse_integer<const RADIX: u8>(s: &str) -> f64 {
14    debug_assert!(matches!(RADIX, 2 | 8 | 10 | 16));
15    debug_assert!(!s.is_empty());
16    debug_assert!(!s.contains('_'));
17
18    if RADIX == 10 {
19        parse_integer_from_dec(s)
20    } else {
21        parse_integer_radix::<RADIX>(s)
22    }
23}

25fn parse_integer_radix<const RADIX: u8>(s: &str) -> f64 {
26    debug_assert!(s.chars().all(|c| c.is_digit(RADIX as u32)));
27
28    // Accumulate integers that fit in u64 exactly, then round only once.
29    let max_fast_len = (u64::BITS / RADIX.trailing_zeros()) as usize;
30    if s.len() <= max_fast_len {
31        u64::from_str_radix(s, RADIX as u32).unwrap() as f64
32    } else {
33        parse_radix_slow::<RADIX>(s)
34    }
35}

37#[cold]
38#[inline(never)]
39fn parse_radix_slow<const RADIX: u8>(raw: &str) -> f64 {
40    debug_assert!(RADIX.is_power_of_two());
41
42    // These radices encode an integer as fixed-width bit groups. Retain its
43    // leading significand plus guard and sticky bits, then round exactly once.
44    const SIGNIFICAND_BITS: usize = f64::MANTISSA_DIGITS as usize;
45    const MAX_EXPONENT: usize = f64::MAX_EXP as usize - 1;
46    const EXPONENT_BIAS: u64 = 1023;
47
48    let raw = raw.trim_start_matches('0');
49    if raw.is_empty() {
50        return 0.0;
51    }
52
53    let bits_per_digit = RADIX.trailing_zeros() as usize;
54    // Bound the length before multiplication, including inputs larger than usize
55    // can represent in bits. One leading digit may use fewer bits than the rest.
56    if raw.len() > (MAX_EXPONENT + 1) / bits_per_digit + 1 {
57        return f64::INFINITY;
58    }
59    let first = raw.as_bytes()[0];
60    let first = if RADIX == 16 && first >= b'A' {
61        (first & 15) + 9
62    } else {
63        first & 15
64    };
65    let first_bits = (u8::BITS - first.leading_zeros()) as usize;
66    let bit_length = (raw.len() - 1) * bits_per_digit + first_bits;
67    if bit_length > MAX_EXPONENT + 1 {
68        return f64::INFINITY;
69    }
70    if bit_length <= SIGNIFICAND_BITS {
71        return u64::from_str_radix(raw, RADIX as u32).unwrap() as f64;
72    }
73
74    // Read enough whole digits for the significand and guard bit. Any extra
75    // bits in that prefix and all remaining digits contribute to the sticky bit.
76    let prefix_len = 1 + (SIGNIFICAND_BITS + 1 - first_bits).div_ceil(bits_per_digit);
77    let prefix = u64::from_str_radix(&raw[..prefix_len], RADIX as u32).unwrap();
78    let prefix_bits = first_bits + (prefix_len - 1) * bits_per_digit;
79    let discarded_bits = prefix_bits - SIGNIFICAND_BITS;
80    let mut significand = prefix >> discarded_bits;
81    let guard_mask = 1_u64 << (discarded_bits - 1);
82    let guard = prefix & guard_mask != 0;
83    let sticky = prefix & (guard_mask - 1) != 0
84        || raw.as_bytes()[prefix_len..]
85            .iter()
86            .any(|&byte| byte != b'0');
87
88    let round_up = guard && (sticky || significand & 1 != 0);
89    let mut exponent = bit_length - 1;
90    if round_up {
91        significand += 1;
92        if significand == 1 << SIGNIFICAND_BITS {
93            significand >>= 1;
94            exponent += 1;
95        }
96    }
97    if exponent > MAX_EXPONENT {
98        return f64::INFINITY;
99    }
100
101    let exponent = exponent as u64 + EXPONENT_BIAS;
102    let fraction_mask = (1_u64 << (SIGNIFICAND_BITS - 1)) - 1;
103    f64::from_bits((exponent << (SIGNIFICAND_BITS - 1)) | (significand & fraction_mask))
104}

106fn parse_integer_from_dec(s: &str) -> f64 {
107    debug_assert!(s.chars().all(|c| c.is_ascii_digit()));
108    const MAX_FAST_INT_LEN: usize = MAX_SAFE_INT.ilog10() as usize;
109    if s.len() > MAX_FAST_INT_LEN {
110        s.parse::<f64>().unwrap()
111    } else {
112        s.parse::<u64>().unwrap() as f64
113    }
