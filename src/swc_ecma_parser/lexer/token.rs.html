322impl<'a> Token {
323    #[inline(always)]
324    pub fn jsx_name(name: &str, lexer: &mut crate::Lexer) -> Self {
325        let name = lexer.atoms.atom(name);
326        lexer.set_token_value(Some(TokenValue::Word(name)));
327        Token::JSXName
328    }

330    #[inline(always)]

333    }

339    }

350        Token::Regex
351    }

356        Self::Num
357    }

362        Self::BigInt
363    }
364
365    #[inline(always)]
366    pub fn unknown_ident(value: Atom, lexer: &mut crate::Lexer<'a>) -> Self {
367        lexer.set_token_value(Some(TokenValue::Word(value)));
368        Token::Ident
369    }
370
371    #[inline(always)]
372    pub fn take_error<I: Tokens>(self, buffer: &mut Buffer<I>) -> Error {
373        buffer.expect_error_token_value()
374    }
375
376    #[inline(always)]
377    pub fn take_word<I: Tokens>(self, buffer: &Buffer<I>) -> Atom {
378        if self == Token::Ident {
379            let value = buffer.get_token_value();
380            let Some(TokenValue::Word(word)) = value else {
381                unreachable!("{:#?}", value)
382            };
383            return word.clone();
384        }
385
386        let span = buffer.cur.span;
387        let atom = Atom::new(buffer.iter.read_string(span));
388        atom
389    }
390
391    #[inline(always)]
392    pub fn take_known_ident<I: Tokens>(self, buffer: &Buffer<I>) -> Atom {
393        let span = buffer.cur.span;
394        let atom = Atom::new(buffer.iter.read_string(span));
395        atom
396    }
397
398    #[inline(always)]
399    pub fn take_unknown_ident_ref<'b, I: Tokens>(&'b self, buffer: &'b Buffer<I>) -> &'b Atom {
400        buffer.expect_word_token_value_ref()
401    }
402
403    #[inline(always)]
404    pub fn shebang(value: Atom, lexer: &mut Lexer) -> Self {
405        lexer.set_token_value(Some(TokenValue::Word(value)));
406        Token::Shebang
407    }
408
409    #[inline(always)]
410    pub fn take_shebang<I: Tokens>(self, buffer: &mut Buffer<I>) -> Atom {
411        buffer.expect_word_token_value()
412    }
413}

415impl std::fmt::Debug for Token {
416    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
417        let s = match self {
418            Token::Str => "<string literal>",
419            Token::Num => "<number literal>",
420            Token::BigInt => "<bigint literal>",
421            Token::Regex => "<regexp literal>",
422            Token::Template | Token::NoSubstitutionTemplateLiteral => "<template literal>",
423            Token::TemplateHead => "<template head `...${ >",
424            Token::TemplateMiddle => "<template middle ...${ >",
425            Token::TemplateTail => "<template tail ` >",
426            Token::JSXName => "<jsx name>",
427            Token::JSXText => "<jsx text>",
428            Token::Ident => "<identifier>",
429            Token::Error => "<error>",
430            _ => &self.to_string(),
431        };
432        f.write_str(s)
433    }

436impl Token {
437    pub const fn is_reserved(self, ctx: Context) -> bool {
438        match self {
439            Token::Let | Token::Static => ctx.contains(Context::Strict),
440            Token::Await => {
441                ctx.contains(Context::InAsync)
442                    || ctx.contains(Context::InStaticBlock)
443                    || ctx.contains(Context::Strict)
444            }
445            Token::Yield => ctx.contains(Context::InGenerator) || ctx.contains(Context::Strict),
446
447            Token::Null
448            | Token::True
449            | Token::False
450            | Token::Break
451            | Token::Case
452            | Token::Catch
453            | Token::Continue
454            | Token::Debugger
455            | Token::Default
456            | Token::Do
457            | Token::Export
458            | Token::Else
459            | Token::Finally
460            | Token::For
461            | Token::Function
462            | Token::If
463            | Token::Return
464            | Token::Switch
465            | Token::Throw
466            | Token::Try
467            | Token::Var
468            | Token::Const
469            | Token::While
470            | Token::With
471            | Token::New
472            | Token::This
473            | Token::Super
474            | Token::Class
475            | Token::Extends
476            | Token::Import
477            | Token::In
478            | Token::InstanceOf
479            | Token::TypeOf
480            | Token::Void
481            | Token::Delete => true,
482
483            // Future reserved word
484            Token::Enum => true,
485
486            Token::Implements
487            | Token::Package
488            | Token::Protected
489            | Token::Interface
490            | Token::Private
491            | Token::Public
492                if ctx.contains(Context::Strict) =>
493            {
494                true
495            }
496
497            _ => false,
498        }
499    }
500
501    #[inline(always)]
502    pub const fn is_keyword(self) -> bool {
503        let t = self as u8;
504        t >= Token::Await as u8 && t <= Token::Module as u8
505    }
506
507    #[inline(always)]
508    pub const fn is_known_ident(self) -> bool {
509        let t = self as u8;
510        t >= Token::Abstract as u8 && t <= Token::Target as u8
511    }
512
513    pub const fn is_word(self) -> bool {
514        matches!(
515            self,
516            Token::Null | Token::True | Token::False | Token::Ident
517        ) || self.is_known_ident()
518            || self.is_keyword()
519    }
520
521    #[inline(always)]
522    pub const fn is_bin_op(self) -> bool {
523        let t = self as u8;
524        (t >= Token::EqEq as u8 && t <= Token::NullishCoalescing as u8)
525            || (t >= Token::Plus as u8 && t <= Token::Ampersand as u8)
526    }
527
528    #[inline(always)]
529    pub const fn needs_unary_expr_prefix_parse(self) -> bool {
530        let t = self as u8;
531        (t >= Token::Bang as u8 && t <= Token::Minus as u8)
532            || (t >= Token::PlusPlus as u8 && t <= Token::MinusMinus as u8)
533            || matches!(
534                self,
535                Token::Lt | Token::Await | Token::Delete | Token::TypeOf | Token::Void
536            )
537    }

546            Token::LtEq => Some(swc_ecma_ast::BinaryOp::LtEq),
547            Token::Gt => Some(swc_ecma_ast::BinaryOp::Gt),

549            Token::LShift => Some(swc_ecma_ast::BinaryOp::LShift),
