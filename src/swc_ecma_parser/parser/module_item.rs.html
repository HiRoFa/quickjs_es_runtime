22    /// Parses attributes shared by imports and re-exports. Only the legacy
23    /// `assert` keyword has a no-line-terminator restriction before it.
24    fn parse_import_attributes(&mut self) -> PResult<Option<Box<ObjectLit>>> {
25        if !self.input().syntax().import_attributes() {
26            return Ok(None);
27        }
28
29        match self.input().cur() {
30            Token::With => {}
31            Token::Assert if !self.input().had_line_break_before_cur() => {}
32            _ => return Ok(None),
33        }
34        self.bump();
35
36        match self.parse_object_expr()? {
37            Expr::Object(v) => Ok(Some(Box::new(v))),
38            _ => unreachable!(),
39        }
40    }

42    /// Parses `from 'foo.js' with {};` or `from 'foo.js' assert {};`
43    fn parse_from_clause_and_semi(&mut self) -> PResult<(Box<Str>, Option<Box<ObjectLit>>)> {
44        expect!(self, Token::From);
45
46        let cur = self.input().cur();
47        let src = if cur == Token::Str {
48            Box::new(self.parse_str_lit())
49        } else {
50            unexpected!(self, "a string literal")
51        };
52        let with = self.parse_import_attributes()?;
53        self.expect_general_semi()?;
54        Ok((src, with))
55    }

57    fn parse_named_export_specifier(&mut self, type_only: bool) -> PResult<ExportNamedSpecifier> {
58        let start = self.cur_pos();
59
60        let mut is_type_only = false;
61
62        let orig_token = self.input().cur();
63        let orig = match self.parse_module_export_name()? {
64            ModuleExportName::Ident(orig_ident) => {
65                // Handle:
66                // `export { type xx }`
67                // `export { type xx as yy }`
68                // `export { type as }`
69                // `export { type as as }`
70                // `export { type as as as }`
71                // These names may also be string ModuleExportNames.
72                if self.syntax().typescript()
73                    && orig_token == Token::Type
74                    && (self.input().cur().is_word() || self.input().cur() == Token::Str)
75                {
76                    let possibly_orig_token = self.input().cur();
77                    let possibly_orig = self.parse_module_export_name()?;
78                    if possibly_orig_token == Token::As {
79                        // `export { type as }`
80                        if !(self.input().cur().is_word() || self.input().cur() == Token::Str) {
81                            if type_only {
82                                self.emit_err(orig_ident.span, SyntaxError::TS2207);
83                            }
84
85                            return Ok(ExportNamedSpecifier {
86                                span: self.span(start),
87                                orig: possibly_orig,
88                                exported: None,
89                                is_type_only: true,
90                            });
91                        }
92
93                        let maybe_as_token = self.input().cur();
94                        let maybe_as = self.parse_module_export_name()?;
95                        if maybe_as_token == Token::As {
96                            if self.input().cur().is_word() || self.input().cur() == Token::Str {
97                                // `export { type as as as }`
98                                // `export { type as as foo }`
99                                let exported = self.parse_module_export_name()?;
100
101                                if type_only {
102                                    self.emit_err(orig_ident.span, SyntaxError::TS2207);
103                                }
104
105                                return Ok(ExportNamedSpecifier {
106                                    span: self.span(start),
107                                    orig: possibly_orig,
108                                    exported: Some(exported),
109                                    is_type_only: true,
110                                });
111                            } else {
112                                // `export { type as as }`
113                                return Ok(ExportNamedSpecifier {
114                                    span: self.span(start),
115                                    orig: ModuleExportName::Ident(orig_ident),
116                                    exported: Some(maybe_as),
117                                    is_type_only: false,
118                                });
119                            }
120                        } else {
121                            // `export { type as xxx }`
122                            return Ok(ExportNamedSpecifier {
123                                span: self.span(start),
124                                orig: ModuleExportName::Ident(orig_ident),
125                                exported: Some(maybe_as),
126                                is_type_only: false,
127                            });
128                        }
129                    } else {
130                        // `export { type xx }`
131                        // `export { type xx as yy }`
132                        if type_only {
133                            self.emit_err(orig_ident.span, SyntaxError::TS2207);
134                        }
135
136                        is_type_only = true;
137                        possibly_orig
138                    }
139                } else {
140                    ModuleExportName::Ident(orig_ident)
141                }
142            }
143            module_export_name => module_export_name,
144        };
145
146        let exported = if self.input_mut().eat(Token::As) {
147            Some(self.parse_module_export_name()?)
148        } else {
149            None
150        };
151
152        Ok(ExportNamedSpecifier {
153            span: self.span(start),
154            orig,
155            exported,
156            is_type_only,
157        })
158    }

172    /// Parse `foo`, `foo2 as bar` in `import { foo, foo2 as bar }`
173    fn parse_import_specifier(&mut self, type_only: bool) -> PResult<ImportSpecifier> {
174        let start = self.cur_pos();
175        let orig_token = self.input().cur();
176        match self.parse_module_export_name()? {
177            ModuleExportName::Ident(mut orig_name) => {
178                let mut is_type_only = false;
179                // Handle:
180                // `import { type xx } from 'mod'`
181                // `import { type xx as yy } from 'mod'`
182                // `import { type "xx" as yy } from 'mod'`
183                // `import { type as } from 'mod'`
184                // `import { type as as } from 'mod'`
185                // `import { type as as as } from 'mod'`
186                if self.syntax().typescript()
187                    && (orig_token == Token::Type
188                        || (self.syntax().flow_types_enabled() && orig_token == Token::TypeOf))
189                {
190                    if self.syntax().flow_types_enabled()
191                        && orig_token == Token::TypeOf
192                        && self.input().cur().is_word()
193                    {
194                        let imported = self.parse_module_export_name()?;
195                        let local = if self.input_mut().eat(Token::As) {
196                            self.parse_binding_ident(false)?.into()
197                        } else {
198                            match &imported {
199                                ModuleExportName::Ident(i) => i.clone(),
200                                ModuleExportName::Str(s) => {
201                                    syntax_error!(
202                                        self,
203                                        s.span,
204                                        SyntaxError::ImportBindingIsString(
205                                            s.value.to_string_lossy().into()
206                                        )
207                                    )
208                                }
209                                #[cfg(swc_ast_unknown)]
210                                _ => unreachable!(),
211                            }
212                        };
213
214                        if self.input().syntax().flow() {
215                            if let ModuleExportName::Ident(imported) = &imported {
216                                self.emit_flow_reserved_type_name_error(
217                                    imported.span,
218                                    &imported.sym,
219                                );
220                            }
221                            self.emit_flow_reserved_type_name_error(local.span, &local.sym);
222                        }
223
224                        if type_only {
225                            self.emit_err(orig_name.span, SyntaxError::TS2206);
226                        }
227
228                        return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
229                            span: Span::new_with_checked(start, local.span.hi()),
230                            local,
231                            imported: Some(imported),
232                            is_type_only: true,
233                        }));
234                    }
235
236                    if self.syntax().flow_types_enabled()
237                        && orig_token == Token::TypeOf
238                        && !self.input().cur().is_word()
239                    {
240                        self.emit_err(orig_name.span, SyntaxError::TS1003);
241                    }
242
243                    // Handle `import { type "string" as foo }` case
244                    if self.input().cur() == Token::Str {
245                        let imported = self.parse_module_export_name()?;
246                        expect!(self, Token::As);
247                        let local: Ident = self.parse_binding_ident(false)?.into();
248
249                        if type_only {
250                            self.emit_err(orig_name.span, SyntaxError::TS2206);
251                        }
252
253                        return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
254                            span: Span::new_with_checked(start, local.span.hi()),
255                            local,
256                            imported: Some(imported),
257                            is_type_only: true,
258                        }));
259                    }
260
261                    if self.input().cur().is_word() {
262                        let possibly_orig_token = self.input().cur();
263                        let possibly_orig_name = self.parse_ident_name().map(Ident::from)?;
264                        if possibly_orig_token == Token::As {
265                            // `import { type as } from 'mod'`
266                            if !self.input().cur().is_word() {
267                                if self.ctx().is_reserved_word(&possibly_orig_name.sym) {
268                                    syntax_error!(
269                                        self,
270                                        possibly_orig_name.span,
271                                        SyntaxError::ReservedWordInImport
272                                    )
273                                }
274
275                                if type_only {
276                                    self.emit_err(orig_name.span, SyntaxError::TS2206);
277                                }
278
279                                return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
280                                    span: self.span(start),
281                                    local: possibly_orig_name,
282                                    imported: None,
283                                    is_type_only: true,
284                                }));
285                            }
286
287                            let maybe_as_token = self.input().cur();
288                            let maybe_as: Ident = self.parse_binding_ident(false)?.into();
289                            if maybe_as_token == Token::As {
290                                if self.input().cur().is_word() {
291                                    // `import { type as as as } from 'mod'`
292                                    // `import { type as as foo } from 'mod'`
293                                    let local: Ident = self.parse_binding_ident(false)?.into();
294
295                                    if type_only {
296                                        self.emit_err(orig_name.span, SyntaxError::TS2206);
297                                    }
298
299                                    return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
300                                        span: Span::new_with_checked(start, local.span.hi()),
301                                        local,
302                                        imported: Some(ModuleExportName::Ident(possibly_orig_name)),
303                                        is_type_only: true,
304                                    }));
305                                } else {
306                                    // `import { type as as } from 'mod'`
307                                    return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
308                                        span: Span::new_with_checked(start, maybe_as.span.hi()),
309                                        local: maybe_as,
310                                        imported: Some(ModuleExportName::Ident(orig_name)),
311                                        is_type_only: false,
312                                    }));
313                                }
314                            } else {
315                                // `import { type as xxx } from 'mod'`
316                                return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
317                                    span: Span::new_with_checked(start, maybe_as.span.hi()),
318                                    local: maybe_as,
319                                    imported: Some(ModuleExportName::Ident(orig_name)),
320                                    is_type_only: false,
321                                }));
322                            }
323                        } else {
324                            // `import { type xx } from 'mod'`
325                            // `import { type xx as yy } from 'mod'`
326                            if type_only {
327                                self.emit_err(orig_name.span, SyntaxError::TS2206);
328                            }
329
330                            orig_name = possibly_orig_name;
331                            is_type_only = true;
332                        }
333                    }
334                }
335
336                if self.input_mut().eat(Token::As) {
337                    let local: Ident = self.parse_binding_ident(false)?.into();
338                    if self.input().syntax().flow() && (type_only || is_type_only) {
339                        self.emit_flow_reserved_type_name_error(local.span, &local.sym);
340                    }
341                    return Ok(ImportSpecifier::Named(ImportNamedSpecifier {
342                        span: Span::new_with_checked(start, local.span.hi()),
343                        local,
344                        imported: Some(ModuleExportName::Ident(orig_name)),
345                        is_type_only,
346                    }));
347                }
348
349                // Handle difference between
350                //
351                // 'ImportedBinding'
352                // 'IdentifierName' as 'ImportedBinding'
353                if self.input().syntax().flow() && orig_name.sym == *"default" {
354                    self.emit_err(orig_name.span, SyntaxError::TS1003);
355                }
356                if self.ctx().is_reserved_word(&orig_name.sym)
357                    && !(self.input().syntax().flow() && (type_only || is_type_only))
358                {
359                    syntax_error!(self, orig_name.span, SyntaxError::ReservedWordInImport)
360                }
361
362                let local = orig_name;
363                if self.input().syntax().flow() && (type_only || is_type_only) {
364                    self.emit_flow_reserved_type_name_error(local.span, &local.sym);
365                }
366                Ok(ImportSpecifier::Named(ImportNamedSpecifier {
367                    span: self.span(start),
368                    local,
369                    imported: None,
370                    is_type_only,
371                }))
372            }
373            ModuleExportName::Str(orig_str) => {
374                if self.input_mut().eat(Token::As) {
375                    let local: Ident = self.parse_binding_ident(false)?.into();
376                    Ok(ImportSpecifier::Named(ImportNamedSpecifier {
377                        span: Span::new_with_checked(start, local.span.hi()),
378                        local,
379                        imported: Some(ModuleExportName::Str(orig_str)),
380                        is_type_only: false,
381                    }))
382                } else {
383                    syntax_error!(
384                        self,
385                        orig_str.span,
386                        SyntaxError::ImportBindingIsString(orig_str.value.to_string_lossy().into())
387                    )
388                }
389            }
390            #[cfg(swc_ast_unknown)]
391            _ => unreachable!(),
392        }
393    }

395    pub(crate) fn parse_export(&mut self, mut decorators: Vec<Decorator>) -> PResult<ModuleDecl> {
396        if !self.ctx().contains(Context::Module) && self.ctx().contains(Context::TopLevel) {
397            // Switch to module mode
398            let ctx = self.ctx() | Context::Module | Context::Strict;
399            self.set_ctx(ctx);
400        }
401
402        let start = self.cur_pos();
403        self.assert_and_bump(Token::Export);
404
405        let cur = self.input().cur();
406        if cur == Token::Eof {
407            return Err(self.eof_error());
408        }
409
410        let after_export_start = self.cur_pos();
411
412        // "export declare" is equivalent to just "export".
413        let declare = self.input().syntax().typescript() && self.input_mut().eat(Token::Declare);
414
415        if declare {
416            // TODO: Remove
417            if let Some(decl) = self.try_parse_ts_declare(after_export_start, decorators.clone())? {
418                return Ok(ExportDecl {
419                    span: self.span(start),
420                    decl,
421                }
422                .into());
423            }
424        }
425
426        if self.input().syntax().typescript() {
427            let cur = self.input().cur();
428            if cur.is_word() {
429                let sym = cur.take_word(&self.input);
430                // TODO: remove clone
431                if let Some(decl) = self.try_parse_ts_export_decl(decorators.clone(), sym) {
432                    return Ok(ExportDecl {
433                        span: self.span(start),
434                        decl,
435                    }
436                    .into());
437                }
438            }
439
440            if self.input_mut().eat(Token::Import) {
441                let is_type_only =
442                    self.input().is(Token::Type) && peek!(self).is_some_and(|p| p.is_word());
443
444                if is_type_only {
445                    self.assert_and_bump(Token::Type);
446                }
447
448                let id = self.parse_ident_name()?;
449
450                // export import A = B
451                return self
452                    .parse_ts_import_equals_decl(
453                        start,
454                        id.into(),
455                        /* is_export */ true,
456                        is_type_only,
457                    )
458                    .map(From::from);
459            }
460
461            if self.input_mut().eat(Token::Eq) {
462                // `export = x;`
463                let expr = self.parse_expr()?;
464                self.expect_general_semi()?;
465                return Ok(TsExportAssignment {
466                    span: self.span(start),
467                    expr,
468                }
469                .into());
470            }
471
472            if self.input_mut().eat(Token::As) {
473                // `export as namespace A;`
474                // See `parseNamespaceExportDeclaration` in TypeScript's own parser
475                expect!(self, Token::Namespace);
476                let id = self.parse_ident(false, false)?;
477                self.expect_general_semi()?;
478                return Ok(TsNamespaceExportDecl {
479                    span: self.span(start),
480                    id,
481                }
482                .into());
483            }
484        }
485
486        let ns_export_specifier_start = self.cur_pos();
487
488        let type_only = self.input().syntax().typescript() && self.input_mut().eat(Token::Type);
489
490        // Some("default") if default is exported from 'src'
491        let mut export_default = None;
492
493        if !type_only && self.input_mut().eat(Token::Default) {
494            if self.input().is(Token::At) {
495                let start = self.cur_pos();
496                let after_decorators = self.parse_decorators(false)?;
497
498                if !decorators.is_empty() && !self.input().syntax().flow_decorators() {
499                    syntax_error!(self, self.span(start), SyntaxError::TS8038);
500                }
501
502                if self.input().syntax().flow_decorators() {
503                    decorators.extend(after_decorators);
504                } else {
505                    decorators = after_decorators;
506                }
507            }
508
509            if self.input().syntax().typescript() {
510                if self.input().is(Token::Abstract)
511                    && peek!(self).is_some_and(|cur| cur == Token::Class)
512                    && !self.input_mut().has_linebreak_between_cur_and_peeked()
513                {
514                    let class_start = self.cur_pos();
515                    self.assert_and_bump(Token::Abstract);
516                    let cur = self.input().cur();
517                    if cur == Token::Error {
518                        let err = self.input_mut().expect_error_token_and_bump();
519                        return Err(err);
520                    }
521
522                    return self
523                        .parse_default_class(start, class_start, decorators, true)
524                        .map(ModuleDecl::ExportDefaultDecl);
525                }
526                if self.input().is(Token::Abstract)
527                    && peek!(self).is_some_and(|cur| cur == Token::Interface)
528                {
529                    self.emit_err(self.input().cur_span(), SyntaxError::TS1242);
530                    self.assert_and_bump(Token::Abstract);
531                }
532
533                if self.input().is(Token::Interface) {
534                    let interface_start = self.cur_pos();
535                    self.assert_and_bump(Token::Interface);
536                    let decl = self
537                        .parse_ts_interface_decl(interface_start)
538                        .map(DefaultDecl::from)?;
539                    return Ok(ExportDefaultDecl {
540                        span: self.span(start),
541                        decl,
542                    }
543                    .into());
544                }
545
546                if self.input().syntax().flow_components() && self.input().cur().is_word() {
547                    let sym = self.input().cur().take_word(&self.input);
548                    if sym == atom!("component") || sym == atom!("hook") {
549                        if let Some(decl) =
550                            self.try_parse_ts_export_decl(decorators.clone(), sym.clone())
551                        {
552                            return match decl {
553                                Decl::Fn(fn_decl) => Ok(ExportDefaultDecl {
554                                    span: self.span(start),
555                                    decl: DefaultDecl::Fn(FnExpr {
556                                        ident: Some(fn_decl.ident),
557                                        function: fn_decl.function,
558                                    }),
559                                }
560                                .into()),
561                                _ => Ok(ExportDecl {
562                                    span: self.span(start),
563                                    decl,
564                                }
565                                .into()),
566                            };
567                        }
568                    }
569                }
570            }
571
572            if self.input().is(Token::Class) {
573                let class_start = self.cur_pos();
574                let decl = self.parse_default_class(start, class_start, decorators, false)?;
575                return Ok(decl.into());
576            } else if self.input().is(Token::Async)
577                && peek!(self).is_some_and(|cur| cur == Token::Function)
578                && !self.input_mut().has_linebreak_between_cur_and_peeked()
579            {
580                let decl = self.parse_default_async_fn(start, decorators)?;
581                return Ok(decl.into());
582            } else if self.input().is(Token::Function) {
583                let decl = self.parse_default_fn(start, decorators)?;
584                return Ok(decl.into());
585            } else if self.input().syntax().typescript_allows_enum()
586                && self.input().is(Token::Enum)
587                && peek!(self).is_some_and(|peek| peek.is_word())
588                && !self.input_mut().has_linebreak_between_cur_and_peeked()
589            {
590                let enum_start = self.cur_pos();
591                self.assert_and_bump(Token::Enum);
592                let decl = self.parse_ts_enum_decl(enum_start, false)?;
593
594                // `export default enum` is valid in Flow, but SWC's default export decl AST
595                // variant does not model enum declarations.
596                return Ok(ExportDecl {
597                    span: self.span(start),
598                    decl: Decl::TsEnum(decl),
599                }
600                .into());
601            } else if self.input().syntax().export_default_from()
602                && ((self.input().is(Token::From)
603                    && peek!(self).is_some_and(|peek| peek == Token::Str))
604                    || (self.input().is(Token::Comma)
605                        && (peek!(self)
606                            .is_some_and(|peek| matches!(peek, Token::Asterisk | Token::LBrace)))))
607            {
608                export_default = Some(Ident::new_no_ctxt(
609                    atom!("default"),
610                    self.input().prev_span(),
611                ))
612            } else {
613                let expr = self.allow_in_expr(Self::parse_assignment_expr)?;
614                self.expect_general_semi()?;
615                return Ok(ExportDefaultExpr {
616                    span: self.span(start),
617                    expr,
618                }
619                .into());
620            }
621        }
622
623        if self.input().is(Token::At) {
624            let start = self.cur_pos();
625            let after_decorators = self.parse_decorators(false)?;
626
627            if !decorators.is_empty() && !self.input().syntax().flow_decorators() {
628                syntax_error!(self, self.span(start), SyntaxError::TS8038);
629            }
630
631            if self.input().syntax().flow_decorators() {
632                decorators.extend(after_decorators);
633            } else {
634                decorators = after_decorators;
635            }
636        }
637
638        let decl = if !type_only && self.input().is(Token::Class) {
639            let class_start = self.cur_pos();
640            self.parse_class_decl(start, class_start, decorators, false)?
641        } else if !type_only
642            && self.input().is(Token::Async)
643            && peek!(self).is_some_and(|cur| cur == Token::Function)
644            && !self.input_mut().has_linebreak_between_cur_and_peeked()
645        {
646            self.parse_async_fn_decl(decorators)?
647        } else if !type_only && self.input().is(Token::Function) {
648            self.parse_fn_decl(decorators)?
649        } else if !type_only
650            && self.input().syntax().typescript_allows_enum()
651            && self.input().is(Token::Enum)
652            && peek!(self).is_some_and(|peek| peek.is_word())
653            && !self.input_mut().has_linebreak_between_cur_and_peeked()
654        {
655            let enum_start = self.cur_pos();
656            self.assert_and_bump(Token::Enum);
657            self.parse_ts_enum_decl(enum_start, false)
658                .map(Decl::TsEnum)?
659        } else if !type_only
660            && self.input().syntax().typescript_allows_enum()
661            && self.input().is(Token::Const)
662            && peek!(self).is_some_and(|cur| cur == Token::Enum)
663        {
664            let enum_start = self.cur_pos();
665            self.assert_and_bump(Token::Const);
666            self.assert_and_bump(Token::Enum);
667            if self.input().syntax().flow() {
668                self.emit_err(self.span(enum_start), SyntaxError::TS1003);
669            }
670            return self
671                .parse_ts_enum_decl(enum_start, /* is_const */ true)
672                .map(Decl::from)
673                .map(|decl| {
674                    ExportDecl {
675                        span: self.span(start),
676                        decl,
677                    }
678                    .into()
679                });
680        } else if !type_only
681            && (self.input().is(Token::Var)
682                || self.input().is(Token::Const)
683                || (self.input().is(Token::Let))
684                    && peek!(self)
685                        .map(|t| t.follows_keyword_let())
686                        .unwrap_or(false))
687        {
688            self.parse_var_stmt(false).map(Decl::Var)?
689        } else {
690            // ```javascript
691            // export foo, * as bar, { baz } from "mod"; // *
692            // export      * as bar, { baz } from "mod"; // *
693            // export foo,           { baz } from "mod"; // *
694            // export foo, * as bar          from "mod"; // *
695            // export foo                    from "mod"; // *
696            // export      * as bar          from "mod"; //
697            // export                { baz } from "mod"; //
698            // export                { baz }           ; //
699            // export      *                 from "mod"; //
700            // ```
701
702            // export default
703            // export foo
704            let default = match export_default {
705                Some(default) => Some(default),
706                None => {
707                    if self.input().syntax().export_default_from() && self.input().cur().is_word() {
708                        Some(self.parse_ident(false, false)?)
709                    } else {
710                        None
711                    }
712                }
713            };
714
715            if default.is_none()
716                && self.input().is(Token::Asterisk)
717                && !peek!(self).is_some_and(|cur| cur == Token::As)
718            {
719                self.assert_and_bump(Token::Asterisk);
720
721                // improve error message for `export * from foo`
722                let (src, with) = self.parse_from_clause_and_semi()?;
723                return Ok(ExportAll {
724                    span: self.span(start),
725                    src,
726                    type_only,
727                    with,
728                }
729                .into());
730            }
731
732            let mut specifiers = Vec::new();
733
734            let mut has_default = false;
735            let mut has_ns = false;
736
737            if let Some(default) = default {
738                has_default = true;
739                specifiers.push(ExportSpecifier::Default(ExportDefaultSpecifier {
740                    exported: default,
741                }))
742            }
743
744            // export foo, * as bar
745            //           ^
746            if !specifiers.is_empty()
747                && self.input().is(Token::Comma)
748                && peek!(self).is_some_and(|cur| cur == Token::Asterisk)
749            {
750                self.assert_and_bump(Token::Comma);
751
752                has_ns = true;
753            }
754            // export     * as bar
755            //            ^
756            else if specifiers.is_empty() && self.input().is(Token::Asterisk) {
757                has_ns = true;
758            }
759
760            if has_ns {
761                self.assert_and_bump(Token::Asterisk);
762                expect!(self, Token::As);
763                let name = self.parse_module_export_name()?;
764                if self.input().syntax().flow() && type_only {
765                    let name_span = match &name {
766                        ModuleExportName::Ident(ident) => ident.span,
767                        ModuleExportName::Str(str_lit) => str_lit.span,
768                        #[cfg(swc_ast_unknown)]
769                        _ => unreachable!(),
770                    };
771                    self.emit_err(name_span, SyntaxError::TS1003);
772                }
773                specifiers.push(ExportSpecifier::Namespace(ExportNamespaceSpecifier {
774                    span: self.span(ns_export_specifier_start),
775                    name,
776                }));
777            }
778
779            if has_default || has_ns {
780                if self.input().is(Token::From) {
781                    let (src, with) = self.parse_from_clause_and_semi()?;
782                    return Ok(NamedExport {
783                        span: self.span(start),
784                        specifiers,
785                        src: Some(src),
786                        type_only,
787                        with,
788                    }
789                    .into());
790                } else if !self.input().syntax().export_default_from() {
791                    // emit error
792                    expect!(self, Token::From);
793                }
794
795                expect!(self, Token::Comma);
796            }
797
798            expect!(self, Token::LBrace);
799
800            while !self.input().is(Token::RBrace) {
801                let specifier = self.parse_named_export_specifier(type_only)?;
802                specifiers.push(ExportSpecifier::Named(specifier));
803
804                if self.input().is(Token::RBrace) {
805                    break;
806                } else {
807                    expect!(self, Token::Comma);
808                }
809            }
810            expect!(self, Token::RBrace);
811
812            let opt = if self.input().is(Token::From) {
813                Some(self.parse_from_clause_and_semi()?)
814            } else {
815                for s in &specifiers {
816                    match s {
817                        ExportSpecifier::Default(default) => {
818                            self.emit_err(
819                                default.exported.span,
820                                SyntaxError::ExportExpectFrom(default.exported.sym.clone()),
821                            );
822                        }
823                        ExportSpecifier::Namespace(namespace) => {
824                            let export_name = match &namespace.name {
825                                ModuleExportName::Ident(i) => i.sym.clone(),
826                                ModuleExportName::Str(s) => s.value.to_string_lossy().into(),
827                                #[cfg(swc_ast_unknown)]
828                                _ => unreachable!(),
829                            };
830                            self.emit_err(
831                                namespace.span,
832                                SyntaxError::ExportExpectFrom(export_name),
833                            );
834                        }
835                        ExportSpecifier::Named(named) => match &named.orig {
836                            ModuleExportName::Ident(id) if id.is_reserved() => {
837                                self.emit_err(
838                                    id.span,
839                                    SyntaxError::ExportExpectFrom(id.sym.clone()),
840                                );
841                            }
842                            ModuleExportName::Str(s) => {
843                                self.emit_err(s.span, SyntaxError::ExportBindingIsString);
844                            }
845                            _ => {}
846                        },
847                        #[cfg(swc_ast_unknown)]
848                        _ => unreachable!(),
849                    }
850                }
851
852                self.eat_general_semi();
853
854                None
855            };
856            let (src, with) = match opt {
857                Some(v) => (Some(v.0), v.1),
858                None => (None, None),
859            };
860            return Ok(NamedExport {
861                span: self.span(start),
862                specifiers,
863                src,
864                type_only,
865                with,
866            }
867            .into());
868        };
869
870        Ok(ExportDecl {
871            span: self.span(start),
872            decl,
873        }
874        .into())
875    }

877    #[inline]
878    fn enter_import_module_context(&mut self) {
879        if !self.ctx().contains(Context::Module) {
880            let ctx = self.ctx() | Context::Module | Context::Strict;
881            self.set_ctx(ctx);
882        }

885    pub(crate) fn parse_import(&mut self) -> PResult<ModuleItem> {
886        let start = self.cur_pos();
887
888        if peek!(self).is_some_and(|cur| cur == Token::Dot) {
889            let expr = self.parse_expr()?;
890
891            self.eat_general_semi();
892
893            return Ok(ExprStmt {
894                span: self.span(start),
895                expr,
896            }
897            .into());
898        }
899
900        if peek!(self).is_some_and(|cur| cur == Token::LParen) {
901            let expr = self.parse_expr()?;
902
903            self.eat_general_semi();
904
905            return Ok(ExprStmt {
906                span: self.span(start),
907                expr,
908            }
909            .into());
910        }
911
912        expect!(self, Token::Import);
913
914        // Handle import 'mod.js'
915        if self.input().cur() == Token::Str {
916            self.enter_import_module_context();
917            let src = Box::new(self.parse_str_lit());
918            let with = self.parse_import_attributes()?;
919            self.eat_general_semi();
920            return Ok(ImportDecl {
921                span: self.span(start),
922                src,
923                specifiers: Vec::new(),
924                type_only: false,
925                with,
926                phase: Default::default(),
927            }
928            .into());
929        }
930
931        let mut type_only = false;
932        let mut phase = ImportPhase::Evaluation;
933        let mut specifiers = Vec::with_capacity(4);
934        let mut flow_leading_import_type = false;
935
936        'import_maybe_ident: {
937            if self.is_ident_ref()
938                || (self.input().syntax().flow() && self.input().cur().is_word())
939                || (self.input().syntax().flow() && self.input().is(Token::TypeOf))
940                || (self.input().syntax().flow_types_enabled() && self.input().is(Token::TypeOf))
941            {
942                let local_token = self.input().cur();
943                let mut local =
944                    if self.input().syntax().flow_types_enabled() && local_token == Token::TypeOf {
945                        self.bump();
946                        type_only = true;
947                        if self.input().is(Token::LBrace) || self.input().is(Token::Asterisk) {
948                            break 'import_maybe_ident;
949                        }
950                        self.parse_imported_default_binding()?
951                    } else {
952                        self.parse_imported_default_binding()?
953                    };
954
955                if self.input().syntax().typescript() && local_token == Token::Type {
956                    let cur = self.input().cur();
957                    if cur == Token::LBrace || cur == Token::Asterisk {
958                        type_only = true;
959                        if self.input().syntax().flow() {
960                            flow_leading_import_type = true;
961                        }
962                        break 'import_maybe_ident;
963                    }
964
965                    if self.is_ident_ref()
966                        || (self.input().syntax().flow() && self.input().cur().is_word())
967                    {
968                        if !self.input().is(Token::From)
969                            || peek!(self).is_some_and(|cur| cur == Token::From)
970                        {
971                            type_only = true;
972                            if self.input().syntax().flow() {
973                                flow_leading_import_type = true;
974                                local = self.parse_ident_name().map(Ident::from)?;
975                            } else {
976                                local = self.parse_imported_default_binding()?;
977                            }
978                        } else if peek!(self).is_some_and(|cur| cur == Token::Eq) {
979                            type_only = true;
980                            local = self.parse_ident_name().map(From::from)?;
981                        }
982                    }
983                }
984
985                if self.input().syntax().flow() && type_only {
986                    self.emit_flow_reserved_type_name_error(local.span, &local.sym);
987                }
988
989                if self.input().syntax().typescript() && self.input().is(Token::Eq) {
990                    let decl = self.parse_ts_import_equals_decl(start, local, false, type_only)?;
991                    if matches!(&decl.module_ref, TsModuleRef::TsExternalModuleRef(..)) {
992                        self.enter_import_module_context();
993                    }
994                    return Ok(ModuleDecl::TsImportEquals(decl).into());
995                }
996
997                if matches!(&*local.sym, "source" | "defer") {
998                    let new_phase = match &*local.sym {
999                        "source" => ImportPhase::Source,
1000                        "defer" => ImportPhase::Defer,
1001                        _ => unreachable!(),
1002                    };
1003
1004                    let cur = self.input().cur();
1005                    if cur == Token::LBrace || cur == Token::Asterisk {
1006                        phase = new_phase;
1007                        break 'import_maybe_ident;
1008                    }
1009
1010                    if self.is_ident_ref() && !self.input().is(Token::From)
1011                        || peek!(self).is_some_and(|cur| cur == Token::From)
1012                    {
1013                        // For defer phase, we expect only namespace imports, so break here
1014                        // and let the subsequent code handle validation
1015                        if new_phase == ImportPhase::Defer {
1016                            break 'import_maybe_ident;
1017                        }
1018                        phase = new_phase;
1019                        local = self.parse_imported_default_binding()?;
1020                    }
1021                }
1022
1023                //TODO: Better error reporting
1024                if !self.input().is(Token::From) {
1025                    expect!(self, Token::Comma);
1026                }
1027                specifiers.push(ImportSpecifier::Default(ImportDefaultSpecifier {
1028                    span: local.span,
1029                    local,
1030                }));
1031            }
1032        }
1033
1034        self.enter_import_module_context();
1035
1036        {
1037            let import_spec_start = self.cur_pos();
1038            // Namespace imports are not allowed in source phase.
1039            if phase != ImportPhase::Source && self.input_mut().eat(Token::Asterisk) {
1040                if self.input().syntax().flow() && flow_leading_import_type {
1041                    self.emit_err(self.input().prev_span(), SyntaxError::TS1003);
1042                }
1043                expect!(self, Token::As);
1044                let local = self.parse_imported_binding()?;
1045                if self.input().syntax().flow() && type_only {
1046                    self.emit_flow_reserved_type_name_error(local.span, &local.sym);
1047                }
1048                specifiers.push(ImportSpecifier::Namespace(ImportStarAsSpecifier {
1049                    span: self.span(import_spec_start),
1050                    local,
1051                }));
1052                // Named imports are only allowed in evaluation phase.
1053            } else if phase == ImportPhase::Evaluation && self.input_mut().eat(Token::LBrace) {
1054                while !self.input().is(Token::RBrace) {
1055                    specifiers.push(self.parse_import_specifier(type_only)?);
1056
1057                    if self.input().is(Token::RBrace) {
1058                        break;
1059                    } else {
1060                        expect!(self, Token::Comma);
1061                    }
1062                }
1063                expect!(self, Token::RBrace);
1064            }
1065        }
1066
1067        let src = {
1068            expect!(self, Token::From);
1069            if self.input().cur() == Token::Str {
1070                Box::new(self.parse_str_lit())
1071            } else {
1072                unexpected!(self, "a string literal")
1073            }
1074        };
1075
1076        let with = self.parse_import_attributes()?;
1077
1078        self.expect_general_semi()?;
1079
1080        Ok(ImportDecl {
1081            span: self.span(start),
1082            specifiers,
1083            src,
1084            type_only,
1085            with,
1086            phase,
1087        }
1088        .into())
1089    }
1090}
1091
1092fn handle_import_export<I: Tokens>(
1093    p: &mut Parser<I>,
1094    decorators: Vec<Decorator>,
1095) -> PResult<ModuleItem> {
1096    if !p
1097        .ctx()
1098        .intersects(Context::TopLevel.union(Context::TsModuleBlock))
1099    {
1100        syntax_error!(p, SyntaxError::NonTopLevelImportExport);
1101    }
1102
1103    let decl = if p.input().is(Token::Import) {
1104        p.parse_import()?
1105    } else if p.input().is(Token::Export) {
1106        p.parse_export(decorators).map(ModuleItem::from)?
1107    } else {
1108        unreachable!(
1109            "handle_import_export should not be called if current token isn't import nor export"
1110        )
1111    };
1112
1113    Ok(decl)
1114}
