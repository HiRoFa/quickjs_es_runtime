38/// Compiles es2015 to es5.
39///
40/// # Parameters
41///
42/// ## `unresolved_mark`
43///
44/// Used to generate `require` calls.
45/// See the documentation of [regenerator](self::regenerator::regenerator) for
46/// more details.
47pub fn es2015<C>(unresolved_mark: Mark, comments: Option<C>, c: Config) -> impl Pass
48where
49    C: Comments + Clone,
50{
51    (
52        (
53            block_scoped_functions(),
54            template_literal(c.template_literal),
55            classes(c.classes),
56            new_target(),
57            spread(c.spread, unresolved_mark),
58        ),
59        // https://github.com/Microsoft/TypeScript/issues/5441
60        if !c.typescript {
61            Some(object_super())
62        } else {
63            None
64        },
65        function_name(),
66        for_of(c.for_of),
67        // Should come before parameters
68        // See: https://github.com/swc-project/swc/issues/1036
69        parameters(c.parameters, unresolved_mark),
70        (
71            arrow(unresolved_mark),
72            computed_properties(c.computed_props),
73            destructuring(c.destructuring),
74            block_scoping(unresolved_mark),
75            generator::generator(unresolved_mark, comments.clone()),
76        ),
77        swc_ecma_transformer::es2015_runtime_transforms(),
78    )
79}
