15/// Transform to optimize performance of literals.
16///
17///
18/// This transform converts pure object literals like
19///
20/// ```js
21/// {a: 1, b: 2}
22/// ```
23///
24/// to
25///
26/// ```js
27/// JSON.parse('{"a":1, "b"}')
28/// ```
29///
30/// # Conditions
31/// If any of the conditions below is matched, pure object literal is converter
32/// to `JSON.parse`
33///
34///   - Object literal is deeply nested (threshold: )
35///
36/// See https://github.com/swc-project/swc/issues/409
37pub fn json_parse(min_cost: usize) -> impl Pass {
38    visit_mut_pass(JsonParse { min_cost })
39}
40
41struct JsonParse {
42    pub min_cost: usize,
43}
44
45impl Parallel for JsonParse {
46    fn create(&self) -> Self {
47        JsonParse {
48            min_cost: self.min_cost,
49        }
50    }
51
52    fn merge(&mut self, _: Self) {}
53}
54
55impl Default for JsonParse {
56    fn default() -> Self {
57        JsonParse { min_cost: 1024 }
58    }
59}
60
61impl VisitMut for JsonParse {
62    noop_visit_mut_type!(fail);
63
64    /// Handles parent expressions before child expressions.
65    fn visit_mut_expr(&mut self, expr: &mut Expr) {
66        if self.min_cost == usize::MAX {
67            return;
68        }
69
70        let e = match expr {
71            Expr::Array(..) | Expr::Object(..) => {
72                let (is_lit, cost) = calc_literal_cost(&*expr, false);
73                if is_lit && cost >= self.min_cost {
74                    let span = expr.span();
75                    let value = jsonify(expr);
76
77                    *expr = CallExpr {
78                        span,
79                        callee: member_expr!(Default::default(), DUMMY_SP, JSON.parse).as_callee(),
80                        args: vec![Lit::Str(Str {
81                            span: DUMMY_SP,
82                            raw: None,
83                            value: value.into(),
84                        })
85                        .as_arg()],
86                        ..Default::default()
87                    }
88                    .into();
89                    return;
90                }
91
92                expr
93            }
94            _ => expr,
95        };
96
97        e.visit_mut_children_with(self)
98    }
99}

101/// Converts a Wtf8Atom to a JSON-safe string, escaping lone surrogates
102fn wtf8_to_json_string(value: &Wtf8Atom) -> String {
103    if let Some(s) = value.as_str() {
104        // Fast path: valid UTF-8
105        return s.to_string();
106    }
107
108    // Slow path: contains lone surrogates, need to escape them
109    let mut result = String::with_capacity(value.len());
110    for cp in value.as_wtf8().code_points() {
111        if let Some(ch) = cp.to_char() {
112            // Valid Rust char, push directly
113            result.push(ch);
114        } else {
115            // Lone surrogate - escape as \uXXXX
116            write!(&mut result, "\\u{:04X}", cp.to_u32()).unwrap();
117        }
118    }
119    result
120}

122/// Converts a finite ECMAScript number without using a saturating integer cast.
123fn json_number(value: f64) -> serde_json::Number {
124    let is_i64 = value.fract() == 0.0 && value >= i64::MIN as f64 && value < i64::MAX as f64;
125    let preserves_zero_sign = value != 0.0 || value.is_sign_positive();
126
127    if is_i64 && preserves_zero_sign {
128        return (value as i64).into();
129    }
130
131    serde_json::Number::from_f64(value)
132        .unwrap_or_else(|| unreachable!("non-finite numbers require raw JSON serialization"))
133}

135/// Converts a property name to the string key created by an object literal.
136fn json_key(key: &PropName) -> String {
137    match key {
138        PropName::Str(s) => wtf8_to_json_string(&s.value),
139        PropName::Ident(id) => id.sym.to_string(),
140        PropName::Num(n) => n.value.to_js_string(),
141        _ => unreachable!(),
142    }
143}

145/// Serializes a literal expression to JSON text while preserving ECMAScript
146/// numeric values that are not representable by `serde_json::Value`.
147fn jsonify(e: &Expr) -> String {
148    serde_json::to_string(&JsonLiteral(e))
149        .unwrap_or_else(|err| unreachable!("failed to serialize literal as JSON: {err}"))
150}

152/// Adapts an ECMAScript literal expression to Serde's data model.
153struct JsonLiteral<'a>(&'a Expr);
154
155impl Serialize for JsonLiteral<'_> {
156    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
157    where
158        S: Serializer,
159    {
160        match self.0 {
161            Expr::Object(obj) => {
162                let mut values = BTreeMap::new();
163                for prop in &obj.props {
164                    let PropOrSpread::Prop(prop) = prop else {
165                        unreachable!()
166                    };
167                    let Prop::KeyValue(prop) = &**prop else {
168                        unreachable!()
169                    };
170                    values.insert(json_key(&prop.key), &*prop.value);
171                }
172
173                let mut map = serializer.serialize_map(Some(values.len()))?;
174                for (key, value) in values {
175                    map.serialize_entry(&key, &JsonLiteral(value))?;
176                }
177                map.end()
178            }
179            Expr::Array(arr) => {
180                let mut seq = serializer.serialize_seq(Some(arr.elems.len()))?;
181                for value in &arr.elems {
182                    let value = value.as_ref().unwrap();
183                    seq.serialize_element(&JsonLiteral(value.expr.as_ref()))?;
184                }
185                seq.end()
186            }
187            Expr::Lit(Lit::Str(Str { value, .. })) => {
188                wtf8_to_json_string(value).serialize(serializer)
189            }
190            Expr::Lit(Lit::Num(Number { value, .. })) if value.is_infinite() => {
191                let value = if value.is_sign_positive() {
192                    "2e308"
193                } else {
194                    "-2e308"
195                };
196                RawValue::from_string(value.into())
197                    .unwrap()
198                    .serialize(serializer)
199            }
200            Expr::Lit(Lit::Num(Number { value, .. })) => json_number(*value).serialize(serializer),
201            Expr::Lit(Lit::Null(..)) => serializer.serialize_none(),
202            Expr::Lit(Lit::Bool(v)) => serializer.serialize_bool(v.value),
203            Expr::Tpl(Tpl { quasis, .. }) => {
204                let value = match quasis.first() {
205                    Some(TplElement {
206                        cooked: Some(value),
207                        ..
208                    }) => wtf8_to_json_string(value),
209                    _ => String::new(),
210                };
211                value.serialize(serializer)
212            }
213            _ => unreachable!("jsonify: Expr {:?} cannot be converted to json", self.0),
214        }
215    }
216}
